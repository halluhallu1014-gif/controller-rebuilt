    ConfigureList related, fraRelated, False
    Set button = fraOffices.Controls.Add("Forms.OptionButton.1", "OfficeAll", True)

Private Sub ApplyPageFonts()

    For Each page In pages.pages
        For Each control In page.Controls
            If TypeOf control Is MSForms.label Or TypeOf control Is MSForms.CommandButton Or _
               TypeOf control Is MSForms.CheckBox Or TypeOf control Is MSForms.TextBox Or TypeOf control Is MSForms.ComboBox Then
                control.Font.name = "Meiryo UI"
                control.Font.size = 10

        Next control

    lblLegacyDetailTitle.Font.size = 19: lblLegacyDetailTitle.Font.bold = True
    lblRelated.Font.size = 12: lblRelated.Font.bold = True
    lblCount.Font.size = 12: lblCount.Font.bold = True
    cmdSave.Font.bold = True: cmdNew.Font.bold = True
End Sub

Private Sub SearchNow()
    Dim button As MSForms.OptionButton, office As String
    On Error GoTo Failed
    For Each button In officeButtons
        If button.value And button.name <> "OfficeAll" Then office = button.caption
    Next button
    Set hits = ControllerFind(CStr(txtName.value), office, CBool(chkPending.value))
    ShowRows results, hits, True
    lblCount.caption = Format$(hits.count, "#,##0") & " 件"
    Exit Sub
Failed:
    txtNotice.value = Err.Description
End Sub

Private Sub ShowRows(ByVal List As ExtendedListBox, ByVal rows As Collection, ByVal isSearch As Boolean)
    Dim data() As Variant, Row As Long, record As Object
    List.Clear
    If rows.count = 0 Then Exit Sub
    If isSearch Then
        ReDim data(0 To rows.count - 1, 0 To 4)
        For Each record In rows
            data(Row, 0) = record("OfficeName")
            data(Row, 1) = record("PersonName")
            data(Row, 2) = record("TaxPeriod")
            data(Row, 3) = record("ApplicationType")
            data(Row, 4) = record("ProcessingCategory")
            Row = Row + 1
        Next record
    Else
        ReDim data(0 To rows.count - 1, 0 To 3)
        For Each record In rows
            data(Row, 0) = record("TaxPeriod")
            data(Row, 1) = CStr(record("CommonNumber")) & "-" & CStr(record("BranchNumber"))
            data(Row, 2) = record("ApplicationType")
            data(Row, 3) = record("ProcessingCategory")
            Row = Row + 1
        Next record
    End If
    List.List = data
End Sub

Public Sub SelectSearchRow(ByVal index As Long)
    If hits Is Nothing Then Exit Sub
    If index < 1 Or index > hits.count Then Exit Sub
    If Not LeaveDraft() Then Exit Sub
    results.ListIndex = index - 1
    ShowRecord hits(index)
End Sub

Private Sub ShowRecord(ByVal identity As Object)
    Dim master As Object, snapshot As Object, warning As String, other As Object, index As Long, key As String
    On Error GoTo Failed
    Set master = ControllerMasterRecord(identity)
    On Error Resume Next
    Set snapshot = ControllerBranchRecord(master)
    warning = Err.Description
    Err.Clear
    On Error GoTo Failed
    Set current = master
    Set peer = snapshot
    isNew = False
    Set relatedRows = ControllerRelated(current)
    ShowRows related, relatedRows, False
    key = LedgerRecordKey(current, fields)
    For Each other In relatedRows
        If LedgerRecordKey(other, fields) = key Then related.ListIndex = index
        index = index + 1
    Next other
    lblDetail.caption = CStr(current("PersonName")) & "   /   " & CStr(current("OfficeName"))
    lblRelated.caption = "関連事案  " & relatedRows.count & " 件"
    lblDetailNote.caption = vbNullString
    If ControllerLocked(current) Or ControllerLocked(peer) Then
        lblDetailNote.caption = "署引継済みのため閲覧のみ"
    ElseIf peer Is Nothing Then
        lblDetailNote.caption = "対象署の読込み不可。閲覧のみ"
        txtNotice.value = warning
    End If
    FillEditors
    pages.value = 1
    Exit Sub
Failed:
    MsgBox Err.Description, vbExclamation, "事案を開く"
End Sub

Private Sub FillEditors()
    Dim fieldId As Variant, definition As Object, box As MSForms.TextBox, writable As Boolean
    For Each fieldId In editors.Keys
        Set box = editors(fieldId)
        If fields.Exists(fieldId) Then
            Set definition = fields(fieldId)
            writable = isNew
            If isNew Then
                box.value = vbNullString
            Else
                box.value = ControllerDisplay(current(fieldId), definition)
                writable = Not ControllerLocked(current) And Not ControllerLocked(peer) And Not peer Is Nothing And definition("Editable") = ChrW(&H25CB)
            End If
            box.Locked = Not writable
        Else
            box.value = "未割当"
            box.Locked = True
        End If
        If dateButtons.Exists(fieldId) Then dateButtons(fieldId).Enabled = Not box.Locked
        If box.Locked Then
            box.BackColor = RGB(241, 245, 250)
        Else
            box.BackColor = vbWhite
        End If
    Next fieldId
    fraFields.ScrollTop = 0
    If isNew Then cmdSave.caption = "登録・保存" Else cmdSave.caption = "更新・保存"
    cmdSave.Enabled = isNew Or (Not ControllerLocked(current) And Not ControllerLocked(peer) And Not peer Is Nothing)
    cmdSource.Enabled = Not isNew
    cmdPrev.Enabled = Not isNew
    cmdNext.Enabled = Not isNew
End Sub

Private Function HasDraft() As Boolean
    Dim box As MSForms.TextBox, fieldId As Variant, definition As Object, oldText As String
    If pages.value <> 1 Then Exit Function
    For Each fieldId In editors.Keys
        Set box = editors(fieldId)
        If Not box.Locked Then
            oldText = vbNullString
            If Not isNew Then
                Set definition = fields(fieldId)
                oldText = ControllerDisplay(current(fieldId), definition)
            End If
            If CStr(box.value) <> oldText Then
                HasDraft = True
                Exit Function
            End If
        End If
    Next fieldId
End Function

Private Function LeaveDraft() As Boolean
    If Not HasDraft() Then
        LeaveDraft = True
        Exit Function
    End If
    Select Case MsgBox("変更内容を保存しますか？" & vbCrLf & "［はい］保存  ［いいえ］破棄  ［キャンセル］この事案を表示したままにする", _
                       vbYesNoCancel Or vbQuestion Or vbDefaultButton3, "未保存の変更")
        Case vbYes: LeaveDraft = SaveDraft()
        Case vbNo: LeaveDraft = True
    End Select
End Function

Private Function DraftValues() As Object
    Dim patch As Object, fieldId As Variant, box As MSForms.TextBox, definition As Object, oldValue As Variant, newValue As Variant
    Set patch = NewLedgerRecord()
    For Each fieldId In editors.Keys
        Set box = editors(fieldId)
        If Not box.Locked Then
            Set definition = fields(fieldId)
            oldValue = Empty
            If Not isNew Then oldValue = current(fieldId)
            newValue = ControllerInput(CStr(fieldId), CStr(box.value), oldValue, definition)
            If Not LedgerValuesEqual(oldValue, newValue) Then patch.Add fieldId, newValue
        End If
    Next fieldId
    Set DraftValues = patch
End Function

Private Function SaveDraft() As Boolean
    Dim patch As Object, plans As Collection, identity As Object
    On Error GoTo Failed
    Set patch = DraftValues()
    Set plans = ControllerPlanSave(current, peer, patch, isNew)
    If plans.count = 0 Then
        SaveDraft = True
        Exit Function
    End If
    If Not ExecuteFromForm(plans, "事案の保存") Then Exit Function
    SaveDraft = True
    Set identity = current
    If isNew Then Set identity = patch
    isNew = False
    RefreshControllerData
    SearchNow
    ShowRecord identity
    lblDetailNote.caption = "保存しました"
    Exit Function
Failed:
    MsgBox Err.Description, vbExclamation, "事案の保存"
End Function

Private Function ExecuteFromForm(ByVal plans As Collection, ByVal operation As String) As Boolean
    Dim folder As String, journal As String, file As Variant
    If plans.count = 0 Then Exit Function
    If MsgBox("他の利用者が対象管理簿を更新しないよう調整済みですか？" & vbCrLf & _
              "転記指示 " & plans.count & " 件を実行します。", vbYesNo Or vbQuestion Or vbDefaultButton2, operation) <> vbYes Then Exit Function
    folder = ControllerSetting("B8")
    If Len(folder) > 0 Then
        journal = ControllerNewJournal(folder)
    Else
        file = Application.GetOpenFilename("Excel ブック (*.xlsx),*.xlsx", , "空の復旧ジャーナルを選択")
        If VarType(file) = vbBoolean Then Exit Function
        journal = CStr(file)
    End If
    ExecuteFromForm = ControllerExecute(plans, journal, True)
End Function

Private Sub MoveRecord(ByVal offset As Long)
    Dim rows As Collection, record As Object, key As String, index As Long
    If current Is Nothing Then Exit Sub
    key = LedgerRecordKey(current, fields)
    Set rows = hits
    If Not rows Is Nothing Then
        For Each record In rows
            index = index + 1
            If LedgerRecordKey(record, fields) = key Then Exit For
        Next record
        If index <= rows.count Then
            If index > 0 Then
                If LedgerRecordKey(rows(index), fields) = key Then
                    SelectSearchRow index + offset
                    Exit Sub
                End If
            End If
        End If
    End If
    Set rows = ControllerRows()
    index = 0
    For Each record In rows
        index = index + 1
        If LedgerRecordKey(record, fields) = key Then
            If index + offset >= 1 And index + offset <= rows.count Then
                If LeaveDraft() Then ShowRecord rows(index + offset)
            End If
            Exit Sub
        End If
    Next record
End Sub

Public Sub PickFieldDate(ByVal id As String)
    Dim box As MSForms.TextBox, day As Date, initial As Variant
    On Error GoTo Failed
    Set box = editors(id)
    If box.Locked Then Exit Sub
    On Error Resume Next
    initial = ControllerDate(CStr(box.value))
    If Err.number <> 0 Then initial = Empty
    Err.Clear
    On Error GoTo Failed
    If Calendar_Select(day, initial, CStr(fields(id)("Header"))) Then box.value = Format$(day, "yyyy/mm/dd")
    Exit Sub
Failed:
    MsgBox Err.Description, vbExclamation, "日付選択"
End Sub

Private Sub cmdWeekly_Click()
    Dim plans As Collection, conflicts As Variant, issue As Variant, message As String
    On Error GoTo Failed
    Dim file As Variant
    file = Application.GetOpenFilename("Excel ブック (*.xlsx),*.xlsx", , "週次取込元を選択")
    If VarType(file) = vbBoolean Then Exit Sub
    Set plans = PlanWeeklyImport(CStr(file), conflicts)
    For Each issue In conflicts
        message = message & LocalWeeklyIssue(CStr(issue)) & vbCrLf
        If Len(message) > 26000 Then
            message = Left$(message, 26000) & vbCrLf & "表示上限に達しました。"
            Exit For
        End If
    Next issue
    If conflicts.count > 0 Then message = "照合に差異がある事案  " & conflicts.count & " 件" & vbCrLf & message
    If Len(message) = 0 Then message = "追加対象 " & (plans.count \ 2) & " 件"
    txtNotice.value = message
    If ExecuteFromForm(plans, "週次取込") Then
        RefreshControllerData
        SearchNow
        txtNotice.value = "週次取込を完了しました。" & vbCrLf & message
    End If
    Exit Sub
Failed:
    txtNotice.value = Err.Description
    MsgBox Err.Description, vbExclamation, "週次取込"
End Sub

Private Sub cmdReload_Click()
    Dim plans As Collection, message As String
    On Error GoTo SyncUnavailable
    Set plans = PlanBranchSynchronization()
    If plans.count > 0 Then
        If ExecuteFromForm(plans, "署から全体への同期") Then message = "同期 " & plans.count & " 件を保存しました。"
    Else
        message = "同期対象の差分はありません。"
    End If
    GoTo ReloadRows
SyncUnavailable:
    message = "同期を実行できませんでした。" & vbCrLf & Err.Description
    Resume ReloadRows
ReloadRows:
    On Error GoTo Failed
    RefreshControllerData
    SearchNow
    txtNotice.value = message & vbCrLf & "検索データを再取得しました。"
    Exit Sub
Failed:
    txtNotice.value = message & vbCrLf & Err.Description
End Sub

Private Function LocalWeeklyIssue(ByVal issue As String) As String
    Dim separator As Long, tail As String, item As Object, id As Variant, label As String, pattern As Object
    Dim addresses As Object
    separator = InStr(1, issue, ": ", vbBinaryCompare)
    If separator < 1 Then
        LocalWeeklyIssue = issue
        Exit Function
    End If
    tail = Mid$(issue, separator + 2)
    Set pattern = CreateObject("VBScript.RegExp")
    pattern.Global = True
    For Each id In fields.Keys
        If Not CStr(id) Like "*[!A-Za-z0-9_]*" Then
            Set item = fields(id)
            pattern.pattern = "\b" & CStr(id) & "\b"
            tail = pattern.Replace(tail, CStr(item("Header")))
        End If
    Next id
    Set addresses = LoadLedgerAddresses()
    For Each id In addresses.Keys
        If Not CStr(id) Like "*[!A-Za-z0-9_]*" Then
            Set item = addresses(id)
            label = CStr(item("Office"))
            If item("Role") = "Master" Then label = "全体"
            pattern.pattern = "\b" & CStr(id) & "\b"
            tail = pattern.Replace(tail, label)
        End If
    Next id
    tail = Replace(tail, " differs: ", "の差異：")
    tail = Replace(tail, "Missing from ", "未登録：")
    tail = Replace(tail, "Also exists in ", "別の署に登録済み：")
    LocalWeeklyIssue = Left$(issue, separator - 1) & "：" & tail
End Function
