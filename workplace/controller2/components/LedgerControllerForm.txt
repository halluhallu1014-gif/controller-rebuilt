Private Sub UserForm_Initialize()

    Set fields = ControllerFields()
    Set editors = NewLedgerRecord()
    Set dateButtons = NewLedgerRecord()
    Set dateHandlers = New Collection
    Set offices = ControllerOfficeNames()
    Set officeButtons = New Collection
    Set results = New ExtendedListBox
    Set related = New ExtendedListBox
    ConfigureList results, fraResults, True
    ConfigureList related, fraRelated, False
    Set button = fraOffices.Controls.Add("Forms.OptionButton.1", "OfficeAll", True)
    SetupOffice button, "全署", 0
    button.value = True
    For Each office In offices
        index = index + 1
        Set button = fraOffices.Controls.Add("Forms.OptionButton.1", "Office" & index, True)
        SetupOffice button, CStr(office), index
    Next office
    fraOffices.ScrollBars = fmScrollBarsNone
    If (index + 1) * 65 + 12 > fraOffices.InsideWidth Then
        fraOffices.ScrollBars = fmScrollBarsHorizontal
        fraOffices.ScrollWidth = (index + 1) * 65 + 12
        fraOffices.Height = 37
    End If
    BuildFields
    ApplyPageFonts
    pages.value = 0
    UpdateModeLabel
    ResetSearchDisplay "検索条件を入力して検索"
    RenderRelatedPage 1
End Sub

Private Sub ApplyPageFonts()

    For Each Page In pages.pages

            If TypeOf control Is MSForms.label Or TypeOf control Is MSForms.CommandButton Or _
               TypeOf control Is MSForms.CheckBox Or TypeOf control Is MSForms.TextBox Or TypeOf control Is MSForms.ComboBox Then
                control.Font.name = "Meiryo UI"
                control.Font.size = 10
            End If
        Next control

    lblDetail.Font.size = 12: lblDetail.Font.bold = True
    lblLegacyDetailTitle.Font.size = 19: lblLegacyDetailTitle.Font.bold = True
    lblRelated.Font.size = 12: lblRelated.Font.bold = True
    lblCount.Font.size = 12: lblCount.Font.bold = True
    cmdSave.Font.bold = True: cmdNew.Font.bold = True
End Sub

Public Sub ShowInitial()
    Dim owner As window
    If Me.Visible Then Exit Sub
    If Not isPositioned Then
        Set owner = ThisWorkbook.Windows(1)
        Me.StartUpPosition = 0
        Me.Left = Application.Left + (Application.Width - Me.Width) / 2
        Me.Top = Application.Top + (Application.Height - Me.Height) / 2
        isPositioned = True
    End If
    Me.Show vbModeless
End Sub

Private Sub UpdateModeLabel()
    If isModal Then
        cmdMode.caption = "モードレスに切替"
    Else
        cmdMode.caption = "モーダルに切替"
    End If
    cmdDetailMode.caption = cmdMode.caption
End Sub

Private Sub cmdDetailMode_Click()
    cmdMode_Click
End Sub

Private Sub cmdMode_Click()
    On Error GoTo Failed
    If isModal Then
        isModal = False
        Me.Hide
        Exit Sub
    End If

    isModal = True
    UpdateModeLabel
    Me.Hide
    ' A modal Show returns after Hide or Unload, on the original modeless call stack.
    Me.Show vbModal
    If isClosing Then Exit Sub

    UpdateModeLabel
    Me.Show vbModeless
    Exit Sub
Failed:
    Dim message As String
    message = Err.Description
    isModal = False
    UpdateModeLabel
    If Not isClosing And Not Me.Visible Then Me.Show vbModeless
    MsgBox message, vbExclamation, "表示モード"
End Sub

Private Function SelectedOffice() As String
    Dim button As MSForms.OptionButton
    For Each button In officeButtons
        If button.value And button.name <> "OfficeAll" Then
            SelectedOffice = button.caption
            Exit Function
        End If
    Next button
End Function

Private Function SearchCriteriaChanged() As Boolean
    SearchCriteriaChanged = (searchedName <> CStr(txtName.value) Or _
                             searchedOffice <> SelectedOffice() Or searchedPending <> CBool(chkPending.value))
End Function

Private Sub ResetSearchDisplay(ByVal caption As String)
    Set hits = Nothing
    searchPage = GetPageWindow(0, 1, SEARCH_PAGE_SIZE)
    results.Clear
    lblCount.caption = caption
    cmdSearchPreviousPage.Enabled = False
    cmdSearchNextPage.Enabled = False
End Sub

Private Sub SearchNow()
    If Not hasSearch Then Exit Sub
    On Error GoTo Failed
    searchedName = CStr(txtName.value)
    searchedOffice = SelectedOffice()
    searchedPending = CBool(chkPending.value)
    Set hits = ControllerFind(searchedName, searchedOffice, searchedPending)
    RenderSearchPage 1
    Exit Sub
Failed:
    Dim message As String
    message = Err.Description
    ResetSearchDisplay "検索結果を取得できませんでした"
    txtNotice.value = message
End Sub

Private Sub RenderSearchPage(ByVal requested As Long)
    searchPage = GetPageWindow(hits.count, requested, SEARCH_PAGE_SIZE)
    ShowRows results, hits, searchPage, True
    lblCount.caption = PageCaption(hits.count, searchPage)
    cmdSearchPreviousPage.Enabled = (searchPage.Page > 1)
    cmdSearchNextPage.Enabled = (searchPage.Page < searchPage.PageCount)
End Sub

Private Sub RenderRelatedPage(ByVal requested As Long)
    Dim total As Long
    If Not relatedRows Is Nothing Then total = relatedRows.count
    relatedPage = GetPageWindow(total, requested, RELATED_PAGE_SIZE)
    ShowRows related, relatedRows, relatedPage, False
    lblRelated.caption = "関連事案  " & PageCaption(total, relatedPage)
    cmdRelatedPreviousPage.Enabled = (relatedPage.Page > 1)
    cmdRelatedNextPage.Enabled = (relatedPage.Page < relatedPage.PageCount)
End Sub

Private Function PageCaption(ByVal total As Long, ByRef window As PageWindow) As String
    PageCaption = Format$(total, "#,##0") & " 件  |  " & Format$(window.FirstIndex, "#,##0") & "-" & _
                  Format$(window.LastIndex, "#,##0") & " 件  |  " & window.Page & "/" & window.PageCount & " ページ"
End Function

Private Sub ChangeSearchPage(ByVal offset As Long)
    If hits Is Nothing Then Exit Sub
    If SearchCriteriaChanged() Then
        SearchNow
    Else
        RenderSearchPage searchPage.Page + offset
    End If
End Sub

Private Sub ShowRows(ByVal list As ExtendedListBox, ByVal rows As Collection, ByRef window As PageWindow, ByVal isSearch As Boolean)
    Dim data() As Variant, row As Long, index As Long, record As Object
    If window.ItemCount = 0 Then
        list.Clear
        Exit Sub
    End If
    If isSearch Then
        ReDim data(0 To window.ItemCount - 1, 0 To 4)
    Else
        ReDim data(0 To window.ItemCount - 1, 0 To 3)
    End If
    For index = window.FirstIndex To window.LastIndex
        Set record = rows(index)
        If isSearch Then
            data(row, 0) = record("OfficeName")
            data(row, 1) = record("PersonName")
            data(row, 2) = record("TaxPeriod")
            data(row, 3) = record("ApplicationType")
            data(row, 4) = record("ProcessingCategory")
        Else
            data(row, 0) = record("TaxPeriod")
            data(row, 1) = CStr(record("CommonNumber")) & "-" & CStr(record("BranchNumber"))
            data(row, 2) = record("ApplicationType")
            data(row, 3) = record("ProcessingCategory")
        End If
        row = row + 1
    Next index
    list.list = data
    list.ListIndex = -1
    list.TopIndex = 0
End Sub

Public Sub SelectSearchRow(ByVal index As Long)
    If hits Is Nothing Then Exit Sub
    If index < 1 Or index > hits.count Then Exit Sub
    If Not LeaveDraft() Then Exit Sub
    If index < searchPage.FirstIndex Or index > searchPage.LastIndex Then RenderSearchPage PageForItem(index, SEARCH_PAGE_SIZE)
    results.ListIndex = index - searchPage.FirstIndex
    ShowRecord hits(index)
End Sub

Private Sub ShowRecord(ByVal identity As Object)
    Dim master As Object, snapshot As Object, other As Object, index As Long, key As String
    On Error GoTo Failed
    Set master = ControllerMasterRecord(identity)
    Set snapshot = ControllerBranchRecord(master)
    Set current = master
    Set peer = snapshot
    isNew = False
    Set relatedRows = ControllerRelated(current)
    key = LedgerRecordKey(current, fields)
    Dim relatedIndex As Long
    For Each other In relatedRows
        index = index + 1
        If LedgerRecordKey(other, fields) = key Then
            relatedIndex = index
            Exit For
        End If
    Next other
    If relatedIndex > 0 Then
        RenderRelatedPage PageForItem(relatedIndex, RELATED_PAGE_SIZE)
        related.ListIndex = relatedIndex - relatedPage.FirstIndex
    Else
        RenderRelatedPage 1
    End If
    lblDetail.caption = CStr(current("PersonName")) & "   /   " & CStr(current("OfficeName"))
    lblDetailNote.caption = vbNullString
    If ControllerLocked(current) Or ControllerLocked(peer) Then
        lblDetailNote.caption = "署引継済みのため閲覧のみ"
    ElseIf peer Is Nothing Then
        lblDetailNote.caption = "対象署の読込み不可。閲覧のみ"

    End If
    FillEditors
    pages.value = 1
    Exit Sub
Failed:
    MsgBox Err.Description, vbExclamation, "事案を開く"
End Sub

Private Sub FillEditors()
    Dim fieldId As Variant, definition As Object, box As MSForms.TextBox, writable As Boolean
    For Each fieldId In editors.keys
        Set box = editors(fieldId)
        If fields.Exists(fieldId) Then
            Set definition = fields(fieldId)
            writable = isNew
            If isNew Then
                box.value = vbNullString
            Else
                box.value = ControllerDisplay(current(fieldId), definition)
                writable = Not ControllerLocked(current) And Not ControllerLocked(peer) And Not peer Is Nothing And definition("Editable") = ChrW(&H25CB) And definition("Rank") = 0 And CStr(fieldId) <> "OfficeName"
            End If
            box.Locked = Not writable
        Else
            box.value = "未割当"
            box.Locked = True
        End If
        If dateButtons.Exists(fieldId) Then dateButtons(fieldId).Enabled = Not box.Locked
        If box.Locked Then
            box.BackColor = RGB(241, 245, 250)
        Else
            box.BackColor = vbWhite
        End If
    Next fieldId
    fraFields.ScrollTop = 0
    If isNew Then cmdSave.caption = "登録・保存" Else cmdSave.caption = "更新・保存"
    cmdSave.Enabled = isNew Or (Not ControllerLocked(current) And Not ControllerLocked(peer) And Not peer Is Nothing)
    cmdSource.Enabled = Not isNew
    cmdPrev.Enabled = Not isNew
    cmdNext.Enabled = Not isNew
End Sub

Private Function HasDraft() As Boolean
    Dim box As MSForms.TextBox, fieldId As Variant, definition As Object, oldText As String
    If pages.value <> 1 Then Exit Function
    For Each fieldId In editors.keys
        Set box = editors(fieldId)
        If Not box.Locked Then
            oldText = vbNullString
            If Not isNew Then
                Set definition = fields(fieldId)
                oldText = ControllerDisplay(current(fieldId), definition)
            End If
            If CStr(box.value) <> oldText Then
                HasDraft = True
                Exit Function
            End If
        End If
    Next fieldId
End Function

Private Function LeaveDraft() As Boolean
    If Not HasDraft() Then
        LeaveDraft = True
        Exit Function
    End If
    Select Case MsgBox("変更内容を保存しますか？" & vbCrLf & "［はい］保存  ［いいえ］破棄  ［キャンセル］この事案を表示したままにする", _
                       vbYesNoCancel Or vbQuestion Or vbDefaultButton3, "未保存の変更")
        Case vbYes: LeaveDraft = SaveDraft()
        Case vbNo: LeaveDraft = True
    End Select
End Function

Private Function DraftValues() As Object
    Dim patch As Object, fieldId As Variant, box As MSForms.TextBox, definition As Object, oldValue As Variant, newValue As Variant
    Set patch = NewLedgerRecord()
    For Each fieldId In editors.keys
        Set box = editors(fieldId)
        If Not box.Locked Then
            Set definition = fields(fieldId)
            oldValue = Empty
            If Not isNew Then oldValue = current(fieldId)
            newValue = ControllerInput(CStr(fieldId), CStr(box.value), oldValue, definition)
            If Not LedgerValuesEqual(oldValue, newValue) Then patch.Add fieldId, newValue
        End If
    Next fieldId
    Set DraftValues = patch
End Function

Private Function SaveDraft() As Boolean
    Dim patch As Object, plans As Collection, identity As Object
    On Error GoTo Failed
    Set patch = DraftValues()
    Set plans = ControllerPlanSave(current, peer, patch, isNew)
    If plans.count = 0 Then
        SaveDraft = True
        Exit Function
    End If
    If Not ExecuteFromForm(plans, "事案の保存", Not isNew) Then Exit Function
    SaveDraft = True
    Set identity = current
    If isNew Then Set identity = patch
    isNew = False
    InvalidateControllerData
    RefreshControllerData
    SearchNow
    ShowRecord identity
    lblDetailNote.caption = "保存しました"
    Exit Function
Failed:
    If SaveDraft Then
        Dim message As String
        message = Err.Description
        Set hits = Nothing
        results.Clear
        lblCount.caption = "0 件"
        pages.value = 0
        txtNotice.value = "保存は完了しました。表示の再取得に失敗しました。「再読込」を実行してください。 " & message
        Exit Function
    End If
    MsgBox Err.Description, vbExclamation, "事案の保存"
End Function

Private Function ExecuteFromForm(ByVal plans As Collection, ByVal operation As String, Optional ByVal verifyUpdate As Boolean = False) As Boolean
    Dim folder As String, journal As String, file As Variant
    If plans.count = 0 Then Exit Function
    If MsgBox("他の利用者が対象管理簿を更新しないよう調整済みですか？" & vbCrLf & _
              "転記指示 " & plans.count & " 件を実行します。", vbYesNo Or vbQuestion Or vbDefaultButton2, operation) <> vbYes Then Exit Function
    folder = ControllerSetting("B8")
    If Len(folder) > 0 Then
        journal = ControllerNewJournal(folder)
    Else
        file = Application.GetOpenFilename("Excel ブック (*.xlsx),*.xlsx", , "空の復旧ジャーナルを選択")
        If VarType(file) = vbBoolean Then Exit Function
        journal = CStr(file)
    End If
    If verifyUpdate Then ControllerVerifyUpdate current, peer
    ExecuteFromForm = ControllerExecute(plans, journal, True)
End Function

Private Sub MoveRecord(ByVal offset As Long)
    Dim rows As Collection, record As Object, key As String, index As Long
    If current Is Nothing Then Exit Sub
    key = LedgerRecordKey(current, fields)
    Set rows = hits
    If Not rows Is Nothing Then
        For Each record In rows
            index = index + 1
            If LedgerRecordKey(record, fields) = key Then Exit For
        Next record
        If index <= rows.count Then
            If index > 0 Then
                If LedgerRecordKey(rows(index), fields) = key Then
                    SelectSearchRow index + offset
                    Exit Sub
                End If
            End If
        End If
    End If
    Set rows = ControllerRows()
    index = 0
    For Each record In rows
        index = index + 1
        If LedgerRecordKey(record, fields) = key Then
            If index + offset >= 1 And index + offset <= rows.count Then
                If LeaveDraft() Then ShowRecord rows(index + offset)
            End If
            Exit Sub
        End If
    Next record
End Sub

Public Sub PickFieldDate(ByVal id As String)
    Dim box As MSForms.TextBox, day As Date, initial As Variant
    On Error GoTo Failed
    Set box = editors(id)
    If box.Locked Then Exit Sub
    On Error Resume Next
    initial = ControllerDate(CStr(box.value))
    If Err.number <> 0 Then initial = Empty
    Err.Clear
    On Error GoTo Failed
    If Calendar_Select(day, initial, CStr(fields(id)("Header"))) Then box.value = Format$(day, "yyyy/mm/dd")
    Exit Sub
Failed:
    MsgBox Err.Description, vbExclamation, "日付選択"
End Sub

Private Sub cmdWeekly_Click()
    Dim plans As Collection, conflicts As Variant, issue As Variant, message As String
    On Error GoTo Failed
    Dim file As Variant
    file = Application.GetOpenFilename("Excel ブック (*.xlsx),*.xlsx", , "週次取込元を選択")
    If VarType(file) = vbBoolean Then Exit Sub
    Set plans = PlanWeeklyImport(CStr(file), conflicts)
    For Each issue In conflicts
        message = message & LocalWeeklyIssue(CStr(issue)) & vbCrLf
        If Len(message) > 26000 Then
            message = Left$(message, 26000) & vbCrLf & "表示上限に達しました。"
            Exit For
        End If
    Next issue
    If conflicts.count > 0 Then message = "照合に差異がある事案  " & conflicts.count & " 件" & vbCrLf & message
    If Len(message) = 0 Then message = "追加対象 " & (plans.count \ 2) & " 件"
    txtNotice.value = message
    If ExecuteFromForm(plans, "週次取込") Then
        RefreshControllerData
        SearchNow
        txtNotice.value = "週次取込を完了しました。" & vbCrLf & message
    End If
    Exit Sub
Failed:
    txtNotice.value = Err.Description
    MsgBox Err.Description, vbExclamation, "週次取込"
End Sub

Private Sub cmdReload_Click()
    Dim plans As Collection, message As String, batch As LedgerReadBatch
    On Error GoTo SyncUnavailable
    Set batch = New LedgerReadBatch
    Set plans = PlanBranchSynchronization(batch)
    If plans.count > 0 Then
        ' A write or recovery can change the files. Never publish the pre-write batch.
        Set batch = Nothing
        If ExecuteFromForm(plans, "署から全体への同期") Then message = "同期 " & plans.count & " 件を保存しました。"
    Else
        message = "同期対象の差分はありません。"
    End If
    GoTo ReloadRows
SyncUnavailable:
    message = "同期を実行できませんでした。" & vbCrLf & Err.Description
    Resume ReloadRows
ReloadRows:
    On Error GoTo Failed
    RefreshControllerData batch
    SearchNow
    txtNotice.value = message & vbCrLf & "検索データを再取得しました。"
    Exit Sub
Failed:
    txtNotice.value = message & vbCrLf & Err.Description
End Sub

Private Sub cmdRecover_Click()
    Dim path As Variant
    On Error GoTo Failed
    path = Application.GetOpenFilename("Excel ブック (*.xlsx),*.xlsx", , "中断した処理の復旧ジャーナルを選択")
    If VarType(path) = vbBoolean Then Exit Sub
    If MsgBox("他の利用者が更新しないよう調整済みですか？", vbYesNo Or vbQuestion Or vbDefaultButton2, "復旧") <> vbYes Then Exit Sub
    ResumeLedgerTransfer CStr(path), True
    RefreshControllerData
    SearchNow
    txtNotice.value = "復旧が完了しました。"
    Exit Sub
Failed:
    txtNotice.value = Err.Description
    MsgBox Err.Description, vbExclamation, "復旧"
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = vbFormControlMenu Then Cancel = Not LeaveDraft()
    If Cancel = 0 Then
        isClosing = True
        ReleaseDateButtons
    End If
End Sub

Private Sub ReleaseDateButtons()
    Dim handler As ControllerDateButton
    If dateHandlers Is Nothing Then Exit Sub
    For Each handler In dateHandlers
        handler.Unbind
    Next handler
    Set dateHandlers = Nothing
End Sub

Private Sub UserForm_Terminate()
    ReleaseDateButtons
    If Not results Is Nothing Then results.Dispose
    If Not related Is Nothing Then related.Dispose
End Sub


Private Function LocalWeeklyIssue(ByVal issue As String) As String
    Dim separator As Long, tail As String, item As Object, id As Variant, label As String, pattern As Object
    Dim Addresses As Object
    separator = InStr(1, issue, ": ", vbBinaryCompare)
    If separator < 1 Then
        LocalWeeklyIssue = issue
        Exit Function
    End If
    tail = Mid$(issue, separator + 2)
    Set pattern = CreateObject("VBScript.RegExp")
    pattern.Global = True
    For Each id In fields.keys
        If Not CStr(id) Like "*[!A-Za-z0-9_]*" Then
            Set item = fields(id)
            pattern.pattern = "\b" & CStr(id) & "\b"
            tail = pattern.Replace(tail, CStr(item("Header")))
        End If
    Next id
    Set Addresses = LoadLedgerAddresses()
    For Each id In Addresses.keys
        If Not CStr(id) Like "*[!A-Za-z0-9_]*" Then
            Set item = Addresses(id)
            label = CStr(item("Office"))
            If item("Role") = "Master" Then label = "全体"
            pattern.pattern = "\b" & CStr(id) & "\b"
            tail = pattern.Replace(tail, label)
        End If
    Next id
    tail = Replace(tail, " differs: ", "の差異：")
    tail = Replace(tail, "Missing from ", "未登録：")
    tail = Replace(tail, "Also exists in ", "別の署に登録済み：")
    LocalWeeklyIssue = Left$(issue, separator - 1) & "：" & tail
End Function
