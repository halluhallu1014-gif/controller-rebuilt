Public Sub InstallControllerSettings()
    Dim book As Workbook
    Dim page As Worksheet
    Dim preferences As Worksheet
    Dim button As Shape
    Dim phase As String
    On Error GoTo Failed
    Set book = ThisWorkbook
    phase = "空ブックの検証"
    If book.ReadOnly Or book.Worksheets.Count <> 1 Or book.Sheets.Count <> 1 Then Err.Raise 5
    Set page = book.Worksheets(1)
    If page.Shapes.Count <> 0 Or Application.CountA(page.UsedRange) <> 0 Or book.Names.Count <> 0 Then Err.Raise 5
    If page.Name = "設定" Or page.Name = "Preferences" Then Err.Raise 5
    phase = "設定シートの構築"
    page.Name = "設定"
    page.Range("B2:L4").Merge
    page.Range("B2").Value2 = "CONTROLLER 設定"
    page.Range("B2:L4").Interior.Color = &H432C16
    page.Range("B2:L4").Font.Color = vbWhite
    page.Range("B2:L12").Font.Name = "Meiryo UI"
    page.Range("B2").Font.Size = 20
    page.Range("B6").Value2 = "ファイルを選ぶと、年度・シート・項目を読み取ります。"
    Set button = page.Shapes.AddShape(msoShapeRoundedRectangle, 65, 117, 170, 35)
    button.Name = "OpenSetupButton"
    button.TextFrame2.TextRange.Text = "設定を開く"
    button.TextFrame2.TextRange.Font.Name = "Meiryo UI"
    button.TextFrame2.TextRange.Font.Size = 12
    button.Fill.ForeColor.RGB = &H92642E
    button.OnAction = "'" & Replace(book.Name, "'", "''") & "'!OpenSetup"
    Set preferences = book.Worksheets.Add(Before:=page)
    preferences.Name = "Preferences"
    preferences.Range("A1").Value2 = "既定の設定対象"
    preferences.Visible = xlSheetHidden
    page.Activate
    book.Windows(1).DisplayGridlines = False
    phase = "フォームと設定の検証"
    VerifyInstalledSettings
    MsgBox "CONTROLLER設定の構築と構成確認が完了しました。", vbInformation
    Exit Sub
Failed:
    MsgBox "導入不合格。工程 " & phase & "、エラー番号 " & Err.Number & " " & Err.Source & " " & Err.Description & _
        "。ブックを保存せず閉じてください。", vbCritical
End Sub

Private Sub VerifyInstalledSettings()
    Dim book As Workbook
    Set book = ThisWorkbook
    If book.Worksheets.Count <> 2 Then Err.Raise 5, , "sheet count"
    If Replace(book.Worksheets("設定").Shapes("OpenSetupButton").OnAction, "'", "") <> _
        Replace(book.Name, "'", "") & "!OpenSetup" Then Err.Raise 5, , "マクロ割当"
    If book.Worksheets("Preferences").Visible <> xlSheetHidden Then Err.Raise 5, , "visibility"
    Load UserForm1
    Load UserForm2
    Load UserForm3
    If UserForm1.Controls.Count <> 17 Or UserForm2.Controls.Count <> 4 Or UserForm3.Controls.Count <> 10 Then Err.Raise 5, , "control counts"
    Unload UserForm3
    Unload UserForm2
    Unload UserForm1
End Sub

Public Function SheetLastRow(ByVal sheet As Worksheet, ByVal columns As String) As Long
    Dim found As Range
    Set found = sheet.Range(columns).Find(What:="*", After:=sheet.Range(columns).Cells(1, 1), LookIn:=xlFormulas, _
        LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    SheetLastRow = 1
    If Not found Is Nothing Then SheetLastRow = found.Row
End Function

Public Function FileLabel(ByVal address As String) As String
    address = Replace(address, "\", "/")
    FileLabel = Mid$(address, InStrRev(address, "/") + 1)
End Function

Public Function IsWeb(ByVal address As String) As Boolean
    IsWeb = LCase$(Left$(address, 8)) = "https://"
End Function

Public Function IsDirectUrl(ByVal address As String) As Boolean
    Dim parts As Variant
    If Not IsWeb(address) Then Exit Function
    If InStr(address, "?") Or InStr(address, "%") Or InStr(address, "#") Or InStr(address, "\") Or InStr(address, ":/") > 6 Then Exit Function
    parts = Split(Mid$(address, 9), "/")
    If UBound(parts) < 1 Then Exit Function
    If InStr(parts(0), ".") = 0 Or InStr(parts(0), "@") Or InStr(parts(0), " ") Then Exit Function
    If InStr(address, "/../") Or InStr(address, "/./") Or InStr(address, "/:") Then Exit Function
    IsDirectUrl = True
End Function

Public Function IsWorkbookUrl(ByVal address As String) As Boolean
    Dim extension As String
    If Not IsDirectUrl(address) Then Exit Function
    extension = LCase$(Filesystem().GetExtensionName(FileLabel(address)))
    IsWorkbookUrl = (extension = "xlsx" Or extension = "xlsm" Or extension = "xlsb" Or extension = "xls")
End Function

Public Function IsSameAddress(ByVal first As String, ByVal second As String) As Boolean
    IsSameAddress = StrComp(first, second, vbTextCompare) = 0
End Function

Public Function SafeOpen(ByVal address As String, ByVal writableTarget As Boolean, ByRef message As String) As Workbook
    Dim candidate As Workbook
    Dim opened As Workbook
    Dim previousEvents As Boolean
    Dim previousSecurity As Long
    Dim previousAlerts As Boolean
    previousEvents = Application.EnableEvents
    previousSecurity = Application.AutomationSecurity
    previousAlerts = Application.DisplayAlerts
    On Error GoTo Failed
    If Len(address) = 0 Then GoTo Failed
    If IsWeb(address) Then
        If writableTarget Or Not IsWorkbookUrl(address) Then GoTo Failed
    ElseIf Not Filesystem().FileExists(address) Then
        message = FileLabel(address) & " が見つかりません。ファイルを選び直してください。"
        Exit Function
    End If
    For Each candidate In Application.Workbooks
        If IsSameAddress(candidate.FullName, address) Or IsSameAddress(candidate.Name, FileLabel(address)) Then
            message = FileLabel(address) & " または同名のブックを閉じてから操作してください。"
            Exit Function
        End If
    Next
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    If Not writableTarget Then Application.AutomationSecurity = 3
    Set opened = Workbooks.Open(Filename:=address, UpdateLinks:=0, ReadOnly:=Not writableTarget, _
        IgnoreReadOnlyRecommended:=True, Notify:=False, AddToMru:=False)
    If writableTarget And opened.ReadOnly Then
        message = FileLabel(address) & " が読取専用です。別の利用者の編集状況と権限を確認してください。"
        opened.Close False
    Else
        Set SafeOpen = opened
    End If
Finished:
    Application.EnableEvents = previousEvents
    Application.AutomationSecurity = previousSecurity
    Application.DisplayAlerts = previousAlerts
    Exit Function
Failed:
    message = FileLabel(address) & " を開けません。保存先、ファイルの種類、アクセス権を確認してください。"
    Resume Finished
End Function

Public Function HeadersMatch(ByVal sheet As Worksheet, ByVal firstColumn As Long, ByVal expected As Variant) As Boolean
    Dim index As Long
    For index = LBound(expected) To UBound(expected)
        If CStr(sheet.Cells(1, firstColumn + index).Value2) <> CStr(expected(index)) Then Exit Function
    Next
    HeadersMatch = True
End Function

Public Function IsController(ByVal book As Workbook, ByRef message As String) As Boolean
    Dim settings As Worksheet
    Dim fields As Worksheet
    Dim comparison As Worksheet
    Dim other As Worksheet
    On Error GoTo Invalid
    If book.FileFormat <> xlOpenXMLWorkbookMacroEnabled Or book.Date1904 Then GoTo Invalid
    Set settings = book.Worksheets("Settings")
    Set fields = book.Worksheets("Fields")
    Set comparison = book.Worksheets("Comparison")
    Set other = book.Worksheets("検索")
    Set other = book.Worksheets("詳細")
    If Not HeadersMatch(settings, 1, Array("Setting", "Value")) Then GoTo Invalid
    If Not HeadersMatch(settings, 4, Array("Year", "StaffAddress", "Sheet", "Enabled", "AllowWrite", "GeneralAddress")) Then GoTo Invalid
    If Not HeadersMatch(fields, 1, Array("FieldId", "Header", "Type", "Sheet", "Cell", "Required")) Then GoTo Invalid
    If Not HeadersMatch(fields, 8, Array("Year", "FieldId", "HeaderOverride")) Then GoTo Invalid
    If CStr(comparison.Range("A2").Value2) <> "Address" Or CStr(comparison.Range("A3").Value2) <> "Sheet" Then GoTo Invalid
    If CStr(book.Worksheets("Recovery").Range("B1").Value2) <> "" Or _
        CStr(book.Worksheets("MoveRecovery").Range("B1").Value2) <> "" Then
        message = "本体で前回の保存・移動結果を確認してから設定してください。"
        Exit Function
    End If
    IsController = True
    Exit Function
Invalid:
    message = "対応するCONTROLLERの設定構造が見つかりません。統合版の .xlsm を選択してください。"
End Function

Public Function Snapshot(ByVal book As Workbook) As Object
    Dim result As Object
    Dim sheetName As Variant
    Dim sheet As Worksheet
    Dim columnCount As Long
    Dim rowIndex As Long
    Dim colIndex As Long
    Dim area As Range
    Dim formulas As Variant
    Dim values As Variant
    Dim value As Variant
    Set result = Dictionary()
    For Each sheetName In Array("Settings", "Fields", "Comparison")
        Set sheet = book.Worksheets(CStr(sheetName))
        Select Case sheetName
            Case "Settings": columnCount = 25
            Case "Fields": columnCount = 11
            Case "Comparison": columnCount = 6
        End Select
        Set area = sheet.Range(sheet.Cells(1, 1), sheet.Cells(Application.Max(7, SheetLastRow(sheet, "A:" & _
            Split(sheet.Cells(1, columnCount).Address, "$")(1))), columnCount))
        'Formulaは数値定数を文字列で返すことがある。定数の型はValue2で保持する。
        formulas = area.Formula
        values = area.Value2
        For rowIndex = 1 To UBound(values, 1)
            For colIndex = 1 To columnCount
                value = values(rowIndex, colIndex)
                If VarType(formulas(rowIndex, colIndex)) = vbString Then
                    If Left$(formulas(rowIndex, colIndex), 1) = "=" Then
                        If area.Cells(rowIndex, colIndex).HasFormula Then value = formulas(rowIndex, colIndex)
                    End If
                End If
                If IsError(value) Then Err.Raise 5, , "設定シートにExcelエラーがあります。"
                If Len(CStr(value)) > 0 Then
                    result.Add CStr(sheetName) & "|" & rowIndex & "|" & colIndex, value
                End If
            Next
        Next
    Next
    Set Snapshot = result
End Function

Public Function ReadCell(ByVal sheetName As String, ByVal row As Long, ByVal column As Long) As String
    Dim key As String
    key = sheetName & "|" & row & "|" & column
    If draftChanges.Exists(key) Then
        ReadCell = CStr(draftChanges(key))
    ElseIf originalValues.Exists(key) Then
        ReadCell = CStr(originalValues(key))
    End If
End Function

Public Sub StageCell(ByVal sheetName As String, ByVal row As Long, ByVal column As Long, ByVal value As String)
    Dim key As String
    Dim original As String
    key = sheetName & "|" & row & "|" & column
    If originalValues.Exists(key) Then original = CStr(originalValues(key))
    If value = original Then
        If draftChanges.Exists(key) Then draftChanges.Remove key
    Else
        draftChanges(key) = value
    End If
End Sub

Public Function ConfigRow(ByVal name As String) As Long
    Dim key As Variant
    Dim parts As Variant
    For Each key In originalValues.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Settings" And parts(2) = "1" Then
            If originalValues(key) = name Then
                ConfigRow = CLng(parts(1))
                Exit Function
            End If
        End If
    Next
    Err.Raise 5, , "必要な運用設定が本体にありません。"
End Function

Public Function IsDirty() As Boolean
    If Not draftChanges Is Nothing Then IsDirty = draftChanges.Count > 0
End Function

Public Function LoadTarget(ByVal address As String, ByRef message As String) As Boolean
    Dim book As Workbook
    Dim sheet As Worksheet
    Dim row As Long
    Dim fieldId As String
    Dim year As String
    Dim localYears As Object
    Dim localFields As Object
    Dim localValues As Object
    Dim item As Object
    On Error GoTo Failed
    message = ""
    If IsWeb(address) Then
        message = "設定を書き込むCONTROLLERはローカルの .xlsm を選択してください。"
        Exit Function
    End If
    Set book = SafeOpen(address, False, message)
    If book Is Nothing Then Exit Function
    If Not IsController(book, message) Then GoTo Finished
    Set localValues = Snapshot(book)
    Set localFields = Dictionary()
    Set localYears = Dictionary()
    Set sheet = book.Worksheets("Fields")
    For row = 2 To SheetLastRow(sheet, "A:A")
        fieldId = CStr(sheet.Cells(row, 1).Value2)
        If Len(fieldId) > 0 Then
            localFields.Add fieldId, Array(CStr(sheet.Cells(row, 2).Value2), _
                sheet.Cells(row, 6).Value2 = "Yes" Or fieldId = "MaskedNumber" Or fieldId = "Branch", CStr(sheet.Cells(row, 3).Value2))
        End If
    Next
    If Not localFields.Exists("MaskedNumber") Or Not localFields.Exists("Branch") Or Not localFields.Exists("PersonName") Then
        message = "本体の必須項目の定義が不足しています。"
        GoTo Finished
    End If
    Set sheet = book.Worksheets("Settings")
    For row = 2 To SheetLastRow(sheet, "D:I")
        year = CStr(sheet.Cells(row, 4).Value2)
        If Len(year) > 0 Then
            If localYears.Exists(year) Then
                message = year & "年度が本体の設定で重複しています。重複を整理してから開き直してください。"
                GoTo Finished
            End If
            Set item = NewYearItem(row)
            item("Staff") = CStr(sheet.Cells(row, 5).Value2)
            item("Sheet") = CStr(sheet.Cells(row, 6).Value2)
            item("Enabled") = (sheet.Cells(row, 7).Value2 = "Yes")
            item("Write") = CStr(sheet.Cells(row, 8).Value2)
            item("General") = CStr(sheet.Cells(row, 9).Value2)
            localYears.Add year, item
        End If
    Next
    Set sheet = book.Worksheets("Comparison")
    comparisonPath = CStr(sheet.Range("B2").Value2)
    comparisonSheet = CStr(sheet.Range("B3").Value2)
    targetPath = book.FullName
    Set fieldSpecs = localFields
    Set draftYears = localYears
    Set originalValues = localValues
    Set draftChanges = Dictionary()
    LoadTarget = True
Finished:
    If Not book Is Nothing Then book.Close False
    Exit Function
Failed:
    message = "既存設定を読めませんでした。形式・重複・数式エラーを確認してください。"
    Resume Finished
End Function

Public Function NewYearItem(ByVal row As Long) As Object
    Dim item As Object
    Set item = Dictionary()
    item.Add "Row", row
    item.Add "Staff", ""
    item.Add "General", ""
    item.Add "Sheet", ""
    item.Add "Enabled", False
    item.Add "Write", "No"
    item.Add "State", ""
    item.Add "Count", 0
    item.Add "Unmapped", ""
    Set NewYearItem = item
End Function

Public Function NextYearRow() As Long
    Dim year As Variant
    NextYearRow = 2
    For Each year In draftYears.Keys
        If draftYears(year)("Row") >= NextYearRow Then NextYearRow = draftYears(year)("Row") + 1
    Next
    Do While Len(ReadCell("Settings", NextYearRow, 5) & ReadCell("Settings", NextYearRow, 6) & _
        ReadCell("Settings", NextYearRow, 7) & ReadCell("Settings", NextYearRow, 8) & ReadCell("Settings", NextYearRow, 9)) > 0
        NextYearRow = NextYearRow + 1
    Loop
End Function

Public Function SelectFiles(ByVal title As String, ByVal controllerOnly As Boolean, ByVal multiple As Boolean) As Collection
    Dim dialog As Object
    Dim result As New Collection
    Dim item As Variant
    Set dialog = Application.FileDialog(3)
    With dialog
        .Title = title
        .AllowMultiSelect = multiple
        If Len(targetPath) > 0 Then
            .InitialFileName = Filesystem().GetParentFolderName(targetPath) & "\"
        Else
            .InitialFileName = Filesystem().GetParentFolderName(ThisWorkbook.Path) & "\"
        End If
        .Filters.Clear
        If controllerOnly Then
            .Filters.Add "CONTROLLER", "*.xlsm"
        Else
            .Filters.Add "Excelファイル", "*.xlsx;*.xlsm;*.xlsb;*.xls"
        End If
        If .Show = -1 Then
            For Each item In .SelectedItems
                result.Add CStr(item)
            Next
        End If
    End With
    Set SelectFiles = result
End Function

Public Function ExistingFile(ByVal year As String, ByRef message As String) As String
    Dim item As Object
    Dim staff As String
    Dim general As String
    Dim hasStaff As Boolean
    Dim hasGeneral As Boolean
    Set item = draftYears(year)
    staff = CStr(item("Staff"))
    general = CStr(item("General"))
    If IsWeb(staff) Or IsWeb(general) Then
        ExistingFile = PickValue(year & "年度の現在のファイルを選んでください。", Choices(staff, general))
        Exit Function
    End If
    hasStaff = Filesystem().FileExists(staff)
    hasGeneral = Filesystem().FileExists(general)
    If hasStaff = hasGeneral Then
        If hasStaff Then
            message = year & "年度のファイルが両候補にあります。正式な所在を確認してください。"
        Else
            message = year & "年度のファイルが見つかりません。「ファイル変更」で選び直してください。"
        End If
    ElseIf hasStaff Then
        ExistingFile = staff
    Else
        ExistingFile = general
    End If
End Function

Public Function PrepareLedger(ByVal address As String, ByVal year As String, ByRef message As String) As Object
    Dim table As Object
    Dim previous As Object
    Dim mapping As Object
    Dim preferred As String
    Dim side As String
    Dim pairSide As String
    Dim alternate As String
    Dim priorPath As String
    Dim candidate As String
    Dim picked As String
    Dim reply As Variant
    Dim id As Variant
    If draftYears.Exists(year) Then
        Set previous = draftYears(year)
        preferred = CStr(previous("Sheet"))
    End If
    Set table = ReadTable(address, preferred, True, message)
    If table Is Nothing Then Exit Function
    Set mapping = MapLedger(table("Headers"), year)
    For Each id In mapping.Keys
        If Len(CStr(mapping(id))) = 0 Then
            If Not PickHeading(CStr(id), mapping, table("Headers")) Then Exit Function
        End If
    Next
    side = AddressToRole(address, previous)
    candidate = GuessCounterpart(address, pairSide)
    If Len(side) = 0 Then
        picked = PickValue("選んだファイルが現在置かれている側を選んでください。", Choices("常勤", "非常勤"))
        If Len(picked) = 0 Then Exit Function
        side = "Staff"
        If picked = "非常勤" Then side = "General"
    End If
    If Not previous Is Nothing Then
        If side = "Staff" Then
            priorPath = CStr(previous("Staff"))
            alternate = CStr(previous("General"))
        Else
            priorPath = CStr(previous("General"))
            alternate = CStr(previous("Staff"))
        End If
        If Len(alternate) > 0 And Len(priorPath) > 0 Then
            If IsSameAddress(FileLabel(alternate), FileLabel(priorPath)) And Not IsSameAddress(FileLabel(address), FileLabel(priorPath)) Then
                alternate = Left$(alternate, Len(alternate) - Len(FileLabel(alternate))) & FileLabel(address)
            End If
        End If
    End If
    If IsWeb(alternate) <> IsWeb(address) Then alternate = ""
    If Len(alternate) = 0 And side = pairSide Then alternate = candidate
    If Len(alternate) = 0 Then
        picked = "非常勤"
        If side = "General" Then picked = "常勤"
        If IsWeb(address) Then
            reply = AskText(picked & "側の直接ファイルURLを指定してください。")
            If VarType(reply) = vbBoolean Then Exit Function
            alternate = Trim$(CStr(reply))
        Else
            alternate = SelectFolder(picked & "側の保存フォルダを選択")
            If Len(alternate) = 0 Then Exit Function
            alternate = Filesystem().BuildPath(alternate, FileLabel(address))
        End If
    End If
    If Not IsWeb(alternate) Then
        If Filesystem().FileExists(alternate) Then
            message = "両方の候補にファイルがあります。正式な所在を確認してください。"
            Exit Function
        End If
    End If
    table.Add "Staff", address
    table.Add "General", alternate
    If side = "General" Then
        table("Staff") = alternate
        table("General") = address
    End If
    table.Add "Map", mapping
    Set PrepareLedger = table
End Function

Private Function PickHeading(ByVal fieldId As String, ByVal mapping As Object, ByVal headers As Object) As Boolean
    Dim values As New Collection
    Dim heading As Variant
    Dim id As Variant
    Dim isUsed As Boolean
    Dim picked As String
    If Not fieldSpecs(fieldId)(1) Then values.Add "（対応なし）"
    For Each heading In headers.Keys
        isUsed = False
        For Each id In mapping.Keys
            If CStr(id) <> fieldId And CStr(mapping(id)) = CStr(heading) Then isUsed = True
        Next
        If Not isUsed Then values.Add CStr(heading)
    Next
    picked = PickValue("「" & FieldLabel(fieldId) & "」に対応する、元ファイルの見出しを選んでください。", _
        values, CStr(mapping(fieldId)))
    If Len(picked) = 0 Then Exit Function
    If picked = "（対応なし）" Then picked = "[Absent]"
    mapping(fieldId) = picked
    PickHeading = True
End Function

Public Function ReviewMapping(ByVal table As Object) As Boolean
    Dim labels As New Collection
    Dim options As Object
    Dim mapping As Object
    Dim id As Variant
    Dim heading As String
    Dim label As String
    Dim picked As String
    Set mapping = table("Map")
    Set options = Dictionary()
    For Each id In mapping.Keys
        heading = CStr(mapping(id))
        If heading = "[Absent]" Or Len(heading) = 0 Then heading = "対応なし"
        label = FieldLabel(CStr(id)) & "  →  " & heading
        If Not options.Exists(label) Then
            options.Add label, CStr(id)
            labels.Add label
        End If
    Next
    picked = PickValue("変更したい項目だけ選んでください。対応は一覧で確認できます。", labels)
    If Len(picked) > 0 Then ReviewMapping = PickHeading(CStr(options(picked)), mapping, table("Headers"))
End Function

Public Function PrepareComparison(ByVal address As String, ByRef message As String) As Boolean
    Dim table As Object
    Dim existing As Object
    Dim configured As Object
    Dim headers As Object
    Dim yearKeys As Variant
    Dim key As Variant
    Dim row As Long
    Dim id As String
    Dim heading As String
    Dim picked As String
    Dim values As New Collection
    Dim labels As Object
    Set table = ReadTable(address, comparisonSheet, False, message)
    If table Is Nothing Then Exit Function
    Set headers = table("Headers")
    Set existing = Dictionary()
    Set configured = Dictionary()
    For row = 8 To LastComparisonRow()
        id = ReadCell("Comparison", row, 1)
        If Len(id) > 0 Then
            heading = ReadCell("Comparison", row, 2)
            existing.Add id, Array(heading, ReadCell("Comparison", row, 3), ReadCell("Comparison", row, 4), ReadCell("Comparison", row, 5))
        End If
    Next
    For Each key In existing.Keys
        heading = CStr(existing(key)(0))
        If Not headers.Exists(heading) Then heading = SelectExternalHeading(CStr(key), headers, configured)
        If Len(heading) = 0 Then Exit Function
        configured.Add CStr(key), Array(heading, existing(key)(1), existing(key)(2), existing(key)(3))
    Next
    If configured.Count = 0 Then
        yearKeys = Array("SourceYear", "MaskedNumber", "Branch")
        For Each key In yearKeys
            heading = SelectExternalHeading(CStr(key), headers, configured)
            If Len(heading) = 0 Then Exit Function
            configured.Add CStr(key), Array(heading, "Yes", "No", "")
        Next
        Set labels = Dictionary()
        For Each key In fieldSpecs.Keys
            If Not configured.Exists(key) Then
                id = FieldLabel(CStr(key))
                If Not labels.Exists(id) Then
                    labels.Add id, CStr(key)
                    values.Add id
                End If
            End If
        Next
        picked = PickValue("最初に比較する管理簿の項目を選んでください。", values)
        If Len(picked) = 0 Then Exit Function
        id = CStr(labels(picked))
        heading = SelectExternalHeading(id, headers, configured)
        If Len(heading) = 0 Then Exit Function
        configured.Add id, Array(heading, "No", "No", "")
    End If
    PrepareComparison = StageComparison(table, configured, message)
End Function

Private Function SelectExternalHeading(ByVal fieldId As String, ByVal headers As Object, ByVal configured As Object) As String
    Dim items As New Collection
    Dim heading As Variant
    Dim id As Variant
    Dim used As Boolean
    For Each heading In headers.Keys
        used = False
        For Each id In configured.Keys
            If configured(id)(0) = heading Then used = True
        Next
        If Not used Then items.Add CStr(heading)
    Next
    SelectExternalHeading = PickValue("整理簿で「" & FieldLabel(fieldId) & "」に当たる見出しを選んでください。", items)
End Function

Public Function LastComparisonRow() As Long
    Dim key As Variant
    Dim parts As Variant
    LastComparisonRow = 7
    For Each key In originalValues.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Comparison" And CLng(parts(1)) > LastComparisonRow Then LastComparisonRow = CLng(parts(1))
    Next
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Comparison" And CLng(parts(1)) > LastComparisonRow Then LastComparisonRow = CLng(parts(1))
    Next
End Function

Public Function StageComparison(ByVal table As Object, ByVal rules As Object, ByRef message As String) As Boolean
    Dim headers As Object
    Dim used As Object
    Dim id As Variant
    Dim row As Long
    Dim nextRow As Long
    Dim column As Long
    Dim keyCount As Long
    Dim compareCount As Long
    Set headers = table("Headers")
    Set used = Dictionary()
    For Each id In rules.Keys
        If Not headers.Exists(rules(id)(0)) Or used.Exists(rules(id)(0)) Then
            message = "整理簿の見出しが見つからないか、複数の項目に使われています。"
            Exit Function
        End If
        used.Add rules(id)(0), True
        If rules(id)(1) = "Yes" Then
            keyCount = keyCount + 1
        Else
            compareCount = compareCount + 1
        End If
    Next
    If keyCount = 0 Or compareCount = 0 Then
        message = "照合に使う識別項目と、比較する項目を指定してください。"
        Exit Function
    End If
    nextRow = LastComparisonRow() + 1
    For Each id In rules.Keys
        For row = 8 To LastComparisonRow()
            If ReadCell("Comparison", row, 1) = CStr(id) Then Exit For
        Next
        If row > LastComparisonRow() Then
            row = nextRow
            nextRow = nextRow + 1
        End If
        StageCell "Comparison", row, 1, CStr(id)
        For column = 2 To 5
            StageCell "Comparison", row, column, CStr(rules(id)(column - 2))
        Next
    Next
    If Not IsSameAddress(comparisonPath, CStr(table("Address"))) Or comparisonSheet <> CStr(table("Sheet")) Then
        StageCell "Comparison", 4, 2, "No"
    End If
    comparisonPath = CStr(table("Address"))
    comparisonSheet = CStr(table("Sheet"))
    StageCell "Comparison", 2, 2, comparisonPath
    StageCell "Comparison", 3, 2, comparisonSheet
    StageComparison = True
End Function

Public Function FieldLabel(ByVal fieldId As String) As String
    If fieldId = "MaskedNumber" Then
        FieldLabel = "13桁番号"
    ElseIf fieldId = "TaxType" Then
        FieldLabel = "税目"
    ElseIf fieldId = "HistoryMemo" Then
        FieldLabel = "処理履歴メモ"
    ElseIf fieldId = "SourceYear" Then
        FieldLabel = "受付年度"
    Else
        FieldLabel = CStr(fieldSpecs(fieldId)(0))
        If FieldLabel = ABSENT Then FieldLabel = "その他の項目"
    End If
End Function

Private Function NormalHeader(ByVal heading As String) As String
    NormalHeader = Trim$(Replace(Replace(heading, ChrW(&H3000), " "), vbTab, " "))
End Function

Private Function AliasHeader(ByVal fieldId As String) As String
    Select Case fieldId
        Case "MaskedNumber": AliasHeader = "MaskedNumber"
        Case "Furigana": AliasHeader = "氏名(フリガナ)"
        Case "TaxType": AliasHeader = "税目"
    End Select
End Function

Private Function UniqueHeading(ByVal headers As Object, ByVal heading As String) As String
    Dim key As Variant
    If Len(heading) = 0 Or heading = ABSENT Then Exit Function
    If headers.Exists(heading) Then
        UniqueHeading = heading
        Exit Function
    End If
    For Each key In headers.Keys
        If NormalHeader(CStr(key)) = NormalHeader(heading) Then
            If Len(UniqueHeading) > 0 Then
                UniqueHeading = ""
                Exit Function
            End If
            UniqueHeading = CStr(key)
        End If
    Next
End Function

Private Function HeaderList(ByVal sheet As Worksheet, ByRef message As String) As Object
    Dim result As Object
    Dim last As Range
    Dim index As Long
    Dim value As Variant
    Set result = Dictionary()
    Set last = sheet.Rows(1).Find(What:="*", After:=sheet.Cells(1, 1), LookIn:=xlFormulas, LookAt:=xlPart, _
        SearchOrder:=xlByColumns, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    If last Is Nothing Then
        message = sheet.Name & " の1行目に見出しがありません。"
        Exit Function
    End If
    For index = 1 To last.Column
        value = sheet.Cells(1, index).Value2
        If IsError(value) Then
            message = sheet.Name & " の1行目にExcelエラーがあります。"
            Exit Function
        End If
        If Len(CStr(value)) > 0 Then
            If result.Exists(CStr(value)) Then
                message = sheet.Name & " の見出し「" & CStr(value) & "」が重複しています。"
                Exit Function
            End If
            result.Add CStr(value), index
        End If
    Next
    If result.Count < 2 Then
        message = sheet.Name & " の1行目に複数の見出しが必要です。"
        Exit Function
    End If
    Set HeaderList = result
End Function

Private Function LooksLikeLedger(ByVal headers As Object) As Boolean
    Dim id As Variant
    For Each id In Array("MaskedNumber", "Branch", "PersonName")
        If Len(UniqueHeading(headers, CStr(fieldSpecs(id)(0)))) = 0 Then
            If Len(UniqueHeading(headers, AliasHeader(CStr(id)))) = 0 Then Exit Function
        End If
    Next
    LooksLikeLedger = True
End Function

Public Function ReadTable(ByVal address As String, ByVal preferredSheet As String, ByVal isLedger As Boolean, ByRef message As String) As Object
    Dim book As Workbook
    Dim sheet As Worksheet
    Dim selected As Worksheet
    Dim headerMap As Object
    Dim result As Object
    Dim plausible As New Collection
    Dim names As New Collection
    Dim name As Variant
    Dim ignore As String
    On Error GoTo Failed
    message = ""
    If IsWeb(address) And Not IsWorkbookUrl(address) Then
        message = "共有リンクではなく、拡張子までの直接URLを入力してください。"
        Exit Function
    End If
    If IsSameAddress(address, targetPath) Or IsSameAddress(address, ThisWorkbook.FullName) Then
        message = "設定ブックと本体は管理簿として選べません。"
        Exit Function
    End If
    Set book = SafeOpen(address, False, message)
    If book Is Nothing Then Exit Function
    If book.Date1904 Then
        message = "1904年日付形式のファイルは本体で読み込めません。"
        GoTo Finished
    End If
    For Each sheet In book.Worksheets
        names.Add sheet.Name
        If sheet.Name = preferredSheet Then Set selected = sheet
        ignore = ""
        Set headerMap = HeaderList(sheet, ignore)
        If Not headerMap Is Nothing Then
            If Not isLedger Or LooksLikeLedger(headerMap) Then plausible.Add sheet.Name
        End If
    Next
    If selected Is Nothing Then
        If names.Count = 1 Then
            Set selected = book.Worksheets(CStr(names(1)))
        ElseIf plausible.Count = 1 Then
            Set selected = book.Worksheets(CStr(plausible(1)))
        Else
            If plausible.Count > 1 Then Set names = plausible
            name = PickValue("読み込むシートを選択してください。", names)
            If Len(CStr(name)) = 0 Then GoTo Finished
            Set selected = book.Worksheets(CStr(name))
        End If
    End If
    Set headerMap = HeaderList(selected, message)
    If headerMap Is Nothing Then GoTo Finished
    Set result = Dictionary()
    result.Add "Address", address
    result.Add "Sheet", selected.Name
    result.Add "Headers", headerMap
    result.Add "Count", SheetLastRow(selected, "A:XFD") - 1
    Set ReadTable = result
Finished:
    If Not book Is Nothing Then book.Close False
    Exit Function
Failed:
    message = FileLabel(address) & " のシートまたは見出しを読み取れません。"
    Resume Finished
End Function

Public Function GuessYear(ByVal address As String) As String
    Dim result As Object
    Dim year As Variant
    Dim leaf As String
    Dim candidate As String
    Dim position As Long
    Dim endOfEra As Long
    For Each year In draftYears.Keys
        If IsSameAddress(address, draftYears(year)("Staff")) Or IsSameAddress(address, draftYears(year)("General")) Then
            GuessYear = CStr(year)
            Exit Function
        End If
    Next
    Set result = Dictionary()
    leaf = " " & FileLabel(address) & " "
    For position = 2 To Len(leaf) - 4
        If Not Mid$(leaf, position - 1, 1) Like "#" And Not Mid$(leaf, position + 4, 1) Like "#" Then
            candidate = Mid$(leaf, position, 4)
            If IsYear(candidate) Then result(candidate) = True
        End If
        If Mid$(leaf, position, 2) = "令和" Then
            endOfEra = InStr(position + 2, leaf, "年")
            If endOfEra > position + 2 Then
                candidate = Mid$(leaf, position + 2, endOfEra - position - 2)
                If candidate = "元" Then candidate = "1"
                If candidate Like "[1-9]" Or candidate Like "[1-9]#" Then
                    result(CStr(2018 + CLng(candidate))) = True
                End If
            End If
        End If
    Next
    If result.Count = 1 Then GuessYear = CStr(result.Keys()(0))
End Function

Public Function IsYear(ByVal value As String) As Boolean
    IsYear = Len(value) = 4 And value Like "####"
    If IsYear Then IsYear = CLng(value) >= 1900 And CLng(value) <= 2099
End Function

Private Function ParentPath(ByVal address As String) As String
    Dim separator As String
    separator = "\"
    If IsWeb(address) Then separator = "/"
    ParentPath = Left$(address, InStrRev(address, separator) - 1)
End Function

Private Function SiblingRole(ByVal address As String) As String
    Select Case LCase$(FileLabel(ParentPath(address)))
        Case "常勤", "staff": SiblingRole = "Staff"
        Case "非常勤", "general": SiblingRole = "General"
    End Select
End Function

Public Function GuessCounterpart(ByVal address As String, ByRef side As String) As String
    Dim parent As String
    Dim other As String
    Dim separator As String
    Dim folder As String
    side = SiblingRole(address)
    If Len(side) = 0 Then Exit Function
    parent = ParentPath(address)
    separator = "\"
    If IsWeb(address) Then separator = "/"
    Select Case LCase$(FileLabel(parent))
        Case "常勤": other = "非常勤"
        Case "非常勤": other = "常勤"
        Case "staff": other = "General"
        Case "general": other = "Staff"
    End Select
    folder = ParentPath(parent) & separator & other
    If Not IsWeb(address) Then
        If Not Filesystem().FolderExists(folder) Then Exit Function
    End If
    GuessCounterpart = folder & separator & FileLabel(address)
End Function

Public Function AddressToRole(ByVal address As String, ByVal previous As Object) As String
    If Not previous Is Nothing Then
        If Len(previous("Staff")) > 0 Then
            If IsSameAddress(address, previous("Staff")) Or IsSameAddress(ParentPath(address), ParentPath(CStr(previous("Staff")))) Then
                AddressToRole = "Staff"
                Exit Function
            End If
        End If
        If Len(previous("General")) > 0 Then
            If IsSameAddress(address, previous("General")) Or IsSameAddress(ParentPath(address), ParentPath(CStr(previous("General")))) Then
                AddressToRole = "General"
                Exit Function
            End If
        End If
    End If
    AddressToRole = SiblingRole(address)
End Function

Public Function MapLedger(ByVal headers As Object, ByVal year As String) As Object
    Dim result As Object
    Dim used As Object
    Dim id As Variant
    Dim heading As String
    Dim saved As String
    Dim row As Long
    Set result = Dictionary()
    Set used = Dictionary()
    For Each id In fieldSpecs.Keys
        row = OverrideRow(year, CStr(id))
        heading = ""
        If row > 0 Then
            saved = ReadCell("Fields", row, 10)
            If saved = ABSENT Then
                heading = ABSENT
                If id = "TaxType" Then heading = UniqueHeading(headers, "税目")
                If id = "CompletedDate" Then heading = UniqueHeading(headers, "完了日")
            Else
                heading = UniqueHeading(headers, saved)
            End If
        End If
        If Len(heading) = 0 Then heading = UniqueHeading(headers, CStr(fieldSpecs(id)(0)))
        If Len(heading) = 0 Then heading = UniqueHeading(headers, AliasHeader(CStr(id)))
        If Len(heading) > 0 And heading <> ABSENT Then
            If used.Exists(heading) Then
                heading = ""
            Else
                used.Add heading, True
            End If
        End If
        If Len(heading) = 0 And Not fieldSpecs(id)(1) Then heading = ABSENT
        result.Add id, heading
    Next
    Set MapLedger = result
End Function

Private Function OverrideRow(ByVal year As String, ByVal fieldId As String) As Long
    Dim key As Variant
    Dim parts As Variant
    Dim visited As Object
    Dim row As Long
    Set visited = Dictionary()
    For Each key In originalValues.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Fields" And parts(2) = "8" Then visited(parts(1)) = True
    Next
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Fields" And parts(2) = "8" Then visited(parts(1)) = True
    Next
    For Each key In visited.Keys
        row = CLng(key)
        If ReadCell("Fields", row, 8) = year And ReadCell("Fields", row, 9) = fieldId Then
            If OverrideRow > 0 Then Err.Raise 5, , year & "年度の項目対応が重複しています。"
            OverrideRow = row
        End If
    Next
End Function

Private Function NextOverrideRow() As Long
    Dim key As Variant
    Dim parts As Variant
    NextOverrideRow = 2
    For Each key In originalValues.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Fields" And CLng(parts(2)) >= 8 Then
            If CLng(parts(1)) >= NextOverrideRow Then NextOverrideRow = CLng(parts(1)) + 1
        End If
    Next
    Do While Len(ReadCell("Fields", NextOverrideRow, 8) & ReadCell("Fields", NextOverrideRow, 9) & _
        ReadCell("Fields", NextOverrideRow, 10)) > 0
        NextOverrideRow = NextOverrideRow + 1
    Loop
End Function

Public Function CheckMapping(ByVal mapping As Object, ByVal headers As Object, ByRef message As String) As Boolean
    Dim used As Object
    Dim id As Variant
    Dim heading As String
    Set used = Dictionary()
    For Each id In fieldSpecs.Keys
        heading = CStr(mapping(id))
        If heading = ABSENT Or Len(heading) = 0 Then
            If fieldSpecs(id)(1) Then
                message = "「" & FieldLabel(CStr(id)) & "」に対応する見出しを選んでください。"
                Exit Function
            End If
        Else
            If Not headers.Exists(heading) Or used.Exists(heading) Then
                message = "見出し「" & heading & "」が存在しないか、複数の項目に対応しています。"
                Exit Function
            End If
            used.Add heading, True
        End If
    Next
    message = ""
    CheckMapping = True
End Function

Public Function StageLedger(ByVal year As String, ByVal plan As Object, ByRef message As String) As Boolean
    Dim old As Object
    Dim entry As Object
    Dim map As Object
    Dim id As Variant
    Dim row As Long
    Dim override As Long
    Dim heading As String
    Dim transport As String
    Dim transportRow As Long
    Dim candidate As Variant
    Dim missing As String
    On Error GoTo Failed
    If Not IsYear(year) Then
        message = "年度を西暦4桁で指定してください。"
        Exit Function
    End If
    If Len(plan("Staff")) = 0 Or Len(plan("General")) = 0 Or IsSameAddress(plan("Staff"), plan("General")) Then
        message = "常勤・非常勤の候補が両方必要です。それぞれ別の場所を指定してください。"
        Exit Function
    End If
    If IsWeb(plan("Staff")) <> IsWeb(plan("General")) Then
        message = "同じ年度の候補は、両方を同じ保存方式にしてください。"
        Exit Function
    End If
    Set map = plan("Map")
    If Not CheckMapping(map, plan("Headers"), message) Then Exit Function
    For Each candidate In draftYears.Keys
        If CStr(candidate) <> year And draftYears(candidate)("Enabled") Then
            Set entry = draftYears(candidate)
            If IsWeb(entry("Staff")) <> IsWeb(plan("Staff")) Then
                message = "本体はローカルとSharePointを年度ごとに混在できません。運用設定を確認してください。"
                Exit Function
            End If
            If IsSameAddress(entry("Staff"), plan("Staff")) Or IsSameAddress(entry("Staff"), plan("General")) Or _
                IsSameAddress(entry("General"), plan("Staff")) Or IsSameAddress(entry("General"), plan("General")) Then
                message = "このファイルは別の年度で設定済みです。"
                Exit Function
            End If
        End If
    Next
    transport = "Local"
    If IsWeb(plan("Staff")) Then
        If Not IsWorkbookUrl(plan("Staff")) Or Not IsWorkbookUrl(plan("General")) Then
            message = "両候補に共有リンクではない直接ファイルURLを指定してください。"
            Exit Function
        End If
        transport = "SharePoint"
    End If
    transportRow = ConfigRow("Transport")
    Set old = CloneChanges()
    If draftYears.Exists(year) Then
        Set entry = draftYears(year)
        row = entry("Row")
    Else
        row = NextYearRow()
        Set entry = NewYearItem(row)
    End If
    For Each id In map.Keys
        heading = CStr(map(id))
        If heading = ABSENT And CStr(fieldSpecs(id)(0)) <> ABSENT And id <> "TaxType" And id <> "CompletedDate" Then
            If Len(missing) > 0 Then missing = missing & "、"
            missing = missing & FieldLabel(CStr(id))
        End If
        override = OverrideRow(year, CStr(id))
        If heading = CStr(fieldSpecs(id)(0)) Then
            If override > 0 Then
                StageCell "Fields", override, 8, ""
                StageCell "Fields", override, 9, ""
                StageCell "Fields", override, 10, ""
            End If
        Else
            If override = 0 Then override = NextOverrideRow()
            StageCell "Fields", override, 8, year
            StageCell "Fields", override, 9, CStr(id)
            StageCell "Fields", override, 10, heading
        End If
    Next
    StageCell "Settings", row, 4, year
    StageCell "Settings", row, 5, CStr(plan("Staff"))
    StageCell "Settings", row, 6, CStr(plan("Sheet"))
    StageCell "Settings", row, 7, "Yes"
    If Len(entry("Write")) = 0 Then entry("Write") = "No"
    StageCell "Settings", row, 8, CStr(entry("Write"))
    StageCell "Settings", row, 9, CStr(plan("General"))
    StageCell "Settings", transportRow, 2, transport
    entry("Staff") = plan("Staff")
    entry("General") = plan("General")
    entry("Sheet") = plan("Sheet")
    entry("Enabled") = True
    entry("State") = ""
    If YearHasChanges(year, row) Then entry("State") = "確認済み・保存待ち"
    entry("Count") = plan("Count")
    entry("Unmapped") = missing
    If Len(missing) > 0 Then entry("State") = "要確認・任意項目"
    If Not draftYears.Exists(year) Then draftYears.Add year, entry
    StageLedger = True
    Exit Function
Failed:
    If Not old Is Nothing Then Set draftChanges = old
    message = "年度設定を作成できません。既存の項目対応に重複がないか確認してください。"
End Function

Private Function YearHasChanges(ByVal year As String, ByVal row As Long) As Boolean
    Dim key As Variant
    Dim parts As Variant
    Dim overrideRow As Long
    Dim originalKey As String
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Settings" And CLng(parts(1)) = row And CLng(parts(2)) >= 4 And CLng(parts(2)) <= 9 Then
            YearHasChanges = True
            Exit Function
        End If
        If parts(0) = "Fields" And CLng(parts(2)) >= 8 Then
            overrideRow = CLng(parts(1))
            If ReadCell("Fields", overrideRow, 8) = year Then
                YearHasChanges = True
                Exit Function
            End If
            originalKey = "Fields|" & overrideRow & "|8"
            If originalValues.Exists(originalKey) Then
                If CStr(originalValues(originalKey)) = year Then
                    YearHasChanges = True
                    Exit Function
                End If
            End If
        End If
    Next
End Function

    Dim result As Object
    Dim key As Variant
    Set result = Dictionary()
    For Each key In draftChanges.Keys
        result.Add key, draftChanges(key)
    Next
    Set CloneChanges = result
End Function

Private Function IsSameSnapshot(ByVal before As Object, ByVal after As Object) As Boolean
    Dim key As Variant
    If before.Count <> after.Count Then Exit Function
    For Each key In before.Keys
        If Not after.Exists(key) Then Exit Function
        If VarType(before(key)) <> VarType(after(key)) Then Exit Function
        If StrComp(CStr(before(key)), CStr(after(key)), vbBinaryCompare) <> 0 Then Exit Function
    Next
    IsSameSnapshot = True
End Function

Private Function ExpectedSnapshot() As Object
    Dim result As Object
    Dim key As Variant
    Set result = Dictionary()
    For Each key In originalValues.Keys
        result.Add key, originalValues(key)
    Next
    For Each key In draftChanges.Keys
        If Len(CStr(draftChanges(key))) = 0 Then
            If result.Exists(key) Then result.Remove key
        Else
            result(key) = draftChanges(key)
        End If
    Next
    Set ExpectedSnapshot = result
End Function

Private Function CreateBackup(ByVal book As Workbook) As String
    Dim files As Object
    Dim folder As String
    Dim path As String
    Dim suffix As Long
    Set files = Filesystem()
    folder = files.BuildPath(book.Path, "設定バックアップ")
    If Not files.FolderExists(folder) Then files.CreateFolder folder
    Do
        path = files.BuildPath(folder, files.GetBaseName(book.Name) & "_" & Format$(Now, "yyyymmdd_hhnnss") & "_" & suffix & ".xlsm")
        suffix = suffix + 1
    Loop While files.FileExists(path)
    book.SaveCopyAs path
    If Not files.FileExists(path) Then Err.Raise 5, , "退避先にブックを作成できません。"
    If files.GetFile(path).Size = 0 Then Err.Raise 5, , "退避したブックが空です。"
    CreateBackup = path
End Function

Private Function CheckDraft(ByRef message As String) As Boolean
    Dim year As Variant
    Dim item As Object
    Dim site As String
    Dim backup As String
    Dim transport As String
    Dim row As Long
    transport = ReadCell("Settings", ConfigRow("Transport"), 2)
    site = ReadCell("Settings", ConfigRow("SharePointSite"), 2)
    backup = ReadCell("Settings", ConfigRow("BackupFolder"), 2)
    For Each year In draftYears.Keys
        Set item = draftYears(year)
        If item("Enabled") Then
            If transport = "SharePoint" Then
                If Not IsDirectUrl(site) Or Right$(site, 1) = "/" Then
                    message = "運用設定でSharePointのサイトURLを指定してください。"
                    Exit Function
                End If
                If Not IsWorkbookUrl(item("Staff")) Or Not IsWorkbookUrl(item("General")) Then GoTo InvalidYear
                If InStr(1, item("Staff"), site & "/", vbTextCompare) <> 1 Or _
                    InStr(1, item("General"), site & "/", vbTextCompare) <> 1 Then
                    message = year & "年度の候補パスは設定したSharePointサイト内を指定してください。"
                    Exit Function
                End If
            ElseIf transport <> "Local" Or IsWeb(item("Staff")) Or IsWeb(item("General")) Then
                GoTo InvalidYear
            End If
            If item("Write") = "Yes" And transport = "SharePoint" Then
                If ReadCell("Settings", ConfigRow("SharePointExclusive"), 2) <> "Yes" Then
                    message = year & "年度を書込可にする前に必須チェックアウト運用を確認してください。"
                    Exit Function
                End If
            End If
        End If
    Next
    If Len(backup) > 0 Then
        If transport = "SharePoint" Then
            If Not IsDirectUrl(backup) Or InStr(1, backup, site & "/", vbTextCompare) <> 1 Then
                message = "バックアップ先に、設定したSharePointサイト内のフォルダURLを指定してください。"
                Exit Function
            End If
        ElseIf Not Filesystem().FolderExists(backup) Then
            message = "バックアップフォルダが見つかりません。運用設定で選び直してください。"
            Exit Function
        End If
    End If
    CheckDraft = True
    Exit Function
InvalidYear:
    message = year & "年度の候補と保存方式が一致していません。"
End Function

Private Function MacroName(ByVal book As Workbook, ByVal macro As String) As String
    MacroName = "'" & Replace(book.Name, "'", "''") & "'!" & macro
End Function

Private Function CheckRuntime(ByVal book As Workbook, ByRef message As String) As Boolean
    Dim year As Variant
    Dim loaded As Object
    Dim rules As Object
    Dim records As Object
    Dim activeCount As Long
    On Error GoTo Failed
    If CStr(book.Worksheets("Fields").Cells(1, 11).Value2) = "DisplayFormat" Then
        message = "表示形式または選択肢が無効です。設定を確認してください。"
        If Not CBool(Application.Run(MacroName(book, "Presentation.ValidatePresentation"), "")) Then Exit Function
    End If
    For Each year In draftYears.Keys
        If draftYears(year)("Enabled") Then activeCount = activeCount + 1
    Next
    If activeCount = 0 And Len(comparisonPath) > 0 Then
        message = "照合を設定するには、有効な管理簿を1年度以上追加してください。"
        Exit Function
    End If
    message = "本体の設定検証を実行できません。本体のマクロが実行可能か確認してください。"
    If Not CBool(Application.Run(MacroName(book, "Configuration.LoadConfiguration"), "")) Then
        If activeCount > 0 Then Exit Function
        CheckRuntime = True
        Exit Function
    End If
    For Each year In draftYears.Keys
        If draftYears(year)("Enabled") Then
            message = year & "年度を本体が読み込めません。候補の重複、ファイルの所在、シート、見出しを確認してください。"
            Set loaded = Application.Run(MacroName(book, "Storage.ReadYear"), CStr(year), "")
            If loaded Is Nothing Then Exit Function
        End If
    Next
    If Len(comparisonPath) > 0 Then
        message = "一般事務整理簿を本体が読み込めません。照合に使う項目と見出しを確認してください。"
        Set rules = Application.Run(MacroName(book, "CaseComparison.ComparisonRules"), "")
        If rules Is Nothing Then Exit Function
        Set records = Application.Run(MacroName(book, "CaseComparison.ReadComparison"), rules, "")
        If records Is Nothing Then Exit Function
    End If
    message = ""
    CheckRuntime = True
    Exit Function
Failed:
    If Len(message) = 0 Then message = "本体の読込検証に失敗しました。"
End Function

Public Function SaveSetup(ByRef message As String, ByRef backup As String) As Boolean
    Dim book As Workbook
    Dim current As Object
    Dim expected As Object
    Dim key As Variant
    Dim parts As Variant
    Dim cell As Range
    Dim previousEvents As Boolean
    Dim previousAlerts As Boolean
    Dim previousScreen As Boolean
    Dim saveStarted As Boolean
    Dim address As String
    previousEvents = Application.EnableEvents
    previousAlerts = Application.DisplayAlerts
    previousScreen = Application.ScreenUpdating
    On Error GoTo Failed
    message = ""
    backup = ""
    If Not IsDirty() Then
        message = "保存する変更はありません。"
        Exit Function
    End If
    If Not CheckDraft(message) Then Exit Function
    address = targetPath
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.ScreenUpdating = False
    Set book = SafeOpen(address, True, message)
    If book Is Nothing Then GoTo Finished
    If Not IsSameAddress(book.FullName, address) Or Not IsController(book, message) Then GoTo Finished
    Set current = Snapshot(book)
    If Not IsSameSnapshot(originalValues, current) Then
        message = "本体の設定が先に変更されています。変更を控え、本体を選び直してください。"
        GoTo Finished
    End If
    Set expected = ExpectedSnapshot()
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        Set cell = book.Worksheets(parts(0)).Cells(CLng(parts(1)), CLng(parts(2)))
        If cell.HasFormula Or cell.MergeCells Or cell.Parent.ProtectContents Then
            message = "書込先に保護、結合セル、または数式があります。元の設定を確認してください。"
            GoTo Finished
        End If
    Next
    backup = CreateBackup(book)
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        Set cell = book.Worksheets(parts(0)).Cells(CLng(parts(1)), CLng(parts(2)))
        If Len(CStr(draftChanges(key))) = 0 Then
            cell.ClearContents
        Else
            cell.Formula = "'" & CStr(draftChanges(key))
        End If
    Next
    If Not CheckRuntime(book, message) Then GoTo Finished
    saveStarted = True
    book.Save
    If Not book.Saved Then GoTo Failed
    book.Close False
    Set book = Nothing
    Set book = SafeOpen(address, True, message)
    If book Is Nothing Then GoTo Failed
    Set current = Snapshot(book)
    If Not IsSameSnapshot(expected, current) Then GoTo Failed
    If Not CheckRuntime(book, message) Then GoTo Failed
    book.Close False
    Set book = Nothing
    If Not LoadTarget(address, message) Then GoTo Failed
    message = "設定を保存し、本体で読み込めることを確認しました。"
    If ActiveYearCount() = 0 Then message = "対象解除を保存しました。有効な管理簿はありません。"
    SaveSetup = True
Finished:
    On Error Resume Next
    If Not book Is Nothing Then book.Close False
    Application.EnableEvents = previousEvents
    Application.DisplayAlerts = previousAlerts
    Application.ScreenUpdating = previousScreen
    On Error GoTo 0
    Exit Function
Failed:
    If saveStarted Then
        message = "保存結果を確定できません。本体と退避ファイルを確認してください。" & vbCrLf & backup
    Else
        message = "設定を保存できませんでした。書込権限、設定形式、退避先を確認してください。"
    End If
    Resume Finished
End Function

Public Function ActiveYearCount() As Long
    Dim year As Variant
    For Each year In draftYears.Keys
        If draftYears(year)("Enabled") Then ActiveYearCount = ActiveYearCount + 1
    Next
End Function

Public Function SupportsPresentation() As Boolean
    SupportsPresentation = (ReadCell("Fields", 1, 11) = "DisplayFormat" And ReadCell("Settings", 1, 16) = "文書名")
End Function

Public Function DraftChoices(ByVal column As Long) As Collection
    Dim result As New Collection
    Dim row As Long
    Dim value As String
    For row = 2 To 501
        value = ReadCell("Settings", row, column)
        If Len(value) > 0 Then result.Add value
    Next
    Set DraftChoices = result
End Function

Public Function StageChoices(ByVal column As Long, ByVal items As Collection, ByRef message As String) As Boolean
    Dim seen As Object
    Dim item As Variant
    Dim value As String
    Dim index As Long
    message = ""
    If Not SupportsPresentation() Or column < 16 Or column > 18 Then Exit Function
    If items.Count > 500 Then
        message = "選択肢は500件までです。"
        Exit Function
    End If
    Set seen = Dictionary()
    seen.CompareMode = vbTextCompare
    For Each item In items
        value = Trim$(CStr(item))
        If Len(value) = 0 Or Len(value) > 255 Or seen.Exists(value) Or InStr(value, vbCr) > 0 Or InStr(value, vbLf) > 0 Then
            message = "選択肢には重複のない255文字以内の値を入力してください。"
            Exit Function
        End If
        seen.Add value, True
    Next
    index = 2
    For Each item In items
        StageCell "Settings", index, column, Trim$(CStr(item))
        index = index + 1
    Next
    For index = index To 501
        StageCell "Settings", index, column, ""
    Next
    StageChoices = True
End Function

Public Function FormatChoices(ByVal id As String, ByVal fieldType As String) As Object
    Dim result As Object
    Set result = Dictionary()
    result.Add "標準（自動）", ""
    If fieldType = "Date" Then
        result.Add "日付  yyyy/mm/dd", "dateSlash"
        result.Add "日付  yyyy-mm-dd", "dateDash"
        result.Add "日付  yyyy年m月d日", "dateJapanese"
    ElseIf fieldType = "Text" Then
        result.Add "文字列（先頭ゼロ保持）", "text"
        Select Case id
            Case "PostalCode", "PhoneNumber": result.Add "数字のまま（ハイフンなし）", "digits"
            Case "TaxYear"
                result.Add "4桁表示（数字の年分のみ）", "year4"
                result.Add "整数表示（数字の年分のみ）", "integer"
        End Select
    End If
    Set FormatChoices = result
End Function

Public Function StageDisplayFormat(ByVal id As String, ByVal preset As String) As Boolean
    Dim row As Long
    Dim choices As Object
    Dim label As Variant
    If Not SupportsPresentation() Or id = "HistoryMemo" Or Len(id) = 0 Then Exit Function
    For row = 2 To 500
        If ReadCell("Fields", row, 1) = id Then
            Set choices = FormatChoices(id, ReadCell("Fields", row, 3))
            For Each label In choices.Keys
                If choices(label) = preset Then
                    StageCell "Fields", row, 11, preset
                    StageDisplayFormat = True
                    Exit Function
                End If
            Next
            Exit Function
        End If
    Next
End Function

Private Function DictionaryLabels(ByVal dictionary As Object) As Collection
    Dim result As New Collection
    Dim label As Variant
    For Each label In dictionary.Keys
        result.Add CStr(label)
    Next
    Set DictionaryLabels = result
End Function

Public Sub EditDisplayFormat()
    Dim ids As Object
    Dim choices As Object
    Dim row As Long
    Dim id As String
    Dim label As String
    Dim selected As String
    Dim current As String
    Set ids = Dictionary()
    For row = 2 To 500
        id = ReadCell("Fields", row, 1)
        If Len(id) > 0 And id <> "HistoryMemo" Then ids.Add FieldLabel(id), row
    Next
    selected = PickValue("表示形式を設定する項目を選択してください。", DictionaryLabels(ids))
    If Len(selected) = 0 Then Exit Sub
    row = ids(selected)
    id = ReadCell("Fields", row, 1)
    Set choices = FormatChoices(id, ReadCell("Fields", row, 3))
    current = "標準（自動）"
    Dim key As Variant
    For Each key In choices.Keys
        If choices(key) = ReadCell("Fields", row, 11) Then current = CStr(key)
    Next
    label = PickValue(selected & " ／ 現在 " & current & vbCrLf & "管理簿の保存型は変わりません。", DictionaryLabels(choices), current)
    If Len(label) > 0 Then StageDisplayFormat id, CStr(choices(label))
End Sub

Public Function CopyLedgerWithTax(ByVal source As String, ByVal sheetName As String, ByVal documentHeader As String, _
    ByVal destination As String, ByRef message As String) As Boolean
    Dim book As Workbook
    Dim ws As Worksheet
    Dim lastColumn As Long
    Dim index As Long
    Dim documentColumn As Long
    Dim heading As String
    Dim taxColumn As Range
    Dim oldEvents As Boolean
    Dim temporary As String
    Dim files As Object
    oldEvents = Application.EnableEvents
    Set files = Filesystem()
    On Error GoTo Failed
    message = "ローカルの .xlsx を選び、元ファイルとは別の未使用の保存先を指定してください。"
    If IsWeb(source) Or IsWeb(destination) Or IsSameAddress(source, destination) Then Exit Function
    If LCase$(Right$(source, 5)) <> ".xlsx" Or LCase$(Right$(destination, 5)) <> ".xlsx" Then Exit Function
    If Filesystem().FileExists(destination) Then Exit Function
    Application.EnableEvents = False
    Set book = SafeOpen(source, False, message)
    If book Is Nothing Then GoTo Finished
    Set ws = book.Worksheets(sheetName)
    message = "対象シートが保護されているか、文書名の見出しを一意に確認できません。"
    If ws.ProtectContents Then GoTo Finished
    lastColumn = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For index = 1 To lastColumn
        If IsError(ws.Cells(1, index).Value2) Then GoTo Finished
        heading = CStr(ws.Cells(1, index).Value2)
        If Trim$(Replace(heading, "　", " ")) = "税目" Then
            message = "この管理簿には税目列が既にあります。"
            GoTo Finished
        End If
        If heading = documentHeader Then
            If documentColumn > 0 Or ws.Cells(1, index).MergeCells Then GoTo Finished
            documentColumn = index
        End If
    Next
    If documentColumn = 0 Then GoTo Finished
    Set taxColumn = ws.Columns(documentColumn + 1)
    taxColumn.Insert Shift:=xlToRight, CopyOrigin:=xlFormatFromRightOrBelow
    ws.Cells(1, documentColumn + 1).Value2 = "税目"
    ws.Cells(2, documentColumn + 1).Resize(Application.Max(1, SheetLastRow(ws, "A:XFD") - 1), 1).NumberFormat = "@"
    If CStr(ws.Cells(1, documentColumn).Value2) <> documentHeader Then GoTo Finished
    If Filesystem().FileExists(destination) Then GoTo Finished
    temporary = files.BuildPath(files.GetParentFolderName(destination), files.GetTempName() & ".xlsx")
    book.SaveCopyAs temporary
    files.MoveFile temporary, destination
    If Not Filesystem().FileExists(destination) Then GoTo Finished
    CopyLedgerWithTax = True
    message = "税目列を文書名の直後に追加した作業コピーを保存しました。" & vbCrLf & destination
Finished:
    On Error Resume Next
    If Not book Is Nothing Then book.Close False
    Application.EnableEvents = oldEvents
    If Len(temporary) > 0 Then
        If files.FileExists(temporary) Then files.DeleteFile temporary
    End If
    Exit Function
Failed:
    message = "作業コピーを作成できませんでした。シート名、保存先、ファイルの権限を確認してください。"
    Resume Finished
End Function
