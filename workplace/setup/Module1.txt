Public Sub InstallControllerSettings()
    Dim book As Workbook
    Dim page As Worksheet
    Dim preferences As Worksheet
    Dim button As Shape
    Dim phase As String
    On Error GoTo Failed
    Set book = ThisWorkbook
    phase = "空ブックの検証"
    If book.ReadOnly Or book.Worksheets.Count <> 1 Or book.Sheets.Count <> 1 Then Err.Raise 5
    Set page = book.Worksheets(1)
    If page.Shapes.Count <> 0 Or Application.CountA(page.UsedRange) <> 0 Or book.Names.Count <> 0 Then Err.Raise 5
    If page.Name = "設定" Or page.Name = "Preferences" Then Err.Raise 5
    phase = "設定シートの構築"
    page.Name = "設定"
    page.Range("B2:L4").Merge
    page.Range("B2").Value2 = "CONTROLLER 設定"
    page.Range("B2:L4").Interior.Color = &H432C16
    page.Range("B2:L4").Font.Color = vbWhite
    page.Range("B2:L12").Font.Name = "Meiryo UI"
    page.Range("B2").Font.Size = 20
    page.Range("B6").Value2 = "ファイルを選ぶと、年度・シート・項目を読み取ります。"
    Set button = page.Shapes.AddShape(msoShapeRoundedRectangle, 65, 117, 170, 35)
    button.Name = "OpenSetupButton"
    button.TextFrame2.TextRange.Text = "設定を開く"
    button.TextFrame2.TextRange.Font.Name = "Meiryo UI"
    button.TextFrame2.TextRange.Font.Size = 12
    button.Fill.ForeColor.RGB = &H92642E
    button.OnAction = "'" & Replace(book.Name, "'", "''") & "'!OpenSetup"
    Set preferences = book.Worksheets.Add(Before:=page)
    preferences.Name = "Preferences"
    preferences.Range("A1").Value2 = "既定の設定対象"
    preferences.Visible = xlSheetHidden
    page.Activate
    book.Windows(1).DisplayGridlines = False
    phase = "フォームと設定の検証"
    VerifyInstalledSettings
    MsgBox "CONTROLLER設定の構築と構成確認が完了しました。", vbInformation
    Exit Sub
Failed:
    MsgBox "導入不合格。工程 " & phase & "、エラー番号 " & Err.Number & " " & Err.Source & " " & Err.Description & _
        "。ブックを保存せず閉じてください。", vbCritical
End Sub

Private Sub VerifyInstalledSettings()
    Dim book As Workbook
    Set book = ThisWorkbook
    If book.Worksheets.Count <> 2 Then Err.Raise 5, , "sheet count"
    If Replace(book.Worksheets("設定").Shapes("OpenSetupButton").OnAction, "'", "") <> _
        Replace(book.Name, "'", "") & "!OpenSetup" Then Err.Raise 5, , "マクロ割当"
    If book.Worksheets("Preferences").Visible <> xlSheetHidden Then Err.Raise 5, , "visibility"
    Load UserForm1
    Load UserForm2
    Load UserForm3
    If UserForm1.Controls.Count <> 17 Or UserForm2.Controls.Count <> 4 Or UserForm3.Controls.Count <> 10 Then Err.Raise 5, , "control counts"
    Unload UserForm3
    Unload UserForm2
    Unload UserForm1
End Sub

Public Function SheetLastRow(ByVal sheet As Worksheet, ByVal columns As String) As Long
    Dim found As Range
    Set found = sheet.Range(columns).Find(What:="*", After:=sheet.Range(columns).Cells(1, 1), LookIn:=xlFormulas, _
        LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    SheetLastRow = 1
    If Not found Is Nothing Then SheetLastRow = found.Row
End Function

Public Function FileLabel(ByVal address As String) As String
    address = Replace(address, "\", "/")
    FileLabel = Mid$(address, InStrRev(address, "/") + 1)
End Function

Public Function IsWeb(ByVal address As String) As Boolean
    IsWeb = LCase$(Left$(address, 8)) = "https://"
End Function

Public Function IsDirectUrl(ByVal address As String) As Boolean
    IsDirectUrl = IsHttpsAddress(address) And Len(CanonicalAddress(address)) > 0
End Function

Public Function IsWorkbookUrl(ByVal address As String) As Boolean
    IsWorkbookUrl = IsHttpsAddress(address) And IsDirectWorkbookAddress(address)
End Function

Public Function IsSameAddress(ByVal first As String, ByVal second As String) As Boolean
    If IsWeb(first) Or IsWeb(second) Then
        IsSameAddress = IsSameWorkbookAddress(first, second)
    Else
        IsSameAddress = StrComp(first, second, vbTextCompare) = 0
    End If
End Function

Public Function SafeOpen(ByVal address As String, ByVal writableTarget As Boolean, ByRef message As String) As Workbook
    Set SafeOpen = OpenAddressWorkbook(address, Not writableTarget, writableTarget, message)
End Function

Public Function HeadersMatch(ByVal sheet As Worksheet, ByVal firstColumn As Long, ByVal expected As Variant) As Boolean
    Dim index As Long
    For index = LBound(expected) To UBound(expected)
        If CStr(sheet.Cells(1, firstColumn + index).Value2) <> CStr(expected(index)) Then Exit Function
    Next
    HeadersMatch = True
End Function

Public Function IsController(ByVal book As Workbook, ByRef message As String) As Boolean
    Dim settings As Worksheet
    Dim fields As Worksheet
    Dim comparison As Worksheet
    Dim other As Worksheet
    On Error GoTo Invalid
    If book.FileFormat <> xlOpenXMLWorkbookMacroEnabled Or book.Date1904 Then GoTo Invalid
    Set settings = book.Worksheets("Settings")
    Set fields = book.Worksheets("Fields")
    Set comparison = book.Worksheets("Comparison")
    Set other = book.Worksheets("検索")
    Set other = book.Worksheets("詳細")
    If Not HeadersMatch(settings, 1, Array("Setting", "Value")) Then GoTo Invalid
    If Not HeadersMatch(settings, 4, Array("Year", "（未使用）", "Sheet", "Enabled", "AllowWrite", "（未使用）")) Then GoTo Invalid
    If Not HeadersMatch(fields, 1, Array("FieldId", "Header", "Type", "Sheet", "Cell", "Required")) Then GoTo Invalid
    If Not HeadersMatch(fields, 8, Array("Year", "FieldId", "HeaderOverride")) Then GoTo Invalid
    If CStr(comparison.Range("A2").Value2) <> "Address" Or CStr(comparison.Range("A3").Value2) <> "Sheet" Then GoTo Invalid
    If CStr(book.Worksheets("Recovery").Range("B1").Value2) <> "" Or _
        CStr(book.Worksheets("MoveRecovery").Range("B1").Value2) <> "" Then
        message = "本体で前回の保存・移動結果を確認してから設定してください。"
        Exit Function
    End If
    IsController = True
    Exit Function
Invalid:
    message = "対応するCONTROLLERの設定構造が見つかりません。統合版の .xlsm を選択してください。"
End Function

Public Function Snapshot(ByVal book As Workbook) As Object
    Dim result As Object
    Dim sheetName As Variant
    Dim sheet As Worksheet
    Dim columnCount As Long
    Dim rowIndex As Long
    Dim colIndex As Long
    Dim area As Range
    Dim formulas As Variant
    Dim values As Variant
    Dim value As Variant
    Set result = Dictionary()
    For Each sheetName In Array("Settings", "Fields", "Comparison")
        Set sheet = book.Worksheets(CStr(sheetName))
        Select Case sheetName
            Case "Settings": columnCount = 25
            Case "Fields": columnCount = 11
            Case "Comparison": columnCount = 6
        End Select
        Set area = sheet.Range(sheet.Cells(1, 1), sheet.Cells(Application.Max(7, SheetLastRow(sheet, "A:" & _
            Split(sheet.Cells(1, columnCount).Address, "$")(1))), columnCount))
        'Formulaは数値定数を文字列で返すことがある。定数の型はValue2で保持する。
        formulas = area.Formula
        values = area.Value2
        For rowIndex = 1 To UBound(values, 1)
            For colIndex = 1 To columnCount
                value = values(rowIndex, colIndex)
                If VarType(formulas(rowIndex, colIndex)) = vbString Then
                    If Left$(formulas(rowIndex, colIndex), 1) = "=" Then
                        If area.Cells(rowIndex, colIndex).HasFormula Then value = formulas(rowIndex, colIndex)
                    End If
                End If
                If IsError(value) Then Err.Raise 5, , "設定シートにExcelエラーがあります。"
                If Len(CStr(value)) > 0 Then
                    result.Add CStr(sheetName) & "|" & rowIndex & "|" & colIndex, value
                End If
            Next
        Next
    Next
    Set Snapshot = result
End Function

Public Function ReadCell(ByVal sheetName As String, ByVal row As Long, ByVal column As Long) As String
    Dim key As String
    key = sheetName & "|" & row & "|" & column
    If draftChanges.Exists(key) Then
        ReadCell = CStr(draftChanges(key))
    ElseIf originalValues.Exists(key) Then
        ReadCell = CStr(originalValues(key))
    End If
End Function

Public Sub StageCell(ByVal sheetName As String, ByVal row As Long, ByVal column As Long, ByVal value As String)
    Dim key As String
    Dim original As String
    key = sheetName & "|" & row & "|" & column
    If originalValues.Exists(key) Then original = CStr(originalValues(key))
    If value = original Then
        If draftChanges.Exists(key) Then draftChanges.Remove key
    Else
        draftChanges(key) = value
    End If
End Sub

Public Function ConfigRow(ByVal name As String) As Long
    Dim key As Variant
    Dim parts As Variant
    For Each key In originalValues.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Settings" And parts(2) = "1" Then
            If originalValues(key) = name Then
                ConfigRow = CLng(parts(1))
                Exit Function
            End If
        End If
    Next
    Err.Raise 5, , "必要な運用設定が本体にありません。"
End Function

Public Function IsDirty() As Boolean
    If Not draftChanges Is Nothing Then IsDirty = draftChanges.Count > 0
End Function

Public Function LoadTarget(ByVal address As String, ByRef message As String) As Boolean
    Dim book As Workbook
    Dim sheet As Worksheet
    Dim row As Long
    Dim fieldId As String
    Dim year As String
    Dim localYears As Object
    Dim localFields As Object
    Dim localValues As Object
    Dim item As Object
    On Error GoTo Failed
    message = ""
    Set book = SafeOpen(address, False, message)
    If book Is Nothing Then Exit Function
    If Not IsController(book, message) Then GoTo Finished
    Set localValues = Snapshot(book)
    Set localFields = Dictionary()
    Set localYears = Dictionary()
    Set sheet = book.Worksheets("Fields")
    For row = 2 To SheetLastRow(sheet, "A:A")
        fieldId = CStr(sheet.Cells(row, 1).Value2)
        If Len(fieldId) > 0 Then
            localFields.Add fieldId, Array(CStr(sheet.Cells(row, 2).Value2), _
                sheet.Cells(row, 6).Value2 = "Yes" Or fieldId = "CommonNumber", CStr(sheet.Cells(row, 3).Value2))
        End If
    Next
    If Not localFields.Exists("CommonNumber") Or Not localFields.Exists("PersonName") Then
        message = "本体の必須項目の定義が不足しています。"
        GoTo Finished
    End If
    Set sheet = book.Worksheets("Settings")
    For row = 2 To SheetLastRow(sheet, "D:I")
        year = CStr(sheet.Cells(row, 4).Value2)
        If Len(year) > 0 Then
            If localYears.Exists(year) Then
                message = year & "年度が本体の設定で重複しています。重複を整理してから開き直してください。"
                GoTo Finished
            End If
            Set item = NewYearItem(row)
            item("Staff") = CStr(sheet.Range("B7").Value2)
            item("Sheet") = CStr(sheet.Cells(row, 6).Value2)
            item("Enabled") = (sheet.Cells(row, 7).Value2 = "Yes")
            item("Write") = CStr(sheet.Cells(row, 8).Value2)
            item("General") = CStr(sheet.Range("B8").Value2)
            localYears.Add year, item
        End If
    Next
    Set sheet = book.Worksheets("Comparison")
    comparisonPath = CStr(sheet.Range("B2").Value2)
    comparisonSheet = CStr(sheet.Range("B3").Value2)
    targetPath = book.FullName
    Set fieldSpecs = localFields
    Set draftYears = localYears
    Set originalValues = localValues
    Set draftChanges = Dictionary()
    LoadTarget = True
Finished:
    If Not book Is Nothing Then book.Close False
    Exit Function
Failed:
    message = "既存設定を読めませんでした。形式・重複・数式エラーを確認してください。"
    Resume Finished
End Function

Public Function NewYearItem(ByVal row As Long) As Object
    Dim item As Object
    Set item = Dictionary()
    item.Add "Row", row
    item.Add "Staff", ""
    item.Add "General", ""
    item.Add "Sheet", ""
    item.Add "Enabled", False
    item.Add "Write", "No"
    item.Add "State", ""
    item.Add "Count", 0
    item.Add "Unmapped", ""
    Set NewYearItem = item
End Function

Public Function NextYearRow() As Long
    Dim year As Variant
    NextYearRow = 2
    For Each year In draftYears.Keys
        If draftYears(year)("Row") >= NextYearRow Then NextYearRow = draftYears(year)("Row") + 1
    Next
    Do While Len(ReadCell("Settings", NextYearRow, 5) & ReadCell("Settings", NextYearRow, 6) & _
        ReadCell("Settings", NextYearRow, 7) & ReadCell("Settings", NextYearRow, 8) & ReadCell("Settings", NextYearRow, 9)) > 0
        NextYearRow = NextYearRow + 1
    Loop
End Function

Public Function SelectFiles(ByVal title As String, ByVal controllerOnly As Boolean, ByVal multiple As Boolean) As Collection
    Dim dialog As Object
    Dim result As New Collection
    Dim item As Variant
    Set dialog = Application.FileDialog(3)
    With dialog
        .Title = title
        .AllowMultiSelect = multiple
        If Len(targetPath) > 0 Then
            .InitialFileName = Filesystem().GetParentFolderName(targetPath) & "\"
        Else
            .InitialFileName = Filesystem().GetParentFolderName(ThisWorkbook.Path) & "\"
        End If
        .Filters.Clear
        If controllerOnly Then
            .Filters.Add "CONTROLLER", "*.xlsm"
        Else
            .Filters.Add "Excelファイル", "*.xlsx;*.xlsm;*.xlsb;*.xls"
        End If
        If .Show = -1 Then
            For Each item In .SelectedItems
                result.Add CStr(item)
            Next
        End If
    End With
    Set SelectFiles = result
End Function

Public Function PrepareComparison(ByVal address As String, ByRef message As String) As Boolean
    Dim table As Object
    Dim existing As Object
    Dim configured As Object
    Dim headers As Object
    Dim yearKeys As Variant
    Dim key As Variant
    Dim row As Long
    Dim id As String
    Dim heading As String
    Dim picked As String
    Dim values As New Collection
    Dim labels As Object
    Set table = ReadTable(address, comparisonSheet, False, message)
    If table Is Nothing Then Exit Function
    Set headers = table("Headers")
    Set existing = Dictionary()
    Set configured = Dictionary()
    For row = 8 To LastComparisonRow()
        id = ReadCell("Comparison", row, 1)
        If Len(id) > 0 Then
            heading = ReadCell("Comparison", row, 2)
            existing.Add id, Array(heading, ReadCell("Comparison", row, 3), ReadCell("Comparison", row, 4), ReadCell("Comparison", row, 5))
        End If
    Next
    For Each key In existing.Keys
        heading = CStr(existing(key)(0))
        If Not headers.Exists(heading) Then heading = SelectExternalHeading(CStr(key), headers, configured)
        If Len(heading) = 0 Then Exit Function
        configured.Add CStr(key), Array(heading, existing(key)(1), existing(key)(2), existing(key)(3))
    Next
    If configured.Count = 0 Then
        yearKeys = Array("SourceYear", "CommonNumber")
        For Each key In yearKeys
            heading = SelectExternalHeading(CStr(key), headers, configured)
            If Len(heading) = 0 Then Exit Function
            configured.Add CStr(key), Array(heading, "Yes", "No", "")
        Next
        Set labels = Dictionary()
        For Each key In fieldSpecs.Keys
            If Not configured.Exists(key) Then
                id = FieldLabel(CStr(key))
                If Not labels.Exists(id) Then
                    labels.Add id, CStr(key)
                    values.Add id
                End If
            End If
        Next
        picked = PickValue("最初に比較する管理簿の項目を選んでください。", values)
        If Len(picked) = 0 Then Exit Function
        id = CStr(labels(picked))
        heading = SelectExternalHeading(id, headers, configured)
        If Len(heading) = 0 Then Exit Function
        configured.Add id, Array(heading, "No", "No", "")
    End If
    PrepareComparison = StageComparison(table, configured, message)
End Function

Private Function SelectExternalHeading(ByVal fieldId As String, ByVal headers As Object, ByVal configured As Object) As String
    Dim items As New Collection
    Dim heading As Variant
    Dim id As Variant
    Dim used As Boolean
    For Each heading In headers.Keys
        used = False
        For Each id In configured.Keys
            If configured(id)(0) = heading Then used = True
        Next
        If Not used Then items.Add CStr(heading)
    Next
    SelectExternalHeading = PickValue("整理簿で「" & FieldLabel(fieldId) & "」に当たる見出しを選んでください。", items)
End Function

Public Function LastComparisonRow() As Long
    Dim key As Variant
    Dim parts As Variant
    LastComparisonRow = 7
    For Each key In originalValues.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Comparison" And CLng(parts(1)) > LastComparisonRow Then LastComparisonRow = CLng(parts(1))
    Next
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        If parts(0) = "Comparison" And CLng(parts(1)) > LastComparisonRow Then LastComparisonRow = CLng(parts(1))
    Next
End Function

Public Function StageComparison(ByVal table As Object, ByVal rules As Object, ByRef message As String) As Boolean
    Dim headers As Object
    Dim used As Object
    Dim id As Variant
    Dim row As Long
    Dim nextRow As Long
    Dim column As Long
    Dim keyCount As Long
    Dim compareCount As Long
    Set headers = table("Headers")
    Set used = Dictionary()
    For Each id In rules.Keys
        If Not headers.Exists(rules(id)(0)) Or used.Exists(rules(id)(0)) Then
            message = "整理簿の見出しが見つからないか、複数の項目に使われています。"
            Exit Function
        End If
        used.Add rules(id)(0), True
        If rules(id)(1) = "Yes" Then
            keyCount = keyCount + 1
        Else
            compareCount = compareCount + 1
        End If
    Next
    If keyCount = 0 Or compareCount = 0 Then
        message = "照合に使う識別項目と、比較する項目を指定してください。"
        Exit Function
    End If
    nextRow = LastComparisonRow() + 1
    For Each id In rules.Keys
        For row = 8 To LastComparisonRow()
            If ReadCell("Comparison", row, 1) = CStr(id) Then Exit For
        Next
        If row > LastComparisonRow() Then
            row = nextRow
            nextRow = nextRow + 1
        End If
        StageCell "Comparison", row, 1, CStr(id)
        For column = 2 To 5
            StageCell "Comparison", row, column, CStr(rules(id)(column - 2))
        Next
    Next
    If Not IsSameAddress(comparisonPath, CStr(table("Address"))) Or comparisonSheet <> CStr(table("Sheet")) Then
        StageCell "Comparison", 4, 2, "No"
    End If
    comparisonPath = CStr(table("Address"))
    comparisonSheet = CStr(table("Sheet"))
    StageCell "Comparison", 2, 2, comparisonPath
    StageCell "Comparison", 3, 2, comparisonSheet
    StageComparison = True
End Function

Public Function FieldLabel(ByVal fieldId As String) As String
    If fieldId = "CommonNumber" Then
        FieldLabel = "共通番号"
    ElseIf fieldId = "SubmissionFormat" Then
        FieldLabel = "提出形式"
    ElseIf fieldId = "HistoryMemo" Then
        FieldLabel = "処理履歴メモ"
    ElseIf fieldId = "SourceYear" Then
        FieldLabel = "受付年度"
    Else
        FieldLabel = CStr(fieldSpecs(fieldId)(0))
        If FieldLabel = ABSENT Then FieldLabel = "その他の項目"
    End If
End Function

Private Function NormalHeader(ByVal heading As String) As String
    NormalHeader = Trim$(Replace(Replace(heading, ChrW(&H3000), " "), vbTab, " "))
End Function

Private Function AliasHeader(ByVal fieldId As String) As String
    Select Case fieldId
        Case "CommonNumber": AliasHeader = "CommonNumber"
        Case "Furigana": AliasHeader = "氏名(フリガナ)"
        Case "SubmissionFormat": AliasHeader = "提出形式"
    End Select
End Function

Private Function UniqueHeading(ByVal headers As Object, ByVal heading As String) As String
    Dim key As Variant
    If Len(heading) = 0 Or heading = ABSENT Then Exit Function
    If headers.Exists(heading) Then
        UniqueHeading = heading
        Exit Function
    End If
    For Each key In headers.Keys
        If NormalHeader(CStr(key)) = NormalHeader(heading) Then
            If Len(UniqueHeading) > 0 Then
                UniqueHeading = ""
                Exit Function
            End If
            UniqueHeading = CStr(key)
        End If
    Next
End Function

Private Function HeaderList(ByVal sheet As Worksheet, ByRef message As String) As Object
    Dim result As Object
    Dim last As Range
    Dim index As Long
    Dim value As Variant
    Set result = Dictionary()
    Set last = sheet.Rows(1).Find(What:="*", After:=sheet.Cells(1, 1), LookIn:=xlFormulas, LookAt:=xlPart, _
        SearchOrder:=xlByColumns, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    If last Is Nothing Then
        message = sheet.Name & " の1行目に見出しがありません。"
        Exit Function
    End If
    For index = 1 To last.Column
        value = sheet.Cells(1, index).Value2
        If IsError(value) Then
            message = sheet.Name & " の1行目にExcelエラーがあります。"
            Exit Function
        End If
        If Len(CStr(value)) > 0 Then
            If result.Exists(CStr(value)) Then
                message = sheet.Name & " の見出し「" & CStr(value) & "」が重複しています。"
                Exit Function
            End If
            result.Add CStr(value), index
        End If
    Next
    If result.Count < 2 Then
        message = sheet.Name & " の1行目に複数の見出しが必要です。"
        Exit Function
    End If
    Set HeaderList = result
End Function

Private Function LooksLikeLedger(ByVal headers As Object) As Boolean
    Dim id As Variant
    For Each id In Array("CommonNumber", "PersonName")
        If Len(UniqueHeading(headers, CStr(fieldSpecs(id)(0)))) = 0 Then
            If Len(UniqueHeading(headers, AliasHeader(CStr(id)))) = 0 Then Exit Function
        End If
    Next
    LooksLikeLedger = True
End Function

Public Function ReadTable(ByVal address As String, ByVal preferredSheet As String, ByVal isLedger As Boolean, ByRef message As String) As Object
    Dim book As Workbook
    Dim sheet As Worksheet
    Dim selected As Worksheet
    Dim headerMap As Object
    Dim result As Object
    Dim plausible As New Collection
    Dim names As New Collection
    Dim name As Variant
    Dim ignore As String
    On Error GoTo Failed
    message = ""
    If IsWeb(address) And Not IsWorkbookUrl(address) Then
        message = "共有リンクではなく、拡張子までの直接URLを入力してください。"
        Exit Function
    End If
    If IsSameAddress(address, targetPath) Or IsSameAddress(address, ThisWorkbook.FullName) Then
        message = "設定ブックと本体は管理簿として選べません。"
        Exit Function
    End If
    Set book = SafeOpen(address, False, message)
    If book Is Nothing Then Exit Function
    If book.Date1904 Then
        message = "1904年日付形式のファイルは本体で読み込めません。"
        GoTo Finished
    End If
    For Each sheet In book.Worksheets
        names.Add sheet.Name
        If sheet.Name = preferredSheet Then Set selected = sheet
        ignore = ""
        Set headerMap = HeaderList(sheet, ignore)
        If Not headerMap Is Nothing Then
            If Not isLedger Or LooksLikeLedger(headerMap) Then plausible.Add sheet.Name
        End If
    Next
    If selected Is Nothing Then
        If names.Count = 1 Then
            Set selected = book.Worksheets(CStr(names(1)))
        ElseIf plausible.Count = 1 Then
            Set selected = book.Worksheets(CStr(plausible(1)))
        Else
            If plausible.Count > 1 Then Set names = plausible
            name = PickValue("読み込むシートを選択してください。", names)
            If Len(CStr(name)) = 0 Then GoTo Finished
            Set selected = book.Worksheets(CStr(name))
        End If
    End If
    Set headerMap = HeaderList(selected, message)
    If headerMap Is Nothing Then GoTo Finished
    Set result = Dictionary()
    result.Add "Address", address
    result.Add "Sheet", selected.Name
    result.Add "Headers", headerMap
    result.Add "Count", SheetLastRow(selected, "A:XFD") - 1
    Set ReadTable = result
Finished:
    If Not book Is Nothing Then book.Close False
    Exit Function
Failed:
    message = FileLabel(address) & " のシートまたは見出しを読み取れません。"
    Resume Finished
End Function

Private Function IsSameSnapshot(ByVal before As Object, ByVal after As Object) As Boolean
    Dim key As Variant
    If before.Count <> after.Count Then Exit Function
    For Each key In before.Keys
        If Not after.Exists(key) Then Exit Function
        If VarType(before(key)) <> VarType(after(key)) Then Exit Function
        If StrComp(CStr(before(key)), CStr(after(key)), vbBinaryCompare) <> 0 Then Exit Function
    Next
    IsSameSnapshot = True
End Function

Private Function ExpectedSnapshot() As Object
    Dim result As Object
    Dim key As Variant
    Set result = Dictionary()
    For Each key In originalValues.Keys
        result.Add key, originalValues(key)
    Next
    For Each key In draftChanges.Keys
        If Len(CStr(draftChanges(key))) = 0 Then
            If result.Exists(key) Then result.Remove key
        Else
            result(key) = draftChanges(key)
        End If
    Next
    Set ExpectedSnapshot = result
End Function

Private Function CreateBackup(ByVal book As Workbook) As String
    Dim files As Object
    Dim folder As String
    Dim path As String
    Dim suffix As Long
    If IsWeb(book.FullName) Then
        folder = ReadCell("Settings", ConfigRow("BackupFolder"), 2)
        If Not IsDirectUrl(folder) Then Err.Raise 5, , "SharePointの設定バックアップ先を指定してください。"
        If Right$(folder, 1) = "/" Then folder = Left$(folder, Len(folder) - 1)
        Randomize
        path = folder & "/CONTROLLER_config_" & Format$(Now, "yyyymmdd_hhnnss") & "_" & Format$(Rnd() * 100000000, "00000000") & ".xlsm"
        book.SaveCopyAs path
        Dim backupBook As Workbook
        Dim message As String
        Set backupBook = SafeOpen(path, False, message)
        If backupBook Is Nothing Then Err.Raise 5, , message
        Dim matches As Boolean
        matches = IsSameSnapshot(Snapshot(book), Snapshot(backupBook))
        backupBook.Close False
        If Not matches Then Err.Raise 5, , "設定のバックアップ内容が一致しません。"
        CreateBackup = path
        Exit Function
    End If
    Set files = Filesystem()
    folder = files.BuildPath(book.Path, "設定バックアップ")
    If Not files.FolderExists(folder) Then files.CreateFolder folder
    Do
        path = files.BuildPath(folder, files.GetBaseName(book.Name) & "_" & Format$(Now, "yyyymmdd_hhnnss") & "_" & suffix & ".xlsm")
        suffix = suffix + 1
    Loop While files.FileExists(path)
    book.SaveCopyAs path
    If Not files.FileExists(path) Then Err.Raise 5, , "退避先にブックを作成できません。"
    If files.GetFile(path).Size = 0 Then Err.Raise 5, , "退避したブックが空です。"
    CreateBackup = path
End Function

Private Function CheckDraft(ByRef message As String) As Boolean
    Dim staff As String
    Dim general As String
    Dim staffFolder As String
    Dim generalFolder As String
    Dim transport As String
    Dim backup As String
    transport = ReadCell("Settings", ConfigRow("Transport"), 2)
    staff = ReadCell("Settings", ConfigRow("StaffLedger"), 2)
    general = ReadCell("Settings", ConfigRow("GeneralLedger"), 2)
    staffFolder = ReadCell("Settings", ConfigRow("StaffFolder"), 2)
    generalFolder = ReadCell("Settings", ConfigRow("GeneralFolder"), 2)
    backup = ReadCell("Settings", ConfigRow("BackupFolder"), 2)
    message = "常勤・非常勤の異なるフォルダと管理簿.xlsmの直接アドレスを確認してください。"
    If Not IsDirectWorkbookAddress(staff) Or Not IsDirectWorkbookAddress(general) Then Exit Function
    If AddressFileName(staff) <> "管理簿.xlsm" Or AddressFileName(general) <> "管理簿.xlsm" Then Exit Function
    If IsSameWorkbookAddress(staffFolder, generalFolder) Then Exit Function
    If Not IsSameWorkbookAddress(AddressFolder(staff), staffFolder) Then Exit Function
    If Not IsSameWorkbookAddress(AddressFolder(general), generalFolder) Then Exit Function

    ElseIf transport = "Local" Then
        If IsWeb(staff) Or IsWeb(general) Or IsWeb(targetPath) Then Exit Function
    Else
        Exit Function
    End If
    message = ""
    CheckDraft = True
End Function

Private Function CheckRuntime(ByVal book As Workbook, ByRef message As String) As Boolean
    Dim year As Variant
    Dim loaded As Object
    Dim rules As Object
    Dim records As Object
    Dim activeCount As Long
    On Error GoTo Failed
    If CStr(book.Worksheets("Fields").Cells(1, 11).Value2) = "DisplayFormat" Then
        message = "表示形式または選択肢が無効です。設定を確認してください。"
        If Not CBool(Application.Run(MacroName(book, "Presentation.ValidatePresentation"), "")) Then Exit Function
    End If
    For Each year In draftYears.Keys
        If draftYears(year)("Enabled") Then activeCount = activeCount + 1
    Next
    If activeCount = 0 And Len(comparisonPath) > 0 Then
        message = "照合を設定するには、有効な管理簿を1年度以上追加してください。"
        Exit Function
    End If
    message = "本体の設定検証を実行できません。本体のマクロが実行可能か確認してください。"
    If Not CBool(Application.Run(MacroName(book, "Configuration.LoadConfiguration"), "")) Then
        If activeCount > 0 Then Exit Function
        CheckRuntime = True
        Exit Function
    End If
    For Each year In draftYears.Keys
        If draftYears(year)("Enabled") Then
            message = year & "年度を本体が読み込めません。候補の重複、ファイルの所在、シート、見出しを確認してください。"
            Set loaded = Application.Run(MacroName(book, "Storage.ReadYear"), CStr(year), "")
            If loaded Is Nothing Then Exit Function
        End If
    Next
    If Len(comparisonPath) > 0 Then
        message = "一般事務整理簿を本体が読み込めません。照合に使う項目と見出しを確認してください。"
        Set rules = Application.Run(MacroName(book, "CaseComparison.ComparisonRules"), "")
        If rules Is Nothing Then Exit Function
        Set records = Application.Run(MacroName(book, "CaseComparison.ReadComparison"), rules, "")
        If records Is Nothing Then Exit Function
    End If
    message = ""
    CheckRuntime = True
    Exit Function
Failed:
    If Len(message) = 0 Then message = "本体の読込検証に失敗しました。"
End Function

Public Function SaveSetup(ByRef message As String, ByRef backup As String) As Boolean
    Dim book As Workbook
    Dim current As Object
    Dim expected As Object
    Dim key As Variant
    Dim parts As Variant
    Dim cell As Range
    Dim previousEvents As Boolean
    Dim previousAlerts As Boolean
    Dim previousScreen As Boolean
    Dim saveStarted As Boolean
    Dim address As String
    previousEvents = Application.EnableEvents
    previousAlerts = Application.DisplayAlerts
    previousScreen = Application.ScreenUpdating
    On Error GoTo Failed
    message = ""
    backup = ""
    If Not IsDirty() Then
        message = "保存する変更はありません。"
        Exit Function
    End If
    If Not CheckDraft(message) Then Exit Function
    address = targetPath
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.ScreenUpdating = False
    Set book = SafeOpen(address, True, message)
    If book Is Nothing Then GoTo Finished
    If Not IsSameAddress(book.FullName, address) Or Not IsController(book, message) Then GoTo Finished
    Set current = Snapshot(book)
    If Not IsSameSnapshot(originalValues, current) Then
        message = "本体の設定が先に変更されています。変更を控え、本体を選び直してください。"
        GoTo Finished
    End If
    Set expected = ExpectedSnapshot()
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        Set cell = book.Worksheets(parts(0)).Cells(CLng(parts(1)), CLng(parts(2)))
        If cell.HasFormula Or cell.MergeCells Or cell.Parent.ProtectContents Then
            message = "書込先に保護、結合セル、または数式があります。元の設定を確認してください。"
            GoTo Finished
        End If
    Next
    backup = CreateBackup(book)
    For Each key In draftChanges.Keys
        parts = Split(CStr(key), "|")
        Set cell = book.Worksheets(parts(0)).Cells(CLng(parts(1)), CLng(parts(2)))
        If Len(CStr(draftChanges(key))) = 0 Then
            cell.ClearContents
        Else
            cell.Formula = "'" & CStr(draftChanges(key))
        End If
    Next
    If Not CheckRuntime(book, message) Then GoTo Finished
    saveStarted = True
    book.Save
    If Not book.Saved Then GoTo Failed
    book.Close False
    Set book = Nothing
    Set book = SafeOpen(address, True, message)
    If book Is Nothing Then GoTo Failed
    Set current = Snapshot(book)
    If Not IsSameSnapshot(expected, current) Then GoTo Failed
    If Not CheckRuntime(book, message) Then GoTo Failed
    book.Close False
    Set book = Nothing
    If Not LoadTarget(address, message) Then GoTo Failed
    message = "設定を保存し、本体で読み込めることを確認しました。"
    If ActiveYearCount() = 0 Then message = "対象解除を保存しました。有効な管理簿はありません。"
    SaveSetup = True
Finished:
    On Error Resume Next
    If Not book Is Nothing Then book.Close False
    Application.EnableEvents = previousEvents
    Application.DisplayAlerts = previousAlerts
    Application.ScreenUpdating = previousScreen
    On Error GoTo 0
    Exit Function
Failed:
    If saveStarted Then
        message = "保存結果を確定できません。本体と退避ファイルを確認してください。" & vbCrLf & backup
    Else
        message = "設定を保存できませんでした。書込権限、設定形式、退避先を確認してください。"
    End If
    Resume Finished
End Function

Public Function StageChoices(ByVal column As Long, ByVal items As Collection, ByRef message As String) As Boolean

    For Each item In items
        value = Trim$(CStr(item))

    index = 2

        StageCell "Settings", index, column, Trim$(CStr(item))
        index = index + 1
    Next
    For index = index To 501
        StageCell "Settings", index, column, ""
    Next
    StageChoices = True
End Function

Public Function IsHttpsAddress(ByVal address As String) As Boolean
    IsHttpsAddress = (StrComp(Left$(address, 8), "https://", vbTextCompare) = 0)
End Function

Public Function CanonicalAddress(ByVal address As String) As String
    Dim slash As Long
    Dim host As String
    Dim path As String
    Dim segment As Variant
    If Not IsHttpsAddress(address) Then
        If Mid$(address, 2, 2) = ":\" Or Left$(address, 2) = "\\" Then CanonicalAddress = Replace(address, "/", "\")
        Exit Function
    End If
    slash = InStr(9, address, "/")
    If slash = 0 Then
        host = Mid$(address, 9)
    Else
        host = Mid$(address, 9, slash - 9)
    End If
    If Len(host) = 0 Or InStr(host, ".") = 0 Or host Like "*[!A-Za-z0-9.-]*" Then Exit Function
    If InStr(address, "?") > 0 Or InStr(address, "\") > 0 Then Exit Function
    If slash = 0 Then
        CanonicalAddress = "https://" & LCase$(host)
        Exit Function
    End If
    path = DecodeAddressPath(Mid$(address, slash))
    If Len(path) = 0 Then Exit Function
    If InStr(path, "\") > 0 Or InStr(path, vbCr) > 0 Or InStr(path, vbLf) > 0 Or InStr(path, Chr$(0)) > 0 Then Exit Function
    If InStr(path, "/:") > 0 Or InStr(path, "//") > 0 Or InStr(1, path, "/_layouts/", vbTextCompare) > 0 Then Exit Function
    For Each segment In Split(path, "/")
        If segment = "." Or segment = ".." Then Exit Function
    Next
    If Right$(path, 1) = "/" Then path = Left$(path, Len(path) - 1)
    CanonicalAddress = "https://" & LCase$(host) & path
End Function

Private Function DecodeAddressPath(ByVal path As String) As String
    Dim i As Long
    Dim octet As Long
    Dim code As Long
    Dim pending As Long
    Dim minimum As Long
    Dim pair As String
    Dim result As String
    i = 1
    Do While i <= Len(path)
        pair = Mid$(path, i + 1, 2)
        If Mid$(path, i, 1) = "%" And Len(pair) = 2 And UCase$(pair) Like "[0-9A-F][0-9A-F]" Then
            octet = CLng("&H" & pair)
            If pending > 0 Then
                If octet < 128 Or octet > 191 Then Exit Function
                code = code * 64 + octet - 128
                pending = pending - 1
                If pending = 0 Then
                    If code < minimum Or code > &H10FFFF Or (code >= &HD800& And code <= &HDFFF&) Then Exit Function
                    If code > 65535 Then
                        code = code - 65536
                        result = result & ChrW(&HD800& + code \ 1024) & ChrW(&HDC00& + code Mod 1024)
                    Else
                        result = result & ChrW(code)
                    End If
                End If
            ElseIf octet < 128 Then
                '区切りをエンコードしたリンクは、別パスに解釈しない。
                If octet = 47 Or octet = 92 Then Exit Function
                result = result & ChrW(octet)
            ElseIf octet >= 194 And octet <= 223 Then
                code = octet - 192
                pending = 1
                minimum = 128
            ElseIf octet >= 224 And octet <= 239 Then
                code = octet - 224
                pending = 2
                minimum = 2048
            ElseIf octet >= 240 And octet <= 244 Then
                code = octet - 240
                pending = 3
                minimum = 65536
            Else
                Exit Function
            End If
            i = i + 3
        Else
            If pending > 0 Then Exit Function
            result = result & Mid$(path, i, 1)
            i = i + 1
        End If
    Loop
    If pending = 0 Then DecodeAddressPath = result
End Function

Public Function IsSameWorkbookAddress(ByVal first As String, ByVal second As String) As Boolean
    Dim leftAddress As String
    Dim rightAddress As String
    leftAddress = CanonicalAddress(first)
    rightAddress = CanonicalAddress(second)
    If Len(leftAddress) = 0 Or Len(rightAddress) = 0 Then Exit Function
    IsSameWorkbookAddress = (StrComp(leftAddress, rightAddress, vbTextCompare) = 0)
End Function

Public Function IsDirectWorkbookAddress(ByVal address As String) As Boolean
    Dim path As String
    Dim extension As String
    path = CanonicalAddress(address)
    If Len(path) = 0 Then Exit Function
    extension = LCase$(Mid$(path, InStrRev(path, ".") + 1))
    IsDirectWorkbookAddress = (extension = "xlsm" Or extension = "xlsx" Or extension = "xlsb" Or extension = "xls")
End Function

Public Function AddressFileName(ByVal address As String) As String
    Dim path As String
    path = Replace(CanonicalAddress(address), "\", "/")
    If Len(path) > 0 Then AddressFileName = Mid$(path, InStrRev(path, "/") + 1)
End Function

Public Function AddressFolder(ByVal address As String) As String
    Dim slash As Long
    slash = InStrRev(address, IIf(IsHttpsAddress(address), "/", "\"))
    If slash > 0 Then AddressFolder = Left$(address, slash - 1)
End Function

Public Function OpenAddressWorkbook(ByVal address As String, ByVal isReadOnly As Boolean, _
    ByVal allowTrustedMacros As Boolean, ByRef message As String) As Workbook
    Dim candidate As Workbook
    Dim book As Workbook
    Dim previousEvents As Boolean
    Dim previousSecurity As Long

    message = ""
    If Not IsDirectWorkbookAddress(address) Then
        message = "Excelファイルの直接URLまたは検証用の絶対パスを指定してください。"
        Exit Function
    End If
    For Each candidate In Application.Workbooks
        If IsSameWorkbookAddress(candidate.FullName, address) Or _
            StrComp(candidate.Name, AddressFileName(address), vbTextCompare) = 0 Then
            message = "対象または同名のブックを閉じてから操作してください。"
            Exit Function
        End If
    Next
    previousEvents = Application.EnableEvents
    previousSecurity = Application.AutomationSecurity
    On Error GoTo Failed
    Application.EnableEvents = False
    If Not allowTrustedMacros Then Application.AutomationSecurity = msoAutomationSecurityForceDisable
    Set book = Application.Workbooks.Open(Filename:=ExcelOpenAddress(address), UpdateLinks:=0, ReadOnly:=isReadOnly, _
        Notify:=False, AddToMru:=False, IgnoreReadOnlyRecommended:=True)
    If Not IsSameWorkbookAddress(book.FullName, address) Then
        message = "Excelが開いたブックのアドレスが設定と一致しません。直接URLを確認してください。"
    ElseIf Not isReadOnly And book.ReadOnly Then
        message = "読み取り専用です。編集権限と他の利用者の使用状況を確認してください。"
    ElseIf book.Date1904 Then
        message = "1904年日付システムのブックは対象外です。"
    Else
        If Not isReadOnly And IsHttpsAddress(address) Then
            'オートセーブが既に無効な環境では、書き込みを試みない。
            If book.AutoSaveOn Then book.AutoSaveOn = False
            If book.AutoSaveOn Then Err.Raise 5, , "自動保存を停止できません。"
        End If
        Set OpenAddressWorkbook = book
        Set book = Nothing
    End If
Finished:
    On Error Resume Next
    If Not book Is Nothing Then book.Close SaveChanges:=False
    Application.AutomationSecurity = previousSecurity
    Application.EnableEvents = previousEvents
    On Error GoTo 0
    Exit Function
Failed:
    message = "Excelで開けません。直接URL、Officeのサインイン、権限、使用状況を確認してください。エラー番号 " & Err.Number
    Resume Finished
End Function

Public Function ExcelOpenAddress(ByVal address As String) As String
    Dim i As Long
    Dim ch As String
    If Not IsHttpsAddress(address) Then
        ExcelOpenAddress = address
        Exit Function
    End If
    For i = 1 To Len(address)
        ch = Mid$(address, i, 1)
        If ch = "#" Then
            ch = "%23"
        ElseIf ch = "%" Then
            If Not UCase$(Mid$(address, i + 1, 2)) Like "[0-9A-F][0-9A-F]" Then ch = "%25"
        End If
        ExcelOpenAddress = ExcelOpenAddress & ch
    Next
End Function
