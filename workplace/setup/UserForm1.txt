Private Sub UserForm_Initialize()
    Dim address As String
    Dim message As String
    BuildFormControls
    StyleControls Me
    Banner.Font.Size = 17
    Banner.Font.Bold = True
    TargetName.Font.Bold = True
    SaveChanges.Font.Bold = True
    StartUpPosition = 2
    If Len(targetPath) = 0 Then
        address = CStr(ThisWorkbook.Worksheets("Preferences").Range("B1").Value2)
        If Len(address) > 0 Then
            If Filesystem().FileExists(address) Then LoadTarget address, message
        End If
    End If
    DrawEntries
    If Len(message) > 0 Then Status.Caption = message
End Sub

Private Sub DrawEntries(Optional ByVal selected As String = "")
    Dim year As Variant
    Dim item As Object
    Dim address As String
    Dim state As String
    Dim index As Long
    Entries.Clear
    If Len(targetPath) = 0 Then
        TargetName.Caption = "設定対象を選んでください"
        Status.Caption = "最初に「対象を選択」で CONTROLLER.xlsm を開きます。"
    Else
        TargetName.Caption = "設定対象   " & FileLabel(targetPath) & "   ・   " & FileLabel(Filesystem().GetParentFolderName(targetPath))
        TargetName.ControlTipText = targetPath
        For Each year In draftYears.Keys
            Set item = draftYears(year)
            If Len(item("Staff") & item("General")) > 0 Then
                address = item("Staff")
                If Len(address) = 0 Then address = item("General")
                state = CStr(item("State"))
                If Len(state) = 0 Then
                    If Not item("Enabled") Then
                        state = "対象外"
                    ElseIf Len(item("Staff")) = 0 Or Len(item("General")) = 0 Or Len(item("Sheet")) = 0 Then
                        state = "要確認"
                    Else
                        state = CurrentState(item)
                    End If
                End If
                AddEntry CStr(year) & "年度 管理簿", state, FileLabel(address) & " / " & item("Sheet"), CStr(year)
            End If
        Next
        If Len(comparisonPath) > 0 Then
            AddEntry "一般事務整理簿", "設定あり", FileLabel(comparisonPath) & " / " & comparisonSheet, "Comparison"
        Else
            AddEntry "一般事務整理簿", "未設定（任意）", "必要な場合に選んで「ファイル変更」", "Comparison"
        End If
        If IsDirty() Then
            Status.Caption = "一覧の内容を確認し、「設定を保存」を押してください。"
        ElseIf ActiveYearCount() = 0 Then
            Status.Caption = "「管理簿を追加」でExcelファイルを選んでください。複数年度をまとめて選べます。"
        Else
            Status.Caption = "変更する行を選ぶと、保存先や項目対応を確認できます。"
        End If
    End If
    For index = 0 To Entries.ListCount - 1
        If Entries.List(index, 3) = selected Then Entries.ListIndex = index
    Next
    UpdateButtons
End Sub

Private Function CurrentState(ByVal item As Object) As String
    Dim files As Object
    CurrentState = "設定済み"
    If IsWeb(item("Staff")) Then Exit Function
    Set files = Filesystem()
    If files.FileExists(item("Staff")) And files.FileExists(item("General")) Then
        CurrentState = "要確認・両方に存在"
    ElseIf Not files.FileExists(item("Staff")) And Not files.FileExists(item("General")) Then
        CurrentState = "要確認・所在不明"
    End If
End Function

Private Sub AddEntry(ByVal label As String, ByVal state As String, ByVal description As String, ByVal key As String)
    Dim index As Long
    Entries.AddItem label
    index = Entries.ListCount - 1
    Entries.List(index, 1) = state
    Entries.List(index, 2) = description
    Entries.List(index, 3) = key
End Sub

Private Function SelectedKey() As String
    If Entries.ListIndex >= 0 Then SelectedKey = CStr(Entries.List(Entries.ListIndex, 3))
End Function

Private Sub UpdateButtons()
    Dim selected As String
    selected = SelectedKey()
    AddFiles.Enabled = Len(targetPath) > 0
    Operations.Enabled = AddFiles.Enabled
    SaveChanges.Enabled = IsDirty()
    ChangeFile.Enabled = Len(selected) > 0
    RefreshFile.Enabled = Len(selected) > 0
    MatchFields.Enabled = Len(selected) > 0
    DisableYear.Enabled = Len(selected) > 0
    If selected = "Comparison" Then
        RefreshFile.Enabled = Len(comparisonPath) > 0
        MatchFields.Enabled = RefreshFile.Enabled
        DisableYear.Enabled = RefreshFile.Enabled
    ElseIf Len(selected) > 0 Then
        DisableYear.Enabled = draftYears(selected)("Enabled")
    End If
End Sub

Private Sub Entries_Change()
    Dim selected As String
    UpdateButtons
    selected = SelectedKey()
    If Len(selected) = 0 Or selected = "Comparison" Then Exit Sub
    If Len(draftYears(selected)("Unmapped")) > 0 Then
        Status.Caption = "任意項目の対応なし  " & draftYears(selected)("Unmapped") & vbCrLf & _
            "必要な項目だけ「対応を確認」から指定できます。"
    End If
End Sub

Private Sub ChooseTarget_Click()
    Dim selected As Collection
    Dim message As String
    If Not CanDiscard() Then Exit Sub
    Set selected = SelectFiles("設定するCONTROLLERを選択", True, False)
    If selected.Count = 0 Then Exit Sub
    If LoadTarget(CStr(selected(1)), message) Then
        RememberTarget
        DrawEntries
    Else
        Status.Caption = message
    End If
End Sub

Private Sub RememberTarget()
    Dim events As Boolean
    If ThisWorkbook.ReadOnly Then Exit Sub
    events = Application.EnableEvents
    On Error GoTo Finished
    Application.EnableEvents = False
    ThisWorkbook.Worksheets("Preferences").Range("B1").Value2 = targetPath
    ThisWorkbook.Save
Finished:
    Application.EnableEvents = events
End Sub

Private Sub AddFiles_Click()
    Dim files As Collection
    Dim address As Variant
    Dim message As String
    Dim summary As String
    Dim added As Long
    Set files = SelectFiles("追加する管理簿を選択（複数可）", False, True)
    For Each address In files
        message = ""
        If AddLedger(CStr(address), "", False, message) Then
            added = added + 1
        ElseIf Len(message) > 0 Then
            If Len(summary) = 0 Then summary = message
        End If
    Next
    If files.Count > 0 Then
        DrawEntries
        If Len(summary) > 0 Then
            Status.Caption = added & "件を追加しました。" & vbCrLf & summary
        ElseIf added > 0 Then
            Status.Caption = added & "件を読み取りました。内容を確認して「設定を保存」を押してください。"
        End If
    End If
End Sub

Private Sub ChangeFile_Click()
    Dim files As Collection
    Dim selected As String
    Dim message As String
    selected = SelectedKey()
    If Len(selected) = 0 Then Exit Sub
    If selected = "Comparison" Then
        Set files = SelectFiles("一般事務整理簿を選択", False, False)
    Else
        Set files = SelectFiles(selected & "年度の新しい管理簿を選択", False, False)
    End If
    If files.Count = 0 Then Exit Sub
    If selected = "Comparison" Then
        If PrepareComparison(CStr(files(1)), message) Then DrawEntries selected
    Else
        If AddLedger(CStr(files(1)), selected, False, message) Then DrawEntries selected
    End If
    If Len(message) > 0 Then Status.Caption = message
End Sub

Private Sub RefreshFile_Click()
    Recheck False

End Sub

Private Sub Recheck(ByVal review As Boolean)
    Dim selected As String
    Dim message As String
    Dim address As String
    selected = SelectedKey()
    If Len(selected) = 0 Then Exit Sub
    If selected = "Comparison" Then
        If Len(comparisonPath) = 0 Then Exit Sub
        If PrepareComparison(comparisonPath, message) Then DrawEntries selected
    Else
        address = ExistingFile(selected, message)
        If Len(address) > 0 Then
            If AddLedger(address, selected, review, message) Then
                DrawEntries selected
                If Not IsDirty() Then Status.Caption = selected & "年度の見出しを確認しました。設定の変更はありません。"
            End If
        End If
    End If
    If Len(message) > 0 Then Status.Caption = message
End Sub

Private Function AddLedger(ByVal address As String, ByVal selectedYear As String, ByVal review As Boolean, ByRef message As String) As Boolean
    Dim year As String
    Dim detected As String
    Dim answer As Variant
    Dim plan As Object
    On Error GoTo Failed
    detected = GuessYear(address)
    year = selectedYear
    If Len(year) = 0 Then
        year = detected
        If Len(year) = 0 Then
            answer = AskText(FileLabel(address) & " の受付年度を西暦4桁で指定してください。")
            If VarType(answer) = vbBoolean Then Exit Function
            year = Trim$(CStr(answer))
        End If
        If draftYears.Exists(year) Then
            If Len(draftYears(year)("Staff") & draftYears(year)("General")) > 0 Then
                message = year & "年度は設定済みです。行を選んで「ファイル変更」を使ってください。"
                Exit Function
            End If
        End If
    ElseIf Len(detected) > 0 And detected <> year Then
        If MsgBox("ファイル名から読み取った年度は " & detected & " です。" & year & "年度の管理簿として使用しますか？", _
            vbYesNo Or vbQuestion Or vbDefaultButton2) <> vbYes Then Exit Function
    End If
    If Not IsYear(year) Then
        message = "受付年度は1900～2099年の西暦4桁を指定してください。"
        Exit Function
    End If
    Set plan = PrepareLedger(address, year, message)
    If plan Is Nothing Then Exit Function
    If review Then
        If Not ReviewMapping(plan) Then Exit Function
    End If
    AddLedger = StageLedger(year, plan, message)
    Exit Function
Failed:
    message = FileLabel(address) & " の設定を作成できません。ファイルと既存の項目対応を確認してください。"
End Function

Private Sub DisableYear_Click()
    Dim selected As String
    selected = SelectedKey()
    If Len(selected) = 0 Then Exit Sub
    If MsgBox("選んだ表を読込対象から外しますか？ 元のExcelファイルは残ります。", vbYesNo Or vbQuestion Or vbDefaultButton2) <> vbYes Then Exit Sub
    If selected = "Comparison" Then
        StageCell "Comparison", 2, 2, ""
        StageCell "Comparison", 3, 2, ""
        StageCell "Comparison", 4, 2, "No"
        comparisonPath = ""
        comparisonSheet = ""
    Else
        StageDisable selected
    End If
    DrawEntries selected
End Sub

Private Sub Operations_Click()
    Dim choice As String
    Dim picked As String
    Dim row As Long
    Dim answer As Variant
    Dim message As String
    Dim selected As String
    selected = SelectedKey()
    choice = PickValue("必要な運用設定だけ変更できます。", Choices("担当区分", "選択年度の更新権限", "バックアップ先", _
        "SharePointのサイト・排他", "管理簿を直接URLで指定", "整理簿を直接URLで指定", "署の一覧", "終了状態の一覧"))
    Select Case choice
        Case "担当区分"
            picked = PickValue("この本体を使う担当区分を選んでください。", Choices("常勤", "非常勤"))
            If Len(picked) = 0 Then Exit Sub
            StageCell "Settings", ConfigRow("ControllerRole"), 2, IIf(picked = "常勤", "Staff", "General")
        Case "選択年度の更新権限"
            If Len(selected) = 0 Or selected = "Comparison" Then
                Status.Caption = "先に、更新権限を設定する管理簿の行を選んでください。"
                Exit Sub
            End If
            picked = PickValue(selected & "年度を本体から更新できるようにしますか？", Choices("読込のみ", "更新を許可"))
            If Len(picked) = 0 Then Exit Sub
            draftYears(selected)("Write") = IIf(picked = "更新を許可", "Yes", "No")
            StageCell "Settings", CLng(draftYears(selected)("Row")), 8, CStr(draftYears(selected)("Write"))
        Case "バックアップ先"
            row = ConfigRow("BackupFolder")
            If ReadCell("Settings", ConfigRow("Transport"), 2) = "SharePoint" Then
                answer = AskText("バックアップフォルダの直接URLを入力してください。", ReadCell("Settings", row, 2))
                If VarType(answer) = vbBoolean Then Exit Sub
                picked = Trim$(CStr(answer))
            Else
                picked = SelectFolder("業務開始時のバックアップ先を選択")
                If Len(picked) = 0 Then Exit Sub
            End If
            StageCell "Settings", row, 2, picked
        Case "SharePointのサイト・排他"
            row = ConfigRow("SharePointSite")
            answer = AskText("SharePointのサイトURLを入力してください。", ReadCell("Settings", row, 2))
            If VarType(answer) = vbBoolean Then Exit Sub
            If Not IsDirectUrl(Trim$(CStr(answer))) Then
                Status.Caption = "サイトの直接URLを確認してください。"
                Exit Sub
            End If
            picked = PickValue("必須チェックアウトが有効だと管理担当が確認しましたか？", Choices("未確認", "確認済み"))
            If Len(picked) = 0 Then Exit Sub
            StageCell "Settings", row, 2, Trim$(CStr(answer))
            StageCell "Settings", ConfigRow("SharePointExclusive"), 2, IIf(picked = "確認済み", "Yes", "No")
        Case "管理簿を直接URLで指定", "整理簿を直接URLで指定"
            answer = AskText("拡張子まで含む直接ファイルURLを入力してください。")
            If VarType(answer) = vbBoolean Then Exit Sub
            picked = Trim$(CStr(answer))
            If Not IsWorkbookUrl(picked) Then
                Status.Caption = "共有リンクではなく、拡張子まで含む直接URLを入力してください。"
                Exit Sub
            End If
            If choice = "整理簿を直接URLで指定" Then
                PrepareComparison picked, message
            Else
                If selected = "Comparison" Then selected = ""
                AddLedger picked, selected, False, message
            End If
        Case "署の一覧"
            EditList 14, "署名を「、」で区切って入力してください。"
        Case "終了状態の一覧"
            EditList 15, "終了として扱う状態を「、」で区切って入力してください。"
        Case Else
            Exit Sub
    End Select
    DrawEntries selected
    If Len(message) > 0 Then Status.Caption = message
End Sub

Private Sub EditList(ByVal column As Long, ByVal prompt As String)
    Dim row As Long
    Dim previous As String
    Dim answer As Variant
    Dim items As Variant
    Dim index As Long
    row = 2
    Do While Len(ReadCell("Settings", row, column)) > 0
        If Len(previous) > 0 Then previous = previous & "、"
        previous = previous & ReadCell("Settings", row, column)
        row = row + 1
    Loop
    answer = AskText(prompt, previous)
    If VarType(answer) = vbBoolean Then Exit Sub
    items = Split(CStr(answer) & "、", "、")
    For index = 0 To Application.Max(row - 3, UBound(items) - 1)
        If index < UBound(items) Then
            StageCell "Settings", index + 2, column, Trim$(CStr(items(index)))
        Else
            StageCell "Settings", index + 2, column, ""
        End If
    Next
End Sub

Private Sub PresentationSettings_Click()
    Dim choice As String
    Dim selected As String
    Dim column As Long
    If Len(targetPath) = 0 Then
        Status.Caption = "設定対象のCONTROLLERを選んでください。"
        Exit Sub
    End If
    If Not SupportsPresentation() Then
        Status.Caption = "選択肢・表示形式に対応したCONTROLLERを選んでください。"
        Exit Sub
    End If
    choice = PickValue("変更する内容を選んでください。", Choices("選択肢の設定", "表示形式の設定", "税目列の追加コピー"))
    Select Case choice
        Case "選択肢の設定"
            selected = PickValue("候補を編集する項目を選んでください。", Choices("文書名", "税目", "処理状態"))
            Select Case selected
                Case "文書名": column = 16
                Case "税目": column = 17
                Case "処理状態": column = 18
                Case Else: Exit Sub
            End Select
            UserForm3.EditChoices column
        Case "表示形式の設定"
            EditDisplayFormat
        Case "税目列の追加コピー"
            MakeTaxCopy
        Case Else
            Exit Sub
    End Select
    UpdateButtons
End Sub

Private Sub MakeTaxCopy()
    Dim selected As String
    Dim source As String
    Dim destination As Variant
    Dim item As Object
    Dim mapping As Object
    Dim table As Object
    Dim message As String
    selected = SelectedKey()
    If Not draftYears.Exists(selected) Then
        Status.Caption = "先に一覧で移行したい管理簿の年度を選んでください。"
        Exit Sub
    End If
    Set item = draftYears(selected)
    source = PickValue("原本は変更しません。税目列を追加する元の管理簿を選んでください。", Choices(CStr(item("Staff")), CStr(item("General"))))
    If Len(source) = 0 Then Exit Sub
    Set table = ReadTable(source, CStr(item("Sheet")), True, message)
    If table Is Nothing Then
        Status.Caption = message
        Exit Sub
    End If
    Set mapping = MapLedger(table("Headers"), selected)
    destination = Application.GetSaveAsFilename(InitialFileName:=Left$(source, Len(source) - 5) & "_税目追加.xlsx", _
        FileFilter:="Excel ブック (*.xlsx),*.xlsx", Title:="税目列を追加した作業コピーの保存先")
    If VarType(destination) = vbBoolean Then Exit Sub
    If MsgBox("文書名の直後に税目を追加したコピーを作成します。" & vbCrLf & CStr(destination), _
        vbYesNo Or vbQuestion Or vbDefaultButton2, "税目列の追加コピー") <> vbYes Then Exit Sub
    If CopyLedgerWithTax(source, CStr(item("Sheet")), CStr(mapping("DocumentName")), CStr(destination), message) Then
        Status.Caption = "コピーを作成しました。確認後に「ファイル変更」で指定してください。"
        Status.ControlTipText = CStr(destination)
    Else
        Status.Caption = message
    End If
End Sub

End Function
