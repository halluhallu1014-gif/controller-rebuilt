Option Explicit

Private Prompt As MSForms.Label
Private Choices As MSForms.ComboBox
Private WithEvents CancelChoice As MSForms.CommandButton
Private WithEvents AcceptChoice As MSForms.CommandButton

Private result As String

Public Function Choose(ByVal message As String, ByVal options As Collection, ByVal current As String) As String
    Dim value As Variant
    Dim index As Long
    StyleControls Me
    Prompt.Caption = message
    Choices.Clear
    For Each value In options
        Choices.AddItem CStr(value)
        If CStr(value) = current Then Choices.ListIndex = index
        index = index + 1
    Next
    If Choices.ListCount = 1 Then Choices.ListIndex = 0
    Me.Show
    Choose = result
End Function

Private Sub AcceptChoice_Click()
    If Choices.ListIndex < 0 Then Exit Sub
    result = CStr(Choices.Value)
    Me.Hide
End Sub

Private Sub CancelChoice_Click()
    Me.Hide
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = vbFormControlMenu Then
        Cancel = True
        Me.Hide
    End If
End Sub
Private Sub UserForm_Initialize()
    BuildFormControls
End Sub

Private Sub BuildFormControls()
    Dim stage As String
    On Error GoTo Failed
    Me.Caption = "選択"
    Me.Width = 468.75
    Me.Height = 207.75
    Me.BackColor = 16315889
    Me.StartUpPosition = 1

    stage = "Prompt"
    Set Prompt = Me.Controls.Add("Forms.Label.1", "Prompt", True)
    With Prompt
        .Move 16, 15, 427, 42
        .TabIndex = 0
        .BackColor = 16315889
        .ForeColor = 4404002
        .BackStyle = 0
        .BorderStyle = 0
        .WordWrap = True
    End With

    stage = "Choices"
    Set Choices = Me.Controls.Add("Forms.ComboBox.1", "Choices", True)
    With Choices
        .Move 16, 63, 427, 24
        .TabIndex = 1
        .TabStop = True
        .BackColor = -2147483643
        .ForeColor = -2147483640
        .BackStyle = 1
        .BorderStyle = 0
        .ColumnCount = 1
        .Style = 2
    End With

    stage = "CancelChoice"
    Set CancelChoice = Me.Controls.Add("Forms.CommandButton.1", "CancelChoice", True)
    With CancelChoice
        .Move 232, 130, 99, 28
        .Caption = "戻る"
        .TabIndex = 2
        .TabStop = True
        .BackColor = 15590875
        .ForeColor = 4404002
        .WordWrap = False
        .Default = False
        .Cancel = False
    End With

    stage = "AcceptChoice"
    Set AcceptChoice = Me.Controls.Add("Forms.CommandButton.1", "AcceptChoice", True)
    With AcceptChoice
        .Move 341, 130, 102, 28
        .Caption = "選択"
        .TabIndex = 3
        .TabStop = True
        .BackColor = 15590875
        .ForeColor = 4404002
        .WordWrap = False
        .Default = False
        .Cancel = False
    End With
    Exit Sub
Failed:
    Err.Raise Err.Number, "PickForm " & stage, Err.Description
End Sub
