Option Explicit

Public Function ControllerFolderRole(ByVal address As String) As String
    Dim folder As String
    folder = AddressFolder(address)
    If IsSameWorkbookAddress(Setting("StaffFolder"), Setting("GeneralFolder")) Then Exit Function
    If IsSameWorkbookAddress(folder, Setting("StaffFolder")) Then
        ControllerFolderRole = "Staff"
    ElseIf IsSameWorkbookAddress(folder, Setting("GeneralFolder")) Then
        ControllerFolderRole = "General"
    End If
End Function

Public Function CanMoveLedger(ByRef message As String) As Boolean
    Select Case ControllerFolderRole(ThisWorkbook.FullName)
        Case "Staff": CanMoveLedger = True
        Case "General": message = "この機能は非常勤フォルダからは使用できません。常勤職員フォルダのCONTROLLERから実行してください。"

        Case Else
            If IsLocalDevelopment() Then
                CanMoveLedger = True
            Else
                message = "移動は常勤フォルダのCONTROLLERから実行してください。ローカル開発用では3つのローカルフォルダを指定してください。"
            End If
    End Select
End Function


Public Function IsLocalDevelopment() As Boolean
    Dim files As Object
    Dim path As String
    Dim folder As Variant
    path = ThisWorkbook.Path
    If Mid$(path, 2, 2) <> ":\" Then Exit Function
    Set files = CreateObject("Scripting.FileSystemObject")
    If Not files.FolderExists(path & "\src") Then Exit Function
    If Not files.FileExists(path & "\CONTROLLER設定.xlsm") Then Exit Function
    If Not HasLedgerFolders() Then Exit Function
    For Each folder In Array(Setting("StaffFolder"), Setting("GeneralFolder"), Setting("BackupFolder"))
        If Mid$(CStr(folder), 2, 2) <> ":\" Then Exit Function
        If Not files.FolderExists(CStr(folder)) Then Exit Function
        If files.GetDrive(files.GetDriveName(CStr(folder))).DriveType <> 2 Then Exit Function
    Next
    IsLocalDevelopment = True
End Function


Public Function IsLocalSingleLedger() As Boolean
    Dim address As String
    If HasLedgerFolders() Or Len(Setting("GeneralLedger")) > 0 Then Exit Function
    address = Setting("StaffLedger")
    IsLocalSingleLedger = IsDirectWorkbookAddress(address) And Not IsHttpsAddress(address)
End Function

Private Function HasLedgerFolders() As Boolean
    Dim staff As String
    Dim general As String
    staff = Setting("StaffFolder")
    general = Setting("GeneralFolder")
    If Len(CanonicalAddress(staff)) = 0 Or Len(CanonicalAddress(general)) = 0 Then Exit Function
    If IsHttpsAddress(staff) <> IsHttpsAddress(general) Then Exit Function
    HasLedgerFolders = Not IsSameWorkbookAddress(staff, general)
End Function

Private Function LedgerInFolder(ByVal folder As String) As String
    Dim separator As String
    separator = IIf(IsHttpsAddress(folder), "/", "\")
    If Right$(folder, 1) <> separator Then folder = folder & separator
    LedgerInFolder = folder & "管理簿.xlsm"
End Function

Public Function ResolveYearAddress(ByVal year As String, ByRef message As String, Optional ByVal isWrite As Boolean = False) As String
    Dim definition As Variant
    definition = years(year)
    If Len(CStr(definition(5))) > 0 Then
        message = year & "年度：" & definition(5)
        Exit Function
    End If


    ResolveYearAddress = LocateFixedLedger(message, isWrite)
End Function

Public Function LocateFixedLedger(ByRef message As String, Optional ByVal isWrite As Boolean = False) As String
    Dim first As String
    Dim second As String


    LedgerAddresses first, second
    If Len(second) = 0 And IsDirectWorkbookAddress(first) Then
        LocateFixedLedger = first
        message = ""
        Exit Function
    End If
    If Len(first) = 0 Or Len(second) = 0 Then
        message = "設定ツールで常勤・非常勤フォルダを指定し、どちらか一方に管理簿.xlsmを配置してください。"
        Exit Function
    End If
    If Not HasLedgerFolders() Or ControllerFolderRole(first) <> "Staff" Or ControllerFolderRole(second) <> "General" Then
        message = "常勤・非常勤フォルダに異なる場所を同じ種類のアドレスで指定してください。"
        Exit Function
    End If
    LocateFixedLedger = ConfirmLedgerLocation(first, second, message, isWrite, IsHttpsAddress(first), True)
End Function

Private Sub LedgerAddresses(ByRef staff As String, ByRef general As String)
    staff = Setting("StaffLedger")
    general = Setting("GeneralLedger")
    If HasLedgerFolders() Then
        staff = LedgerInFolder(Setting("StaffFolder"))
        general = LedgerInFolder(Setting("GeneralFolder"))
    End If
End Sub

Private Function ConfirmLedgerLocation(ByVal first As String, ByVal second As String, _
    ByRef message As String, ByVal isWrite As Boolean, Optional ByVal deriveSite As Boolean = False, Optional ByVal allowLegacy As Boolean = False) As String
    Dim firstResult As String
    Dim secondResult As String
    Dim legacy As String
    Dim firstState As Long
    Dim secondState As Long
    firstState = ProbeExcelAddress(first, firstResult, deriveSite)
    secondState = ProbeExcelAddress(second, secondResult, deriveSite)
    If firstState = 1 And secondState = 1 Then
        message = "両方の候補に管理簿があります。正式データを確定してください。"

    ElseIf firstState = 0 And secondState = 0 Then
        If allowLegacy Then
            legacy = Setting("StaffLedger")
            If IsDirectWorkbookAddress(legacy) And IsHttpsAddress(legacy) = IsHttpsAddress(first) Then
                If Not IsSameWorkbookAddress(legacy, first) And Not IsSameWorkbookAddress(legacy, second) Then
                    If ProbeExcelAddress(legacy, firstResult, deriveSite) = 1 Then
                        ConfirmLedgerLocation = firstResult
                        message = ""
                        Exit Function
                    End If
                End If
            End If
        End If
        message = "両方の候補に管理簿がありません。アドレスと所在を確認してください。"
    ElseIf firstState = 1 And (secondState = 0 Or Not isWrite) Then
        ConfirmLedgerLocation = firstResult
        message = ""
    ElseIf secondState = 1 And (firstState = 0 Or Not isWrite) Then
        ConfirmLedgerLocation = secondResult
        message = ""
    Else
        message = "管理簿の所在を確定できません。アクセス権、開いているブック、ネットワークを確認してください。"
    End If
End Function

Private Function ProbeExcelAddress(ByVal address As String, ByRef actualAddress As String, Optional ByVal deriveSite As Boolean = False) As Long
    Dim book As Workbook
    Dim state As Object
    Dim message As String
    ProbeExcelAddress = -1
    Set book = OpenLedger(address, True, message)
    If Not book Is Nothing Then
        actualAddress = book.FullName
        book.Close False
        ProbeExcelAddress = 1
    Else
        'Excelのエラーだけでは不存在と権限エラーを区別できない。
        Set state = FileState(address, message, deriveSite)
        If Not state Is Nothing Then
            If Not state("Exists") Then ProbeExcelAddress = 0
        End If
    End If
End Function

Private Function SharePointEndpointBase() As String
    Dim url As String
    url = EffectiveSite()
    If Right$(url, 1) = "/" Then url = Left$(url, Len(url) - 1)
    SharePointEndpointBase = ExcelOpenAddress(url)
End Function



Private Function EffectiveSite() As String
    Dim explicitSite As String
    Dim common As String
    Dim address As String
    Dim name As Variant
    explicitSite = CanonicalAddress(Setting("SharePointSite"))
    common = AddressSite(CanonicalAddress(Setting("StaffFolder")))
    For Each name In Array("StaffFolder", "GeneralFolder", "BackupFolder")
        address = CanonicalAddress(Setting(CStr(name)))
        If Len(address) > 0 Then
            If StrComp(Left$(address, Len(explicitSite) + 1), explicitSite & "/", vbTextCompare) <> 0 Then explicitSite = ""
            If StrComp(AddressSite(address), common, vbTextCompare) <> 0 Then common = ""
        End If
    Next
    If IsHttpsAddress(explicitSite) Then
        EffectiveSite = explicitSite
    Else
        EffectiveSite = common
    End If
End Function

Private Function AddressSite(ByVal file As String) As String
    Dim parts As Variant
    If Not IsHttpsAddress(file) Then Exit Function
    parts = Split(file, "/")
    If UBound(parts) < 4 Then Exit Function
    Select Case LCase$(CStr(parts(3)))
        Case "sites", "teams"
            AddressSite = "https://" & parts(2) & "/" & parts(3) & "/" & parts(4)
    End Select
End Function

Private Function ApiFile(ByVal address As String, Optional ByVal deriveSite As Boolean = False) As String
    Dim site As String
    Dim file As String
    Dim slash As Long
    site = EffectiveSite()

    file = CanonicalAddress(address)
    If deriveSite And IsHttpsAddress(file) Then
        If Len(site) = 0 Or StrComp(Left$(file, Len(site) + 1), site & "/", vbTextCompare) <> 0 Then site = AddressSite(file)
    End If
    If Not IsHttpsAddress(site) Or Not IsHttpsAddress(file) Then Exit Function
    If StrComp(Left$(file, Len(site) + 1), site & "/", vbTextCompare) <> 0 Then Exit Function
    slash = InStr(9, file, "/")
    ApiFile = ExcelOpenAddress(site) & "/_api/web/GetFileByServerRelativePath(decodedUrl='" & _
        Application.WorksheetFunction.EncodeURL(Replace(Mid$(file, slash), "'", "''")) & "')"
End Function

Private Function SharePointRequest(ByVal method As String, ByVal url As String, ByVal body As String, _
    ByRef message As String, Optional ByVal digest As String = "") As Object
    Dim request As Object
    On Error GoTo Failed
    If Not IsHttpsAddress(url) Then
        message = "SharePointサイトと対象の直接URLが一致しません。"
        Exit Function
    End If
    Set request = CreateObject("WinHttp.WinHttpRequest.5.1")
    request.SetTimeouts 5000, 8000, 15000, 25000
    request.Option(6) = False
    request.Open method, url, False
    request.SetAutoLogonPolicy 0
    request.SetRequestHeader "Accept", "application/atom+xml"
    If Len(digest) > 0 Then request.SetRequestHeader "X-RequestDigest", digest
    If Len(body) > 0 Then request.SetRequestHeader "Content-Type", "application/json;odata=verbose; charset=utf-8"
    If Len(body) = 0 Then
        request.Send
    Else
        request.Send body
    End If
    Set SharePointRequest = request
    Exit Function
Failed:
    message = "SharePoint通信に失敗しました。エラー " & Err.Number & "。変更要求は再送せず結果を照合してください。"
End Function

Private Function ResponseXmlValue(ByVal request As Object, ByVal name As String) As String
    Dim xml As Object
    Dim node As Object
    Set xml = CreateObject("MSXML2.DOMDocument.6.0")
    xml.async = False
    xml.resolveExternals = False
    xml.SetProperty "ProhibitDTD", True
    If Not xml.LoadXML(request.ResponseText) Then Exit Function
    Set node = xml.SelectSingleNode("//*[local-name()='" & name & "']")
    If Not node Is Nothing Then ResponseXmlValue = node.Text
End Function

Public Function FileState(ByVal address As String, ByRef message As String, Optional ByVal deriveSite As Boolean = False) As Object
    Dim result As Object
    Dim file As Object
    Dim request As Object
    Dim status As Long
    Dim endpoint As String
    On Error GoTo Failed
    message = ""
    Set result = NewDictionary()
    result.Add "Exists", False
    result.Add "Id", ""
    result.Add "ETag", ""
    result.Add "Size", ""
    result.Add "Locked", False
    If IsHttpsAddress(address) Then
        endpoint = ApiFile(address, deriveSite)
        Set request = SharePointRequest("GET", endpoint, "", message)
        If request Is Nothing Then Exit Function
        Select Case request.Status
            Case 200
                result("Id") = ResponseXmlValue(request, "UniqueId")
                result("ETag") = ResponseXmlValue(request, "ETag")
                If Len(result("ETag")) = 0 Then result("ETag") = request.GetResponseHeader("ETag")
                result("Size") = ResponseXmlValue(request, "Length")
                If Len(result("Id")) = 0 Or Len(result("ETag")) = 0 Or Len(result("Size")) = 0 Then GoTo InvalidResponse
                result("Exists") = True
            Case 404
                If InStr(ResponseXmlValue(request, "code"), "System.IO.FileNotFoundException") = 0 Then GoTo InvalidResponse
            Case Else
                message = "SharePointの所在を確定できません。HTTP " & request.Status & "。認証とアクセス権を確認してください。"
                Exit Function
        End Select
    Else
        If Len(CanonicalAddress(address)) = 0 Then Exit Function
        On Error Resume Next
        Set file = CreateObject("Scripting.FileSystemObject").GetFile(address)
        status = Err.Number
        Err.Clear
        On Error GoTo Failed
        If status = 0 Then
            result("Exists") = True
            result("Size") = CStr(file.Size)
            result("ETag") = CStr(CDbl(file.DateLastModified)) & "|" & result("Size")
        ElseIf status <> 53 And status <> 76 Then
            message = "ファイルの所在を確認できません。エラー " & status
            Exit Function
        End If
    End If
    Set FileState = result
    Exit Function
InvalidResponse:
    message = "SharePointの応答から所在と識別情報を確定できません。"
    Exit Function
Failed:
    message = "所在確認に失敗しました。エラー " & Err.Number
End Function

Private Function JsonString(ByVal value As String) As String
    Dim i As Long
    Dim code As Long
    JsonString = """"
    For i = 1 To Len(value)
        code = AscW(Mid$(value, i, 1)) And &HFFFF&
        If code < 32 Or code > 126 Or code = 34 Or code = 92 Then
            JsonString = JsonString & "\u" & Right$("0000" & Hex$(code), 4)
        Else
            JsonString = JsonString & Chr$(code)
        End If
    Next
    JsonString = JsonString & """"
End Function

Private Function ResourcePathJson(ByVal address As String) As String
    ResourcePathJson = "{""__metadata"":{""type"":""SP.ResourcePath""},""DecodedUrl"":" & JsonString(CanonicalAddress(address)) & "}"
End Function

Private Function ServerCopyOrMove(ByVal source As String, ByVal destination As String, ByVal isMove As Boolean, ByRef message As String) As Boolean
    Dim site As String
    Dim digest As String
    Dim endpoint As String
    Dim request As Object
    Dim body As String
    On Error GoTo Failed
    If IsHttpsAddress(source) Then
        If Len(ApiFile(source)) = 0 Or Len(ApiFile(destination)) = 0 Then
            message = "移動元・移動先・退避先は設定したサイト内に指定してください。"
            Exit Function
        End If
        site = SharePointEndpointBase()
        Set request = SharePointRequest("POST", site & "/_api/contextinfo", "", message)
        If request Is Nothing Then Exit Function
        If request.Status <> 200 Then
            message = "変更要求の認証を確立できません。HTTP " & request.Status
            Exit Function
        End If
        digest = ResponseXmlValue(request, "FormDigestValue")
        If Len(digest) = 0 Then
            message = "SharePointの変更用ダイジェストを取得できません。"
            Exit Function
        End If
        endpoint = site & "/_api/SP.MoveCopyUtil." & IIf(isMove, "MoveFileByPath", "CopyFileByPath") & "(overwrite=@a1)?@a1=false"
        body = "{""srcPath"":" & ResourcePathJson(source) & ",""destPath"":" & ResourcePathJson(destination) & _
            ",""options"":{""__metadata"":{""type"":""SP.MoveCopyOptions""},""KeepBoth"":false,""ShouldBypassSharedLocks"":false}}"
        Set request = SharePointRequest("POST", endpoint, body, message, digest)
        If request Is Nothing Then Exit Function
        If request.Status < 200 Or request.Status >= 300 Then
            message = "変更結果を確認してください。HTTP " & request.Status & "。要求は再送しません。"
            Exit Function
        End If
    ElseIf isMove Then
        Name source As destination
    Else
        CreateObject("Scripting.FileSystemObject").CopyFile source, destination, False
    End If
    ServerCopyOrMove = True
    Exit Function
Failed:

    If Err.Number = 70 Or Err.Number = 75 Then
        message = "ファイルを使用中か、保存先への書込権限がありません。権限とファイルの使用状況を確認してください。"
    Else
        message = "コピーまたは移動に失敗しました。再送せず結果を照合してください。エラー " & Err.Number
    End If
End Function

Private Function FileContent(ByVal address As String, ByRef message As String) As Variant
    Dim request As Object
    Dim handle As Integer
    Dim bytes() As Byte
    On Error GoTo Failed
    If IsHttpsAddress(address) Then
        If Len(ApiFile(address)) = 0 Then Exit Function
        Set request = SharePointRequest("GET", ApiFile(address) & "/$value", "", message)
        If request Is Nothing Then Exit Function
        If request.Status <> 200 Then
            message = "ファイル内容を確認できません。HTTP " & request.Status
            Exit Function
        End If
        FileContent = request.ResponseBody
    Else
        handle = FreeFile
        Open address For Binary Access Read Lock Write As #handle
        If LOF(handle) > 0 Then
            ReDim bytes(0 To LOF(handle) - 1)
            Get #handle, , bytes
            FileContent = bytes
        End If
        Close #handle
    End If
    Exit Function
Failed:
    If handle > 0 Then Close #handle
    message = "ファイル内容を読み取れません。エラー " & Err.Number
End Function

Private Function IsEqualFileContent(ByVal first As String, ByVal second As String, ByRef message As String) As Boolean
    Dim leftBytes As Variant
    Dim rightBytes As Variant
    Dim i As Long
    leftBytes = FileContent(first, message)
    If Not IsArray(leftBytes) Then Exit Function
    rightBytes = FileContent(second, message)
    If Not IsArray(rightBytes) Then Exit Function
    If UBound(leftBytes) <> UBound(rightBytes) Then Exit Function
    For i = LBound(leftBytes) To UBound(leftBytes)
        If leftBytes(i) <> rightBytes(i) Then Exit Function
    Next
    IsEqualFileContent = True
End Function


Private Function MoveDirection(ByRef message As String) As Long
    Dim staff As String
    Dim general As String
    Dim first As Object
    Dim second As Object
    If Not CanMoveLedger(message) Then Exit Function
    If PendingSave() Or Len(CStr(Sheet("MoveRecovery").Range("B1").Value2)) > 0 Then
        message = "未確定の保存・移動があります。結果を確認してください。"
        Exit Function
    End If
    LedgerAddresses staff, general
    If ControllerFolderRole(staff) <> "Staff" Or ControllerFolderRole(general) <> "General" Then
        message = "常勤・非常勤フォルダと管理簿の設定を確認してください。"
        Exit Function
    End If
    Set first = FileState(staff, message)
    If first Is Nothing Then Exit Function
    Set second = FileState(general, message)
    If second Is Nothing Then Exit Function
    If first("Exists") And second("Exists") Then
        message = "常勤・非常勤の両方に管理簿があります。正式な1冊を確認してください。"
    ElseIf first("Exists") Then
        MoveDirection = 1
    ElseIf second("Exists") Then
        MoveDirection = -1
    Else
        message = "常勤・非常勤のどちらにも管理簿が見つかりません。"
    End If
End Function

Private Function MoveCaption(ByVal direction As Long) As String
    Select Case direction
        Case 1: MoveCaption = "非常勤へ移す"
        Case -1: MoveCaption = "常勤へ戻す"
        Case Else: MoveCaption = "移動状態を確認"
    End Select
End Function

Public Sub RefreshLedgerMoveButton()
    Dim message As String
    Dim button As Shape
    On Error Resume Next
    Set button = Sheet("検索").Shapes("LedgerMove")
    On Error GoTo 0
    If button Is Nothing Then Exit Sub
    button.TextFrame.Characters.Text = MoveCaption(MoveDirection(message))
    button.AlternativeText = message
End Sub

Public Sub MoveLedgerFromCurrentLocation()
    Dim message As String
    Dim direction As Long
    Dim displayed As String
    Dim prompt As String
    displayed = Sheet("検索").Shapes("LedgerMove").TextFrame.Characters.Text
    direction = MoveDirection(message)
    If direction = 0 Then
        RefreshLedgerMoveButton
        Notify message
        Exit Sub
    End If
    If displayed <> MoveCaption(direction) Then
        RefreshLedgerMoveButton
        Notify "管理簿の所在が変わりました。移動ボタンの表示を確認してください。"
        Exit Sub
    End If
    prompt = IIf(direction = 1, "管理簿を常勤フォルダから非常勤フォルダへ移動します。", _
        "管理簿を非常勤フォルダから常勤フォルダへ戻します。")
    If MsgBox(prompt & vbCrLf & "よろしいですか？", vbYesNo Or vbQuestion Or vbDefaultButton2, "管理簿の移動") <> vbYes Then Exit Sub
    If MoveDirection(message) <> direction Then
        RefreshLedgerMoveButton
        Notify "移動前に管理簿の状態が変わりました。" & vbCrLf & message
        Exit Sub
    End If
    ExecuteBusiness direction = 1, message
    RefreshLedgerMoveButton
    Notify message
End Sub

Public Function ExecuteBusiness(ByVal isStart As Boolean, ByRef message As String) As Boolean
    Dim source As String
    Dim destination As String
    Dim backup As String
    Dim folder As String
    Dim before As Object
    Dim after As Object
    Dim target As Object
    Dim ws As Worksheet
    Dim book As Workbook
    Dim separator As String
    If Not CanMoveLedger(message) Then Exit Function
    If Not BeginAction() Then Exit Function
    On Error GoTo Failed
    Set ws = Sheet("MoveRecovery")
    If PendingSave() Or Len(CStr(ws.Range("B1").Value2)) > 0 Then
        message = "未確定の保存・移動があります。確認後に操作してください。"
        GoTo Finished
    End If

    If isStart Then
        LedgerAddresses source, destination
    Else
        LedgerAddresses destination, source
    End If
    If Len(source) = 0 Or Len(destination) = 0 Then
        message = "常勤・非常勤フォルダと管理簿を設定してください。"
        GoTo Finished
    End If
    If AddressFileName(source) <> "管理簿.xlsm" Or AddressFileName(destination) <> "管理簿.xlsm" Then
        message = "固定アドレスには管理簿.xlsmを指定してください。"
        GoTo Finished
    End If
    If ControllerFolderRole(source) <> IIf(isStart, "Staff", "General") Or _
        ControllerFolderRole(destination) <> IIf(isStart, "General", "Staff") Then
        message = "管理簿のアドレスと常勤・非常勤フォルダの設定が一致しません。"
        GoTo Finished
    End If
    For Each book In Application.Workbooks
        If IsSameWorkbookAddress(book.FullName, source) Or IsSameWorkbookAddress(book.FullName, destination) Then
            message = "管理簿を閉じてから操作してください。"
            GoTo Finished
        End If
    Next
    Set before = FileState(source, message)
    If before Is Nothing Then GoTo Finished
    Set target = FileState(destination, message)
    If target Is Nothing Then GoTo Finished
    If Not before("Exists") Then
        message = "移動元に管理簿が見つかりません。設定と現在の配置を確認してください。"
        GoTo Finished
    End If
    If target("Exists") Then
        message = "移動先に同名の管理簿があります。上書きせず、処理を中止しました。"
        GoTo Finished
    End If
    folder = Setting("BackupFolder")
    If Len(CanonicalAddress(folder)) = 0 Or IsHttpsAddress(folder) <> IsHttpsAddress(source) Then
        message = "管理簿と同じ保存方式のバックアップ先を設定してください。"
        GoTo Finished
    End If
    separator = IIf(IsHttpsAddress(folder), "/", "\")
    If Right$(folder, 1) <> separator Then folder = folder & separator
    backup = folder & "管理簿_" & NewTransactionId() & ".xlsm"
    PutValue ws.Range("B1"), "バックアップ確認待ち"
    PutValue ws.Range("B2"), source
    PutValue ws.Range("B3"), destination
    PutValue ws.Range("B4"), CStr(before("Id"))
    PutValue ws.Range("B5"), backup
    PutValue ws.Range("B6"), CStr(before("ETag"))
    PutValue ws.Range("B7"), Format$(Now, "yyyy/mm/dd hh:nn:ss")
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Failed
    If Not ServerCopyOrMove(source, backup, False, message) Then GoTo Finished
    If Not IsEqualFileContent(source, backup, message) Then
        message = "バックアップと管理簿の内容が一致しないか、比較できません。移動していません。"
        GoTo Finished
    End If
    Set after = FileState(source, message)
    If after Is Nothing Then GoTo Finished
    If Not after("Exists") Or after("Id") <> before("Id") Or after("ETag") <> before("ETag") Then
        message = "退避後に管理簿が変わりました。移動していません。"
        GoTo Finished
    End If
    PutValue ws.Range("B1"), "移動結果確認待ち"
    PutValue ws.Range("B8"), "バックアップ内容確認済み"
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Failed
    Set after = FileState(source, message)
    If after Is Nothing Then GoTo Finished
    If Not after("Exists") Or after("Id") <> before("Id") Or after("ETag") <> before("ETag") Then
        message = "移動要求の送信前に元ファイルが変わりました。記録を照合してください。"
        GoTo Finished
    End If
    If Not ServerCopyOrMove(source, destination, True, message) Then GoTo Finished
    ExecuteBusiness = VerifyMoveJournal(message)
Finished:
    isBusy = False
    Exit Function
Failed:
    message = "朝夕の処理が中断されました。記録と両方の候補を照合し、確認なしで再送しないでください。エラー " & Err.Number
    Resume Finished
End Function

Private Function VerifyMoveJournal(ByRef message As String) As Boolean
    Dim ws As Worksheet
    Dim source As Object
    Dim destination As Object
    Dim oldAddress As String
    Dim newAddress As String
    Set ws = Sheet("MoveRecovery")
    oldAddress = CStr(ws.Range("B2").Value2)
    newAddress = CStr(ws.Range("B3").Value2)
    Set source = FileState(oldAddress, message)
    If source Is Nothing Then Exit Function
    Set destination = FileState(newAddress, message)
    If destination Is Nothing Then Exit Function
    If source("Exists") Or Not destination("Exists") Or CStr(ws.Range("B1").Value2) <> "移動結果確認待ち" Then GoTo Unconfirmed
    If CStr(ws.Range("B4").Value2) <> CStr(destination("Id")) Then GoTo Unconfirmed
    If Not IsEqualFileContent(newAddress, CStr(ws.Range("B5").Value2), message) Then GoTo Unconfirmed
    PutValue ws.Range("B9"), "移動・内容確認済み " & Format$(Now, "yyyy/mm/dd hh:nn:ss")
    ws.Range("B1").ClearContents
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then
        PutValue ws.Range("B1"), "移動結果確認待ち"
        GoTo Unconfirmed
    End If
    message = "管理簿の移動と内容の一致を確認しました。最新を再読込してください。"
    VerifyMoveJournal = True
    Exit Function
Unconfirmed:
    message = "移動結果を確定できません。移動元・移動先・バックアップを確認してください。再送しません。"
End Function

Public Sub ConfirmMoveResult()
    Dim message As String
    If CanMoveLedger(message) Then
        If Len(CStr(Sheet("MoveRecovery").Range("B1").Value2)) = 0 Then
            message = "確認待ちの移動はありません。"
        Else
            VerifyMoveJournal message
        End If
    End If
    Notify message
End Sub

Public Sub RestoreBackup()
    Dim address As Variant
    Dim target As Variant
    Dim state As Object
    Dim message As String
    If Not IsOwnerAuthorized() Or ControllerFolderRole(ThisWorkbook.FullName) <> "Staff" Then
        Notify "オーナー認証と常勤フォルダからの起動が必要です。"
        Exit Sub
    End If
    address = Application.InputBox("退避した管理簿の直接URLを入力してください。", "バックアップの復旧確認コピー", Type:=2)
    If VarType(address) = vbBoolean Then Exit Sub
    target = Application.InputBox("同じ退避フォルダ内の未使用の直接URLを入力してください。名前は復旧確認_から始めます。", "復旧先", Type:=2)
    If VarType(target) = vbBoolean Then Exit Sub
    If Not IsSameWorkbookAddress(AddressFolder(CStr(address)), Setting("BackupFolder")) Or _
        Not IsSameWorkbookAddress(AddressFolder(CStr(target)), Setting("BackupFolder")) Or _
        Left$(AddressFileName(CStr(target)), 5) <> "復旧確認_" Or _
        Not IsDirectWorkbookAddress(CStr(target)) Then
        Notify "元・先の退避フォルダと新しい復旧確認ファイル名を確認してください。"
        Exit Sub
    End If
    Set state = FileState(CStr(target), message)
    If state Is Nothing Then
        Notify message
        Exit Sub
    End If
    If state("Exists") Then
        Notify "復旧先に同名のファイルがあります。上書きしません。"
        Exit Sub
    End If
    If MsgBox("退避元から復旧確認ファイルを作成しますか？", vbYesNo Or vbQuestion Or vbDefaultButton2) <> vbYes Then Exit Sub
    If ServerCopyOrMove(CStr(address), CStr(target), False, message) Then
        If IsEqualFileContent(CStr(address), CStr(target), message) Then
            message = "復旧確認ファイルを作成し、元のバックアップとの一致を確認しました。"
        Else
            message = "復旧確認ファイルの内容を確定できません。再送せず確認してください。"
        End If
    End If
    Notify message
End Sub

Public Sub ReleaseUnsentMove()
    Dim page As Worksheet
    Dim source As Object
    Dim target As Object
    Dim message As String
    On Error GoTo Failed
    If Not CanMoveLedger(message) Then GoTo Finished
    Set page = Sheet("MoveRecovery")
    If CStr(page.Range("B1").Value2) <> "バックアップ確認待ち" Then
        message = "移動要求を送っていないことが記録上明確な場合だけ解除できます。"
        GoTo Finished
    End If
    Set source = FileState(CStr(page.Range("B2").Value2), message)
    If source Is Nothing Then GoTo Finished
    Set target = FileState(CStr(page.Range("B3").Value2), message)
    If target Is Nothing Then GoTo Finished
    If Not source("Exists") Or target("Exists") Or source("Id") <> page.Range("B4").Value2 Or source("ETag") <> page.Range("B6").Value2 Then
        message = "当初の移動元・移動先の状態と一致しないため解除できません。"
        GoTo Finished
    End If
    If MsgBox("移動未送信の記録を解除しますか？バックアップ候補は残ります。", vbYesNo Or vbQuestion Or vbDefaultButton2) <> vbYes Then Exit Sub
    PutValue page.Range("B9"), "移動未送信の確認後に解除 " & Format$(Now, "yyyy/mm/dd hh:nn:ss")
    page.Range("B1").ClearContents
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Failed
    message = "移動未送信の記録を解除しました。"
Finished:
    If Len(message) > 0 Then Notify message
    Exit Sub
Failed:
    PutValue page.Range("B1"), "バックアップ確認待ち"
    Notify "解除の保存結果が不明です。操作を止めて記録を確認してください。"
End Sub
