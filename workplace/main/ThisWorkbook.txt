Option Explicit

Private Sub Workbook_Open()
    On Error GoTo Failed
    SignOutOwner
    If Sheet("Calendar").Visible = xlSheetVisible Then
        Sheet("検索").Activate
        Sheet("Calendar").Visible = xlSheetVeryHidden
    End If
    InitializeControllerUi
    Exit Sub
Failed:
    Sheet("検索").Range("C35").Value2 = "操作画面の初期化に失敗しました。最新を再読込してください。"
End Sub

Private Sub Workbook_SheetSelectionChange(ByVal Sh As Object, ByVal Target As Range)
    If Sh.Name = "検索" Then ResultSelection Target
End Sub

Private Sub Workbook_SheetBeforeDoubleClick(ByVal Sh As Object, ByVal Target As Range, Cancel As Boolean)
    If Sh.Name = "Calendar" Then
        Cancel = True
        PickDispatchDate Target
        Exit Sub
    End If
    If Sh.Name <> "検索" Then Exit Sub
    If Intersect(Target, Sh.Range("C20:U33")) Is Nothing Then Exit Sub
    Cancel = True
    ResultSelection Target
    OpenSelected
End Sub

Private Sub Workbook_SheetActivate(ByVal Sh As Object)
    If Sh.Name = "検索" Then SearchActivated
End Sub
Private Sub Workbook_BeforeSave(ByVal SaveAsUI As Boolean, Cancel As Boolean)
    HideOwnerWorkspace
    If Sheet("CompareResults").Visible = xlSheetVisible Then CloseComparison
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    If Not UiCanLeaveDetail() Then
        Cancel = True
    Else
        SignOutOwner
    End If
End Sub
Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    If Sh.Name = "詳細" Then RefreshEditedStatus Target
End Sub
