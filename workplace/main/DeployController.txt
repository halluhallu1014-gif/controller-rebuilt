Option Explicit

Public Sub InstallControllerMain()
    Dim book As Workbook
    Dim page As Worksheet
    Dim sheetName As Variant
    Dim previousEvents As Boolean
    Dim previousScreen As Boolean
    Dim phase As String
    previousEvents = Application.EnableEvents
    previousScreen = Application.ScreenUpdating
    On Error GoTo Failed
    Set book = ThisWorkbook
    phase = "空のブックを確認"
    If book.ReadOnly Or book.Worksheets.Count <> 1 Or book.Sheets.Count <> 1 Then Err.Raise 5
    Set page = book.Worksheets(1)
    If Application.CountA(page.UsedRange) <> 0 Or page.Shapes.Count <> 0 Or book.Names.Count <> 0 Then Err.Raise 5
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    book.Styles("Normal").Font.Name = "Carlito"
    book.Styles("Normal").Font.Size = 11
    phase = "シートを作成"
    page.Name = "検索"
    For Each sheetName In Array("詳細", "Settings", "Fields", "Comparison", "Recovery", "MoveRecovery", "CompareResults", "History", "Calendar")
        Set page = book.Worksheets.Add(After:=page)
        page.Name = CStr(sheetName)
    Next
    phase = "検索"
    Set page = book.Worksheets("検索")
    InstallSheet1 page
    phase = "詳細"
    Set page = book.Worksheets("詳細")
    InstallSheet2 page
    AlignInstalledDetail page
    phase = "Settings"
    Set page = book.Worksheets("Settings")
    InstallSheet3 page
    phase = "Fields"
    Set page = book.Worksheets("Fields")
    InstallSheet4 page
    phase = "Comparison"
    Set page = book.Worksheets("Comparison")
    InstallSheet5 page
    phase = "Recovery"
    Set page = book.Worksheets("Recovery")
    InstallSheet6 page
    phase = "MoveRecovery"
    Set page = book.Worksheets("MoveRecovery")
    InstallSheet7 page
    phase = "CompareResults"
    Set page = book.Worksheets("CompareResults")
    InstallSheet8 page
    phase = "History"
    Set page = book.Worksheets("History")
    InstallSheet9 page
    phase = "Calendar"
    Set page = book.Worksheets("Calendar")
    InstallSheet10 page
    phase = "選択肢の名前と入力規則"
    AddInstallChoice book, "P", "DocumentName", "F29"
    AddInstallChoice book, "Q", "TaxType", "F27"
    AddInstallChoice book, "R", "Status", "F36"
    book.Worksheets("詳細").Protect DrawingObjects:=False, UserInterfaceOnly:=True
    book.Worksheets("検索").Activate
    phase = "構成と起動を確認"
    InitializeControllerUi
    VerifyControllerInstall book
    Application.EnableEvents = previousEvents
    Application.ScreenUpdating = previousScreen
    MsgBox "CONTROLLERの構築と構成確認が完了しました。", vbInformation
    Exit Sub
Failed:
    Dim message As String
    message = "導入不合格。工程 " & phase & "、エラー番号 " & Err.Number & " " & Err.Description & _
        "。ブックを保存せず閉じてください。"
    Application.EnableEvents = previousEvents
    Application.ScreenUpdating = previousScreen
    MsgBox message, vbCritical
End Sub

Private Sub AddInstallChoice(ByVal book As Workbook, ByVal column As String, ByVal fieldId As String, ByVal address As String)
    Dim first As String
    Dim source As String
    first = "Settings!$" & column & "$2"
    source = first & ":$" & column & "$501"
    book.Names.Add Name:="ControllerChoice" & fieldId, _
        RefersTo:="=" & first & ":INDEX(" & source & ",MAX(1,COUNTA(" & source & ")))"
    With book.Worksheets("詳細").Range(address).Validation
        .Add Type:=xlValidateList, Formula1:="=ControllerChoice" & fieldId
        .ShowError = False
        .InCellDropdown = True
    End With
End Sub

Private Sub VerifyControllerInstall(ByVal book As Workbook)
    Dim shape As Shape
    Dim id As Variant
    Dim message As String
    If book.Worksheets.Count <> 10 Or book.Names.Count <> 3 Then Err.Raise 5, , "シートまたは名前定義"
    Set shape = book.Worksheets("詳細").Shapes("RelatedCasesList")
    If shape.Type <> msoFormControl Then Err.Raise 5, , "関連事案一覧の種類"
    If shape.FormControlType <> xlListBox Or Not shape.OnAction Like "*!RelatedListChanged" Then Err.Raise 5, , "関連事案一覧の割当"
    For Each id In Array("Calendar_ReceivedDate", "Calendar_DraftDate", "Calendar_ApprovedDate", _
        "Calendar_IssuedDate", "UI_コピー取込", "UI_選択項目へ取込")
        If Len(book.Worksheets("詳細").Shapes(CStr(id)).OnAction) > 0 Then Err.Raise 5, , "未接続ボタン"
    Next
    For Each id In Array("F29", "F27", "F36")
        If book.Worksheets("詳細").Range(CStr(id)).Validation.Type <> xlValidateList Then Err.Raise 5, , "入力規則"
    Next
    If Not ValidatePresentation(message) Then Err.Raise 5, , message
End Sub

Private Sub InstallButtonFont(ByVal shape As Shape, ByVal caption As String, ByVal fontName As String, ByVal size As Single)
    With shape.TextFrame2.TextRange
        .Text = caption
        .Font.Name = fontName
        .Font.Size = size
    End With
End Sub

Private Sub AlignInstalledDetail(ByVal page As Worksheet)
    Dim item As Variant
    Dim button As Shape
    Dim anchor As Range
    For Each item In Array(Array("ReceivedDate", "L32"), Array("DraftDate", "L38:L39"), _
        Array("ApprovedDate", "V38:V39"), Array("IssuedDate", "L41"))
        Set button = page.Shapes("Calendar_" & item(0))
        Set anchor = page.Range(item(1))
        button.Top = anchor.Top
        button.Height = anchor.Height
        button.Left = anchor.Left + anchor.Width - button.Width
    Next
    Set anchor = page.Range("C58")
    For Each item In Array("ControllerCopyNumber", "UI_選択項目をコピー", "UI_コピー取込", "UI_選択項目へ取込")
        Set button = page.Shapes(CStr(item))
        button.Top = anchor.Top + (anchor.Height - button.Height) / 2
    Next
End Sub
