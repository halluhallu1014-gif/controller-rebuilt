Public Function LastRow(ByVal ws As Worksheet) As Long
    Dim found As Range
    Set found = ws.Cells.Find(What:="*", LookIn:=xlFormulas, LookAt:=xlPart, _
        SearchOrder:=xlByRows, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    LastRow = 1
    If Not found Is Nothing Then LastRow = found.Row
End Function

Public Function LastColumn(ByVal ws As Worksheet) As Long
    Dim found As Range
    Set found = ws.Cells.Find(What:="*", LookIn:=xlFormulas, LookAt:=xlPart, _
        SearchOrder:=xlByColumns, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    LastColumn = 1
    If Not found Is Nothing Then LastColumn = found.Column
End Function

Public Function SameValue(ByVal first As Variant, ByVal second As Variant) As Boolean
    If IsError(first) Or IsError(second) Then Exit Function
    If IsNull(first) Or IsNull(second) Then Exit Function
    If Len(CStr(first)) = 0 And Len(CStr(second)) = 0 Then
        SameValue = True
    ElseIf VarType(first) = vbString Or VarType(second) = vbString Then
        SameValue = (VarType(first) = VarType(second) And CStr(first) = CStr(second))
    ElseIf IsNumeric(first) And IsNumeric(second) Then
        SameValue = (CDbl(first) = CDbl(second))
    Else
        SameValue = (CStr(first) = CStr(second))
    End If
End Function

Public Sub PutValue(ByVal target As Range, ByVal value As Variant, Optional ByVal isDate As Boolean = False)
    If isDate Then
        target.NumberFormat = "yyyy/mm/dd"
    Else
        target.NumberFormat = "@"
    End If
    target.Value2 = value
End Sub

Public Function IsDigits(ByVal value As Variant, ByVal size As Long) As Boolean
    Dim index As Long
    Dim text As String
    If VarType(value) <> vbString Then Exit Function
    text = CStr(value)
    If Len(text) <> size Then Exit Function
    For index = 1 To size
        If Mid$(text, index, 1) < "0" Or Mid$(text, index, 1) > "9" Then Exit Function
    Next
    IsDigits = True
End Function

Public Function RestoreNumber(ByVal masked As Variant, ByRef number As String) As Boolean
    Dim text As String
    Dim body As String
    Dim index As Long
    number = ""
    If VarType(masked) <> vbString Then Exit Function
    text = CStr(masked)
    If IsDigits(text, 13) Then
        number = text
        RestoreNumber = True
        Exit Function
    End If
    If Len(text) <> 20 Then Exit Function
    If Left$(text, 4) <> "MN1:" Or Mid$(text, 18, 1) <> ":" Then Exit Function
    body = Mid$(text, 5, 13)
    If Not IsDigits(body, 13) Then Exit Function
    For index = 13 To 1 Step -1
        number = number & CStr((CLng(Mid$(body, index, 1)) + 3) Mod 10)
    Next
    If Right$(text, 2) <> NumberChecksum(number) Then
        number = ""
        Exit Function
    End If
    RestoreNumber = True
End Function

Private Function NumberChecksum(ByVal number As String) As String
    Dim index As Long
    Dim total As Long
    For index = 1 To 13
        total = total + index * CLng(Mid$(number, index, 1))
    Next
    NumberChecksum = Format$(total Mod 97, "00")
End Function

Public Function NormalizeValue(ByVal fieldId As String, ByVal value As Variant, ByRef normalized As Variant) As Boolean
    Dim text As String
    normalized = value
    If IsError(value) Or IsNull(value) Then Exit Function
    If Len(CStr(value)) = 0 Then
        normalized = ""
    ElseIf fields(fieldId)(1) = "Date" Then
        If VarType(value) = vbString Or VarType(value) = vbBoolean Then Exit Function
        If Not IsNumeric(value) Then Exit Function
        If CDbl(value) < 1 Or CDbl(value) > 2958465# Then Exit Function
        If CDbl(value) <> Fix(CDbl(value)) Then Exit Function
        normalized = CDbl(value)
    Else
        '文字列項目を数値から推測復元しない。失われた先頭0は補完しない。
        If VarType(value) <> vbString Then Exit Function
        text = CStr(value)
        If fieldId = "PostalCode" Or fieldId = "PhoneNumber" Then
            text = Replace(text, "-", "")
            If fieldId = "PostalCode" Then
                If Not IsDigits(text, 7) Then Exit Function
            Else
                If Len(text) < 9 Or Len(text) > 11 Then Exit Function
                If Not IsDigits(text, Len(text)) Then Exit Function
            End If
        End If
        normalized = text
    End If
    NormalizeValue = True
End Function

Public Function DisplayValue(ByVal fieldId As String, ByVal value As Variant) As Variant
    Dim text As String
    DisplayValue = value
    If fieldId = "PostalCode" Then
        text = CStr(value)
        If IsDigits(text, 7) Then DisplayValue = Left$(text, 3) & "-" & Right$(text, 4)
    ElseIf fieldId = "PhoneNumber" Then
        DisplayValue = FormatPhone(CStr(value))
    End If
End Function

Private Function FormatPhone(ByVal text As String) As String
    Dim prefix As String
    Dim rowIndex As Long
    Dim areaLength As Long
    Dim exchangeLength As Long
    Dim longest As Long
    Dim ws As Worksheet
    FormatPhone = text
    If Len(text) = 11 Then
        Select Case Left$(text, 3)
            Case "070", "080", "090", "050"
                FormatPhone = Left$(text, 3) & "-" & Mid$(text, 4, 4) & "-" & Right$(text, 4)
                Exit Function
        End Select
    End If
    If Len(text) <> 10 Then Exit Function
    Set ws = Sheet("Settings")
    For rowIndex = 2 To LastRow(ws)
        prefix = CStr(ws.Cells(rowIndex, 10).Value2)
        If Len(prefix) > longest And Len(prefix) > 0 Then
            If Left$(text, Len(prefix)) = prefix Then
                If IsNumeric(ws.Cells(rowIndex, 11).Value2) And IsNumeric(ws.Cells(rowIndex, 12).Value2) Then
                    areaLength = CLng(ws.Cells(rowIndex, 11).Value2)
                    exchangeLength = CLng(ws.Cells(rowIndex, 12).Value2)
                    If areaLength >= 2 And areaLength <= 5 And exchangeLength >= 1 Then
                        If areaLength + exchangeLength + 4 = 10 Then
                            FormatPhone = Left$(text, areaLength) & "-" & Mid$(text, areaLength + 1, exchangeLength) & "-" & Right$(text, 4)
                            longest = Len(prefix)
                        End If
                    End If
                End If
            End If
        End If
    Next
End Function

Public Sub WriteHistory(ByVal recovery As Worksheet)
    Dim row As Long
    Dim operation As String
    Dim identity As String
    Dim transactionId As String
    operation = CStr(recovery.Range("B5").Value2)
    If Len(operation) = 0 Then Exit Sub
    identity = CStr(recovery.Range("B2").Value2) & "|" & CStr(recovery.Range("D4").Value2)
    transactionId = CStr(recovery.Range("D5").Value2)
    For row = 6 To LastRow(recovery)
        If recovery.Cells(row, 4).Value2 = "Yes" Then
            AppendHistory CStr(recovery.Range("B3").Value2), identity, operation, CStr(recovery.Cells(row, 1).Value2), _
                recovery.Cells(row, 3).Value2, recovery.Cells(row, 2).Value2, transactionId
        End If
    Next
End Sub

Public Sub AppendHistory(ByVal address As String, ByVal identity As String, ByVal operation As String, _
    ByVal fieldId As String, ByVal oldValue As Variant, ByVal newValue As Variant, ByVal transactionId As String)
    Dim history As Worksheet
    Dim output(1 To 1, 1 To 10) As Variant
    Dim row As Long
    Set history = Sheet("History")
    For row = LastRow(history) To 2 Step -1
        If history.Cells(row, 10).Value2 <> transactionId Then Exit For
        If history.Cells(row, 6).Value2 = fieldId Then Exit Sub
    Next
    output(1, 1) = Format$(Now, "yyyy/mm/dd hh:nn:ss")
    output(1, 2) = Application.UserName
    output(1, 3) = address
    output(1, 4) = identity
    output(1, 5) = operation
    output(1, 6) = fieldId
    output(1, 7) = oldValue
    output(1, 8) = newValue
    output(1, 9) = "保存確認済み"
    output(1, 10) = transactionId
    With history.Cells(LastRow(history) + 1, 1).Resize(1, 10)
        .NumberFormat = "@"
        .Value2 = output
    End With
End Sub

Public Function NewTransactionId() As String
    NewTransactionId = Format$(Now, "yyyymmddhhnnss") & "-" & Format$(Timer * 1000, "00000000") & "-" & CStr(LastRow(Sheet("History")))
End Function

Public Sub ReloadCases(ByRef message As String)
    Dim year As Variant
    Dim loaded As Object
    Dim key As Variant
    Dim detail As String
    Dim invalidCount As Long
    Dim address As String
    Set Cases = NewDictionary()
    Set yearErrors = NewDictionary()
    Set yearColumnMaps = NewDictionary()
    address = LocateFixedLedger(detail)
    If Len(address) = 0 Then
        For Each year In years.Keys
            yearErrors.Add year, detail
        Next
        message = detail
        Exit Sub
    End If
    For Each year In years.Keys
        detail = ""
        Set loaded = ReadYear(CStr(year), detail, address)
        If loaded Is Nothing Then
            yearErrors.Add year, detail
            message = message & CStr(year) & "年度：" & detail & vbCrLf
        Else
            For Each key In loaded.Keys
                Cases.Add key, loaded(key)
                If Len(loaded(key).invalidReason) > 0 Then invalidCount = invalidCount + 1
            Next
        End If
    Next
    message = "読込件数 " & Cases.Count & "。正常年度の検索を利用できます。" & vbCrLf & message
    If invalidCount > 0 Then message = message & vbCrLf & "番号・型・重複などの要確認事案 " & invalidCount & "件は更新できません。詳細画面で理由を確認してください。"
End Sub

Public Function SearchRecords(ByVal query As String, ByVal office As String, ByVal excludeClosed As Boolean) As Collection
    Dim result As New Collection
    Dim closedStates As Object
    Dim key As Variant
    Dim record As CaseRecord
    Dim normalizedQuery As String
    Dim hasName As Boolean
    Dim hasOffice As Boolean
    Dim isClosed As Boolean
    normalizedQuery = SearchText(query)
    Set closedStates = ConfiguredList(15)
    For Each key In Cases.Keys
        Set record = Cases(key)
        hasName = Len(normalizedQuery) = 0
        If Not hasName Then
            hasName = InStr(1, SearchText(CStr(record.values("PersonName"))), normalizedQuery, vbTextCompare) > 0
            If Not hasName Then hasName = InStr(1, SearchText(CStr(record.values("Furigana"))), normalizedQuery, vbTextCompare) > 0
        End If
        hasOffice = Len(office) = 0 Or StrComp(CStr(record.values("Office")), office, vbTextCompare) = 0
        isClosed = excludeClosed And IsCaseClosed(record.values, closedStates)
        If hasName And hasOffice And Not isClosed Then result.Add CStr(key)
    Next
    Set SearchRecords = result
End Function

Private Function SearchText(ByVal text As String) As String
    Dim normalized As String
    normalized = Replace(Replace(Trim$(text), "　", ""), " ", "")
    On Error Resume Next
    SearchText = StrConv(StrConv(normalized, vbWide, 1041), vbKatakana, 1041)
    If Err.Number <> 0 Then SearchText = normalized
    On Error GoTo 0
End Function

Public Function ScreenValues(ByVal record As CaseRecord, ByVal year As String, ByRef message As String) As Object
    Dim result As Object
    Dim id As Variant
    Dim value As Variant
    Dim normalized As Variant
    Set result = NewDictionary()
    For Each id In fields.Keys
        If Not DetailFieldAvailable(CStr(id), year, record, message) And Not record Is Nothing Then
            result.Add id, ""
        Else
            value = FieldCell(CStr(id)).Value2
            If Not record Is Nothing Then
                If SameValue(value, DetailValue(CStr(id), record.values(id))) Then value = record.values(id)
            ElseIf Not DetailFieldAvailable(CStr(id), year, record, message) Then
                If CStr(value) = "（保存先未設定）" Then value = ""
            End If
            If Not NormalizeValue(CStr(id), value, normalized) Then
                message = "入力項目 " & CStr(id) & "の型または形式が不正です。FieldsのTypeと入力値を確認してください。"
                Exit Function
            End If
            result.Add id, normalized
        End If
    Next
    Set ScreenValues = result
End Function

Public Sub DisplayValues(ByVal values As Object)

Public Sub RefreshOneYear(ByVal year As String, ByRef message As String)
    Dim loaded As Object
    Dim key As Variant
    Dim detail As String
    Set loaded = ReadYear(year, detail)
    For Each key In Cases.Keys
        If Cases(key).sourceYear = year Then Cases.Remove key
    Next
    If loaded Is Nothing Then
        yearErrors(year) = detail
        message = message & vbCrLf & "保存は照合済みですが、年度の再読込に失敗しました。" & detail
    Else
        If yearErrors.Exists(year) Then yearErrors.Remove year
        For Each key In loaded.Keys
            Cases.Add key, loaded(key)
        Next
    End If
End Sub

    IsReservedUiCell = target.Parent.Name <> "詳細"
    If Not IsReservedUiCell Then
        IsReservedUiCell = Not Intersect(target, target.Parent.Range("B2:W12,Q32,C60:V64")) Is Nothing
    End If
End Function

Public Sub StartController()
    Dim message As String
    If Not BeginAction() Then Exit Sub
    On Error GoTo Failed
    If Not UiCanLeaveDetail() Then GoTo Finished
    If Not LoadConfiguration(message) Then GoTo Finished
    If PendingSave() Then
        message = "前回の保存結果が未確定です。再送せずオーナーに連絡してください。"
        GoTo Finished
    End If
    ReloadCases message
    hasDetail = False
    Set currentCase = Nothing
    ResetRelatedPanel
    RefreshSearchControls
    RenderSearch
Finished:
    isBusy = False
    If Len(message) > 0 Then Notify message
    Exit Sub
Failed:
    message = SafeFailure("読込", Err.Number)
    Resume Finished
End Sub

Public Sub FindRecords()
    If Cases Is Nothing Then
        StartController
    Else
        RenderSearch
    End If
End Sub

Public Sub PreviousPage()
    If matches Is Nothing Then Exit Sub
    If pageIndex = 0 Then Exit Sub
    pageIndex = pageIndex - 1
    DrawPage
End Sub

Public Sub NextPage()
    If matches Is Nothing Then Exit Sub
    If (pageIndex + 1) * PAGE_SIZE >= matches.Count Then Exit Sub
    pageIndex = pageIndex + 1
    DrawPage
End Sub

Public Sub ResultSelection(ByVal target As Range)
    If matches Is Nothing Then Exit Sub
    If target.CountLarge > 24 Then Exit Sub
    If Intersect(target, Sheet("検索").Range("C20:U33")) Is Nothing Then Exit Sub
    If target.Row = selectedRow Then Exit Sub
    HighlightResult target.Row
End Sub

Public Sub OpenSelected()
    Dim index As Long
    If matches Is Nothing Or selectedRow < FIRST_RESULT_ROW Then Exit Sub
    index = pageIndex * PAGE_SIZE + selectedRow - FIRST_RESULT_ROW + 1
    If index > matches.Count Then Exit Sub
    If Not UiCanLeaveDetail() Then Exit Sub
    ShowRecord CStr(matches(index))
End Sub

Public Sub NewCase()
    Dim year As Variant
    Dim targetYear As String
    If fields Is Nothing Or Cases Is Nothing Then
        StartController
        If fields Is Nothing Or Cases Is Nothing Then Exit Sub
    End If
    If Not UiCanLeaveDetail() Then Exit Sub
    For Each year In years.Keys
        If years(year)(2) = "Yes" And Not yearErrors.Exists(year) Then
            If Len(targetYear) > 0 Then
                targetYear = ""
                Exit For
            End If
            targetYear = CStr(year)
        End If
    Next
    Set currentCase = Nothing
    ShowDetail targetYear
End Sub

Public Sub ReturnToSearch()
    If Not UiCanLeaveDetail() Then Exit Sub
    hasDetail = False
    Set currentCase = Nothing
    ResetRelatedPanel
    Sheet("検索").Activate
End Sub

Public Sub SearchActivated()
    If isNavigating Or Not hasDetail Then Exit Sub
    isNavigating = True
    If UiCanLeaveDetail() Then
        hasDetail = False
        Set currentCase = Nothing
        ResetRelatedPanel
    Else
        Sheet("詳細").Activate
    End If
    isNavigating = False
End Sub

Private Function DetailHasChanges() As Boolean
    Dim id As Variant
    If Not hasDetail Or baseline Is Nothing Then Exit Function
    For Each id In fields.Keys
        If Not SameValue(FieldCell(CStr(id)).Value2, baseline(id)) Then
            DetailHasChanges = True
            Exit Function
        End If
    Next
    DetailHasChanges = (CStr(Sheet("詳細").Range("Q32").Value2) <> originalYear)
End Function

End Function

Public Function CommitDetail(ByVal showConfirmation As Boolean, ByRef message As String) As Boolean
    Dim record As CaseRecord
    Dim inputValues As Object
    Dim year As String
    Dim identity As String
    Dim isNew As Boolean
    If Not hasDetail Then Exit Function
    If Not BeginAction() Then Exit Function
    On Error GoTo Failed
    If relatedCurrentIndex > 0 Then
        If Sheet("詳細").Shapes(RELATED_LIST).ControlFormat.ListIndex <> relatedCurrentIndex Then
            message = "関連事案の選択と表示中の事案が一致していません。事案を選び直してください。"
            GoTo Finished
        End If
    End If
    year = CStr(Sheet("詳細").Range("Q32").Value2)
    isNew = currentCase Is Nothing
    If isNew Then
        Set record = New CaseRecord
        record.sourceYear = year
    Else
        Set record = currentCase
        If year <> record.sourceYear Then
            message = "既存事案の管理簿の年度は変更できません。"
            GoTo Finished
        End If
    End If
    If Not years.Exists(year) Then
        message = "保存先となる管理簿の年度を入力してください。"
        GoTo Finished
    End If
    Set inputValues = ScreenValues(currentCase, year, message)
    If inputValues Is Nothing Then GoTo Finished
    If Not ValidateValues(inputValues, message) Then GoTo Finished
    If Not ValidateDetailChoices(inputValues, currentCase, message) Then GoTo Finished
    If isNew Then
        identity = year & "|" & inputValues("CommonNumber")
    Else
        identity = record.Key
    End If
    If showConfirmation Then
        If MsgBox(year & "年度 / " & CStr(inputValues("PersonName")) & " / " & CStr(inputValues("CommonNumber")) & _
            vbCrLf & IIf(isNew, "新規登録しますか？", "変更した項目を保存しますか？"), _
            vbYesNo Or vbQuestion Or vbDefaultButton2, "CONTROLLER") <> vbYes Then GoTo Finished
    End If
    If SaveCase(record, inputValues, isNew, message) Then
        CommitDetail = True
        RefreshOneYear year, message
        If Cases.Exists(identity) Then
            ShowRecord identity
        Else
            RememberBaseline
        End If
        RenderSearch
    End If
Finished:
    isBusy = False
    Exit Function
Failed:
    If CommitDetail Then
        message = message & vbCrLf & "管理簿への保存は確認済みです。詳細の再表示に失敗したため、最新を再読込してください。"
    Else
        message = SafeFailure("保存", Err.Number)
    End If
    Resume Finished
End Function

Public Sub ConfirmSaveResult()

    If Not IsOwnerAuthorized() Then
        Notify "保存結果はオーナーに確認を依頼してください。"
        Exit Sub
    End If

        If ReconcileSave(message) Then message = message & vbCrLf & "最新を再読込して確認してください。"
    End If
    Notify message
End Sub

            SelectedField = CStr(id)

        End If

End Function

Private Sub BuildSearchFilters()
    Dim ws As Worksheet
    Dim offices As Object
    Dim office As Variant
    Dim index As Long

    Dim button As OptionButton
    Dim check As CheckBox
    Dim frame As GroupBox
    Dim anchor As Range
    Dim groupHeight As Double
    Set ws = Sheet("検索")
    Set offices = ConfiguredList(14)
    If SearchFiltersMatch(ws, offices) Then Exit Sub
    For index = ws.Shapes.Count To 1 Step -1
        If Left$(ws.Shapes(index).Name, Len(CONTROL_PREFIX)) = CONTROL_PREFIX Then ws.Shapes(index).Delete
    Next
    groupHeight = ws.Rows("12:13").Height + 2

    Set frame = ws.GroupBoxes.Add(ws.Range("F12").Left, ws.Range("F12").Top - 1, ws.Range("F12:U13").Width * (1 + offices.Count \ 8), groupHeight)
    frame.Name = CONTROL_PREFIX & "Group"
    frame.Caption = ""
    frame.Visible = False
    AddOfficeOption ws, 0, "すべて"
    index = 1
    For Each office In offices.Keys
        AddOfficeOption ws, index, CStr(office)
        index = index + 1
    Next
    ws.OptionButtons(CONTROL_PREFIX & "Office0").Value = xlOn
    Set anchor = ws.Range("C14")
    Set check = ws.CheckBoxes.Add(anchor.Left + 3, anchor.Top + (anchor.Height - 18) / 2, 120, 18)
    check.Name = CONTROL_PREFIX & "Closed"
    check.Caption = "終了分を除く"
    check.Value = xlOff
    check.OnAction = MacroAddress("FindRecords")
End Sub

Private Function SearchFiltersMatch(ByVal ws As Worksheet, ByVal offices As Object) As Boolean
    Dim index As Long
    Dim office As Variant
    On Error GoTo MissingControl
    If ws.OptionButtons.Count <> offices.Count + 1 Or ws.CheckBoxes.Count <> 1 Then Exit Function
    If ws.GroupBoxes(CONTROL_PREFIX & "Group").Name <> CONTROL_PREFIX & "Group" Then Exit Function
    If ws.CheckBoxes(CONTROL_PREFIX & "Closed").Caption <> "終了分を除く" Then Exit Function
    If ws.OptionButtons(CONTROL_PREFIX & "Office0").Caption <> "すべて" Then Exit Function
    index = 1
    For Each office In offices.Keys
        If ws.OptionButtons(CONTROL_PREFIX & "Office" & index).Caption <> CStr(office) Then Exit Function
        index = index + 1
    Next
    SearchFiltersMatch = True
MissingControl:
End Function

Private Sub AddOfficeOption(ByVal ws As Worksheet, ByVal index As Long, ByVal caption As String)
    Dim anchor As Range
    Dim button As OptionButton
    Dim rowIndex As Long
    rowIndex = 12 + (index Mod 8) \ 4
    Set anchor = ws.Cells(rowIndex, 6 + 4 * (index Mod 4) + 16 * (index \ 8))

Private Sub ShowRelatedCases(ByVal ws As Worksheet)
    Dim number As String
    Dim key As Variant
    Dim record As CaseRecord
    Dim members As Object
    Dim orderedKeys As Collection
    Set orderedKeys = New Collection
    If Not currentCase Is Nothing Then
        number = currentCase.PersonKey
        If Len(number) > 0 And Not Cases Is Nothing Then
            Set members = NewDictionary()
            For Each key In Cases.Keys
                Set record = Cases(key)
                If record.PersonKey = number Then members.Add key, True
            Next
            If Not relatedKeys Is Nothing Then
                For Each key In relatedKeys
                    If members.Exists(key) Then
                        orderedKeys.Add CStr(key)
                        members.Remove key
                    End If
                Next
            End If
            For Each key In members.Keys
                orderedKeys.Add CStr(key)
            Next
        End If
    End If
    ' A new list opens at the first row. Keep the current case visible on its first display.
    Set relatedKeys = orderedKeys
    DrawRelatedList ws
End Sub

Private Sub DrawRelatedList(ByVal ws As Worksheet)
    Dim control As ControlFormat
    Dim key As Variant
    Dim record As CaseRecord
    Dim caption As String
    Dim index As Long
    Dim needsRebuild As Boolean
    Dim wasSelecting As Boolean
    wasSelecting = isSelectingRelated
    isSelectingRelated = True
    On Error GoTo Failed
    Set control = ws.Shapes(RELATED_LIST).ControlFormat
    needsRebuild = (control.ListCount <> relatedKeys.Count)
    Dim captions As New Collection
    relatedCurrentIndex = 0
    For Each key In relatedKeys
        Set record = Cases(key)
        index = index + 1
        caption = "  " & RelatedColumn(record.sourceYear, 18) & RelatedColumn(CStr(record.values("CommonNumber")), 21) & _
            RelatedColumn(CStr(record.values("DocumentName")), 50) & CStr(record.values("Status"))
        captions.Add caption
        If Not needsRebuild Then needsRebuild = (control.List(index) <> caption)
        If CStr(key) = currentCaseKey Then relatedCurrentIndex = index
    Next
    If needsRebuild Then
        If relatedCurrentIndex > 3 Then
            caption = captions(relatedCurrentIndex)
            captions.Remove relatedCurrentIndex
            captions.Add caption, Before:=1
            relatedKeys.Remove relatedCurrentIndex
            relatedKeys.Add currentCaseKey, Before:=1
            relatedCurrentIndex = 1
        End If
        control.RemoveAllItems
        For Each key In captions
            control.AddItem CStr(key)
        Next
    End If
    If control.ListIndex <> relatedCurrentIndex Then control.ListIndex = relatedCurrentIndex
    ws.Range("C8").Value2 = "関連事案　全" & relatedKeys.Count & "件"
    UpdateDetailSummary ws
    isSelectingRelated = wasSelecting
    Exit Sub
Failed:
    Dim failure As Long
    failure = Err.Number
    isSelectingRelated = wasSelecting
    Err.Raise failure
End Sub

Private Function RelatedColumn(ByVal value As String, ByVal width As Long) As String
    Dim normalized As String
    Dim letter As String
    Dim used As Long
    Dim units As Long
    Dim index As Long
    normalized = Replace(Replace(value, vbCr, " "), vbLf, " ")
    For index = 1 To Len(normalized)
        letter = Mid$(normalized, index, 1)
        units = 1
        If (AscW(letter) And &HFFFF&) > 255 Then units = 2
        If used + units > width - 5 Then
            RelatedColumn = RelatedColumn & "…"
            used = used + 2
            Exit For
        End If
        RelatedColumn = RelatedColumn & letter
        used = used + units
    Next
    RelatedColumn = RelatedColumn & Space$(width - used)
End Function

Public Sub RelatedListChanged(Optional ByVal confirmedDecision As VbMsgBoxResult = 0)
    Dim control As ControlFormat
    Dim index As Long
    Dim key As String
    Dim decision As VbMsgBoxResult
    Dim message As String
    If isSelectingRelated Or isBusy Or Not hasDetail Then Exit Sub
    If relatedKeys Is Nothing Or currentCase Is Nothing Then Exit Sub
    isSelectingRelated = True
    On Error GoTo Failed
    Set control = Sheet("詳細").Shapes(RELATED_LIST).ControlFormat
    index = control.ListIndex
    If index = relatedCurrentIndex Then GoTo Finished
    If index < 1 Or index > relatedKeys.Count Then GoTo Finished
    key = CStr(relatedKeys(index))
    decision = vbNo
    If RelatedHasDraft() Then
        control.ListIndex = relatedCurrentIndex
        If confirmedDecision = 0 Then
            decision = MsgBox("変更が保存されていません。" & vbCrLf & _
                "はい：保存して切替" & vbCrLf & "いいえ：破棄して切替" & vbCrLf & "キャンセル：切替中止", _
                vbYesNoCancel Or vbQuestion Or vbDefaultButton3, "関連事案の切替")
        Else
            decision = confirmedDecision
        End If
    End If
    If Not SwitchRelatedCase(key, decision, message) Then control.ListIndex = relatedCurrentIndex
Finished:
    isSelectingRelated = False
    If Len(message) > 0 Then Notify message
    Exit Sub
Failed:
    message = SafeFailure("関連事案の切替", Err.Number)
    Resume Finished
End Sub


Private Function RelatedHasDraft() As Boolean
    RelatedHasDraft = DetailHasChanges()
End Function

Public Function SwitchRelatedCase(ByVal key As String, ByVal decision As VbMsgBoxResult, ByRef message As String) As Boolean
    Dim record As CaseRecord
    If Not hasDetail Or currentCase Is Nothing Or Cases Is Nothing Then Exit Function
    If Not Cases.Exists(key) Or key = currentCaseKey Then Exit Function
    Set record = Cases(key)
    If Len(currentCase.PersonKey) = 0 Or record.PersonKey <> currentCase.PersonKey Then Exit Function
    If decision = vbCancel Then Exit Function
    If RelatedHasDraft() Then
        Select Case decision
            Case vbYes
                If Not CommitDetail(False, message) Then Exit Function
                If Not Cases.Exists(key) Then
                    message = message & vbCrLf & "切替先が見つかりません。検索から開き直してください。"
                    Exit Function
                End If
                Set record = Cases(key)
                If currentCase Is Nothing Then Exit Function
                If record.PersonKey <> currentCase.PersonKey Then Exit Function
            Case vbNo
            Case Else
                Exit Function
        End Select
    End If
    ShowRecord key
    SwitchRelatedCase = True
End Function

Private Sub ImportDetailFields(ByVal onlyId As String)
    Dim answer As Variant
    Dim content As String
    Dim mode As VbMsgBoxResult
    Dim profile As String
    Dim config As Worksheet
    Dim result As Object
    Dim changes As Object
    Dim target As Range
    Dim id As Variant
    Dim issues As String
    Dim summary As String
    Dim restoreDetail As Boolean
    Dim overwrite As Long

    If Not hasDetail Or Not ActiveSheet Is Sheet("詳細") Then Exit Sub
    On Error GoTo Failed
    Set config = Sheet("Settings")
    answer = Application.InputBox("コピー元画面名を入力してください。", "取込ルール", CStr(config.Range("T2").Value2), Type:=2)
    If VarType(answer) = vbBoolean Then Exit Sub
    profile = Trim$(CStr(answer))
    If Len(profile) = 0 Then Exit Sub
    mode = MsgBox("はい：クリップボードを読む" & vbCrLf & "いいえ：別ブックに貼った範囲から読む", _
        vbYesNoCancel Or vbQuestion, "データ取込")
    Select Case mode
        Case vbYes: content = ReadWorkflowClipboard()
        Case vbNo
            restoreDetail = True
            content = ReadWorkflowPaste()
            Sheet("詳細").Activate
        Case Else: Exit Sub
    End Select
    If Len(content) = 0 Then
        Notify "文字列を取得できません。貼り付け範囲からの取込をお試しください。"
        Exit Sub
    End If
    Set result = ParseWorkflowImport(content, config.Range("T2:Y51").Value2, profile, issues)
    Set changes = NewDictionary()
    For Each id In result.Keys
        If Len(onlyId) = 0 Or id = onlyId Then
            If fields.Exists(id) Then
                Set target = FieldCell(CStr(id))
                If Not target.Locked Then
                    If CStr(target.Value2) <> CStr(result(id)) Then

                        changes.Add id, result(id)
                        summary = summary & fields(id)(0) & vbCrLf
                    End If
                End If
            End If
        End If
    Next
    If changes.Count = 0 Then
        Notify "入力できる新しい値がありません。" & vbCrLf & Left$(issues, 500)
        Exit Sub
    End If
    If MsgBox("次の項目を入力欄へ反映しますか？ 既存値の置換 " & overwrite & "件" & vbCrLf & _
        summary & Left$(issues, 350), vbYesNo Or vbQuestion Or vbDefaultButton2) <> vbYes Then Exit Sub

        PutValue FieldCell(CStr(id)), changes(id)
    Next
    Notify changes.Count & "項目を入力欄に反映しました。保存前に内容を確認してください。"
    Exit Sub
Failed:

Public Sub RefreshEditedStatus(ByVal target As Range)
    Dim values As Object
    Dim id As Variant
    Dim events As Boolean
    Dim lastStatus As Variant
    If isRenderingDetail Or Not hasDetail Or fields Is Nothing Or baseline Is Nothing Then Exit Sub
    If target.CountLarge <> 1 Then Exit Sub
    If Intersect(target, Union(FieldCell("Status"), FieldCell("DraftDate"), FieldCell("ApprovedDate"), FieldCell("IssuedDate"), FieldCell("CompletedDate"))) Is Nothing Then Exit Sub
    events = Application.EnableEvents
    On Error GoTo Finished
    Application.EnableEvents = False
    Set values = NewDictionary()
    For Each id In Array("Status", "DraftDate", "ApprovedDate", "IssuedDate", "CompletedDate")
        If IsError(FieldCell(CStr(id)).Value2) Then GoTo Finished
        values.Add id, FieldCell(CStr(id)).Value2
    Next
    If Not Intersect(target, FieldCell("Status")) Is Nothing Then
        If IsTransferStatus(CStr(values("Status"))) And Not HasCompletionValue(values, "CompletedDate") Then
            lastStatus = baseline("Status")
            Application.Undo
            If IsTransferStatus(CStr(FieldCell("Status").Value2)) Then PutValue FieldCell("Status"), lastStatus
            Notify "移行による終了状態には完了日が必要です。状態変更を取り消しました。"
            GoTo Finished
        End If
    End If
    For Each id In Array("DraftDate", "ApprovedDate", "IssuedDate", "CompletedDate")
        If HasCompletionValue(values, CStr(id)) And Not IsNumeric(values(id)) Then GoTo Finished
    Next
    ApplyWorkflowStatus values, CStr(baseline("Status"))
    If Not SameValue(FieldCell("Status").Value2, values("Status")) Then PutValue FieldCell("Status"), values("Status")
    If Not SameValue(FieldCell("CompletedDate").Value2, values("CompletedDate")) Then PutValue FieldCell("CompletedDate"), values("CompletedDate"), True
Finished:
    Application.EnableEvents = events
End Sub

Public Function LocateFixedLedger(ByRef message As String, Optional ByVal isWrite As Boolean = False) As String
    Dim first As String
    Dim second As String
    Dim firstResult As String
    Dim secondResult As String
    Dim firstState As Long
    Dim secondState As Long
    first = Setting("StaffLedger")
    second = Setting("GeneralLedger")
    If AddressFileName(first) <> "管理簿.xlsm" Or AddressFileName(second) <> "管理簿.xlsm" Then
        message = "固定アドレスは常勤と非常勤の管理簿.xlsmを指定してください。"
        Exit Function
    End If
    If Not IsDirectWorkbookAddress(first) Or Not IsDirectWorkbookAddress(second) Or IsSameWorkbookAddress(first, second) Then
        message = "常勤側・非常勤側の異なる直接アドレスを設定してください。"
        Exit Function
    End If
    If ControllerFolderRole(first) <> "Staff" Or ControllerFolderRole(second) <> "General" Then
        message = "固定アドレスのフォルダと常勤・非常勤フォルダの登録が一致しません。"
        Exit Function
    End If
    If Setting("Transport") = "SharePoint" Then
        If Not IsHttpsAddress(first) Or Not IsHttpsAddress(second) Then
            message = "SharePoint運用では2つとも直接URLを指定してください。"
            Exit Function
        End If
    ElseIf Setting("Transport") <> "Local" Or IsHttpsAddress(first) Or IsHttpsAddress(second) Then
        message = "保存方式と固定アドレスが一致しません。"
        Exit Function
    End If
    firstState = ProbeExcelAddress(first, firstResult)
    secondState = ProbeExcelAddress(second, secondResult)
    If firstState = 1 And secondState = 1 Then
        message = "両方の候補に管理簿があります。正式データを確定してください。"
    ElseIf firstState = 0 And secondState = 0 Then
        message = "両方の候補に管理簿がありません。アドレスと所在を確認してください。"
    ElseIf firstState = 1 And (secondState = 0 Or Not isWrite) Then
        LocateFixedLedger = firstResult
        message = ""
    ElseIf secondState = 1 And (firstState = 0 Or Not isWrite) Then
        LocateFixedLedger = secondResult
        message = ""
    Else
        message = "管理簿の所在を確定できません。アクセス権、開いているブック、ネットワークを確認してください。"
    End If
End Function

Private Function ProbeExcelAddress(ByVal address As String, ByRef actualAddress As String) As Long
    Dim book As Workbook

Private Function SharePointEndpointBase() As String
    Dim url As String
    url = Setting("SharePointSite")
    If Right$(url, 1) = "/" Then url = Left$(url, Len(url) - 1)
    SharePointEndpointBase = ExcelOpenAddress(url)
End Function

Private Function ApiFile(ByVal address As String) As String
    Dim site As String
    Dim file As String
    Dim slash As Long
    site = CanonicalAddress(Setting("SharePointSite"))
    file = CanonicalAddress(address)
    If Not IsHttpsAddress(site) Or Not IsHttpsAddress(file) Then Exit Function
    If StrComp(Left$(file, Len(site) + 1), site & "/", vbTextCompare) <> 0 Then Exit Function
    slash = InStr(9, file, "/")
    ApiFile = SharePointEndpointBase() & "/_api/web/GetFileByServerRelativePath(decodedUrl='" & _
        Application.WorksheetFunction.EncodeURL(Replace(Mid$(file, slash), "'", "''")) & "')"
End Function

Private Function SharePointRequest(ByVal method As String, ByVal url As String, ByVal body As String, _
    ByRef message As String, Optional ByVal digest As String = "") As Object
    Dim request As Object
    On Error GoTo Failed
    If Not IsHttpsAddress(url) Then
        message = "SharePointサイトと対象の直接URLが一致しません。"
        Exit Function
    End If
    Set request = CreateObject("WinHttp.WinHttpRequest.5.1")
    request.SetTimeouts 5000, 8000, 15000, 25000
    request.Option(6) = False
    request.Open method, url, False
    request.SetAutoLogonPolicy 0
    request.SetRequestHeader "Accept", "application/atom+xml"
    If Len(digest) > 0 Then request.SetRequestHeader "X-RequestDigest", digest
    If Len(body) > 0 Then request.SetRequestHeader "Content-Type", "application/json;odata=verbose; charset=utf-8"
    If Len(body) = 0 Then
        request.Send
    Else
        request.Send body
    End If
    Set SharePointRequest = request
    Exit Function
Failed:
    message = "SharePoint通信に失敗しました。エラー " & Err.Number & "。変更要求は再送せず結果を照合してください。"
End Function

Private Function ResponseXmlValue(ByVal request As Object, ByVal name As String) As String
    Dim xml As Object
    Dim node As Object
    Set xml = CreateObject("MSXML2.DOMDocument.6.0")
    xml.async = False
    xml.resolveExternals = False
    xml.SetProperty "ProhibitDTD", True
    If Not xml.LoadXML(request.ResponseText) Then Exit Function
    Set node = xml.SelectSingleNode("//*[local-name()='" & name & "']")
    If Not node Is Nothing Then ResponseXmlValue = node.Text
End Function

Public Function FileState(ByVal address As String, ByRef message As String) As Object
    Dim result As Object
    Dim file As Object
    Dim request As Object
    Dim status As Long
    Dim endpoint As String
    On Error GoTo Failed
    message = ""
    Set result = NewDictionary()
    result.Add "Exists", False
    result.Add "Id", ""
    result.Add "ETag", ""
    result.Add "Size", ""
    result.Add "Locked", False
    If IsHttpsAddress(address) Then
        endpoint = ApiFile(address)
        Set request = SharePointRequest("GET", endpoint, "", message)
        If request Is Nothing Then Exit Function
        Select Case request.Status
            Case 200
                result("Id") = ResponseXmlValue(request, "UniqueId")
                result("ETag") = ResponseXmlValue(request, "ETag")
                If Len(result("ETag")) = 0 Then result("ETag") = request.GetResponseHeader("ETag")
                result("Size") = ResponseXmlValue(request, "Length")
                If Len(result("Id")) = 0 Or Len(result("ETag")) = 0 Or Len(result("Size")) = 0 Then GoTo InvalidResponse
                result("Exists") = True
            Case 404
                If InStr(ResponseXmlValue(request, "code"), "System.IO.FileNotFoundException") = 0 Then GoTo InvalidResponse
            Case Else
                message = "SharePointの所在を確定できません。HTTP " & request.Status & "。認証とアクセス権を確認してください。"
                Exit Function
        End Select
    Else
        If Len(CanonicalAddress(address)) = 0 Then Exit Function
        On Error Resume Next
        Set file = CreateObject("Scripting.FileSystemObject").GetFile(address)
        status = Err.Number
        Err.Clear
        On Error GoTo Failed
        If status = 0 Then
            result("Exists") = True
            result("Size") = CStr(file.Size)
            result("ETag") = CStr(CDbl(file.DateLastModified)) & "|" & result("Size")
        ElseIf status <> 53 And status <> 76 Then
            message = "ファイルの所在を確認できません。エラー " & status
            Exit Function
        End If
    End If
    Set FileState = result
    Exit Function
InvalidResponse:
    message = "SharePointの応答から所在と識別情報を確定できません。"
    Exit Function
Failed:
    message = "所在確認に失敗しました。エラー " & Err.Number
End Function

Private Function JsonString(ByVal value As String) As String
    Dim i As Long
    Dim code As Long
    JsonString = """"
    For i = 1 To Len(value)
        code = AscW(Mid$(value, i, 1)) And &HFFFF&
        If code < 32 Or code > 126 Or code = 34 Or code = 92 Then
            JsonString = JsonString & "\u" & Right$("0000" & Hex$(code), 4)
        Else
            JsonString = JsonString & Chr$(code)
        End If
    Next
    JsonString = JsonString & """"
End Function

Private Function ResourcePathJson(ByVal address As String) As String
    ResourcePathJson = "{""__metadata"":{""type"":""SP.ResourcePath""},""DecodedUrl"":" & JsonString(CanonicalAddress(address)) & "}"
End Function

Private Function ServerCopyOrMove(ByVal source As String, ByVal destination As String, ByVal isMove As Boolean, ByRef message As String) As Boolean
    Dim site As String
    Dim digest As String
    Dim endpoint As String
    Dim request As Object
    Dim body As String
    On Error GoTo Failed
    If IsHttpsAddress(source) Then
        If Len(ApiFile(source)) = 0 Or Len(ApiFile(destination)) = 0 Then
            message = "移動元・移動先・退避先は設定したサイト内に指定してください。"
            Exit Function
        End If
        site = SharePointEndpointBase()
        Set request = SharePointRequest("POST", site & "/_api/contextinfo", "", message)
        If request Is Nothing Then Exit Function
        If request.Status <> 200 Then
            message = "変更要求の認証を確立できません。HTTP " & request.Status
            Exit Function
        End If
        digest = ResponseXmlValue(request, "FormDigestValue")
        If Len(digest) = 0 Then
            message = "SharePointの変更用ダイジェストを取得できません。"
            Exit Function
        End If
        endpoint = site & "/_api/SP.MoveCopyUtil." & IIf(isMove, "MoveFileByPath", "CopyFileByPath") & "(overwrite=@a1)?@a1=false"
        body = "{""srcPath"":" & ResourcePathJson(source) & ",""destPath"":" & ResourcePathJson(destination) & _
            ",""options"":{""__metadata"":{""type"":""SP.MoveCopyOptions""},""KeepBoth"":false,""ShouldBypassSharedLocks"":false}}"
        Set request = SharePointRequest("POST", endpoint, body, message, digest)
        If request Is Nothing Then Exit Function
        If request.Status < 200 Or request.Status >= 300 Then
            message = "変更結果を確認してください。HTTP " & request.Status & "。要求は再送しません。"
            Exit Function
        End If
    ElseIf isMove Then
        Name source As destination
    Else
        CreateObject("Scripting.FileSystemObject").CopyFile source, destination, False
    End If
    ServerCopyOrMove = True
    Exit Function
Failed:
    message = "コピーまたは移動に失敗しました。再送せず結果を照合してください。エラー " & Err.Number
End Function

Private Function FileContent(ByVal address As String, ByRef message As String) As Variant
    Dim request As Object
    Dim handle As Integer
    Dim bytes() As Byte
    On Error GoTo Failed
    If IsHttpsAddress(address) Then
        If Len(ApiFile(address)) = 0 Then Exit Function
        Set request = SharePointRequest("GET", ApiFile(address) & "/$value", "", message)
        If request Is Nothing Then Exit Function
        If request.Status <> 200 Then
            message = "ファイル内容を確認できません。HTTP " & request.Status
            Exit Function
        End If
        FileContent = request.ResponseBody
    Else
        handle = FreeFile
        Open address For Binary Access Read Lock Write As #handle
        If LOF(handle) > 0 Then
            ReDim bytes(0 To LOF(handle) - 1)
            Get #handle, , bytes
            FileContent = bytes
        End If
        Close #handle
    End If
    Exit Function
Failed:
    If handle > 0 Then Close #handle
    message = "ファイル内容を読み取れません。エラー " & Err.Number
End Function

Private Function IsEqualFileContent(ByVal first As String, ByVal second As String, ByRef message As String) As Boolean
    Dim leftBytes As Variant
    Dim rightBytes As Variant
    Dim i As Long
    leftBytes = FileContent(first, message)
    If Not IsArray(leftBytes) Then Exit Function
    rightBytes = FileContent(second, message)
    If Not IsArray(rightBytes) Then Exit Function
    If UBound(leftBytes) <> UBound(rightBytes) Then Exit Function
    For i = LBound(leftBytes) To UBound(leftBytes)
        If leftBytes(i) <> rightBytes(i) Then Exit Function
    Next
    IsEqualFileContent = True
End Function

Public Function ExecuteBusiness(ByVal isStart As Boolean, ByRef message As String) As Boolean
    Dim source As String
    Dim destination As String
    Dim backup As String
    Dim folder As String
    Dim before As Object
    Dim after As Object
    Dim target As Object
    Dim ws As Worksheet
    Dim book As Workbook
    Dim separator As String
    If Not CanMoveLedger(message) Then Exit Function
    If Not BeginAction() Then Exit Function
    On Error GoTo Failed
    Set ws = Sheet("MoveRecovery")
    If PendingSave() Or Len(CStr(ws.Range("B1").Value2)) > 0 Then
        message = "未確定の保存・移動があります。確認後に操作してください。"
        GoTo Finished
    End If
    source = Setting(IIf(isStart, "StaffLedger", "GeneralLedger"))
    destination = Setting(IIf(isStart, "GeneralLedger", "StaffLedger"))
    If AddressFileName(source) <> "管理簿.xlsm" Or AddressFileName(destination) <> "管理簿.xlsm" Then
        message = "固定アドレスには管理簿.xlsmを指定してください。"
        GoTo Finished
    End If
    If ControllerFolderRole(source) <> IIf(isStart, "Staff", "General") Or _
        ControllerFolderRole(destination) <> IIf(isStart, "General", "Staff") Then
        message = "管理簿のアドレスと常勤・非常勤フォルダの設定が一致しません。"
        GoTo Finished
    End If
    For Each book In Application.Workbooks
        If IsSameWorkbookAddress(book.FullName, source) Or IsSameWorkbookAddress(book.FullName, destination) Then
            message = "管理簿を閉じてから操作してください。"
            GoTo Finished
        End If
    Next
    Set before = FileState(source, message)
    If before Is Nothing Then GoTo Finished
    Set target = FileState(destination, message)
    If target Is Nothing Then GoTo Finished
    If Not before("Exists") Or target("Exists") Then
        message = "移動元だけに管理簿がある場合に実行できます。両候補を確認してください。"
        GoTo Finished
    End If
    folder = Setting("BackupFolder")
    If Len(CanonicalAddress(folder)) = 0 Or IsHttpsAddress(folder) <> IsHttpsAddress(source) Then
        message = "管理簿と同じ保存方式のバックアップ先を設定してください。"
        GoTo Finished
    End If
    separator = IIf(IsHttpsAddress(folder), "/", "\")
    If Right$(folder, 1) <> separator Then folder = folder & separator
    backup = folder & "管理簿_" & NewTransactionId() & ".xlsm"
    PutValue ws.Range("B1"), "バックアップ確認待ち"
    PutValue ws.Range("B2"), source
    PutValue ws.Range("B3"), destination
    PutValue ws.Range("B4"), CStr(before("Id"))
    PutValue ws.Range("B5"), backup
    PutValue ws.Range("B6"), CStr(before("ETag"))
    PutValue ws.Range("B7"), Format$(Now, "yyyy/mm/dd hh:nn:ss")
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Failed
    If Not ServerCopyOrMove(source, backup, False, message) Then GoTo Finished
    If Not IsEqualFileContent(source, backup, message) Then
        message = "バックアップと管理簿の内容が一致しないか、比較できません。移動していません。"
        GoTo Finished
    End If
    Set after = FileState(source, message)
    If after Is Nothing Then GoTo Finished
    If Not after("Exists") Or after("Id") <> before("Id") Or after("ETag") <> before("ETag") Then
        message = "退避後に管理簿が変わりました。移動していません。"
        GoTo Finished
    End If
    PutValue ws.Range("B1"), "移動結果確認待ち"
    PutValue ws.Range("B8"), "バックアップ内容確認済み"
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Failed
    Set after = FileState(source, message)
    If after Is Nothing Then GoTo Finished
    If Not after("Exists") Or after("Id") <> before("Id") Or after("ETag") <> before("ETag") Then
        message = "移動要求の送信前に元ファイルが変わりました。記録を照合してください。"
        GoTo Finished
    End If
    If Not ServerCopyOrMove(source, destination, True, message) Then GoTo Finished
    ExecuteBusiness = VerifyMoveJournal(message)
Finished:
    isBusy = False
    Exit Function
Failed:
    message = "朝夕の処理が中断されました。記録と両方の候補を照合し、確認なしで再送しないでください。エラー " & Err.Number
    Resume Finished
End Function

Private Function VerifyMoveJournal(ByRef message As String) As Boolean
    Dim ws As Worksheet
    Dim source As Object
    Dim destination As Object
    Dim oldAddress As String
    Dim newAddress As String
    Set ws = Sheet("MoveRecovery")
    oldAddress = CStr(ws.Range("B2").Value2)
    newAddress = CStr(ws.Range("B3").Value2)
    Set source = FileState(oldAddress, message)
    If source Is Nothing Then Exit Function
    Set destination = FileState(newAddress, message)
    If destination Is Nothing Then Exit Function
    If source("Exists") Or Not destination("Exists") Or CStr(ws.Range("B1").Value2) <> "移動結果確認待ち" Then GoTo Unconfirmed
    If CStr(ws.Range("B4").Value2) <> CStr(destination("Id")) Then GoTo Unconfirmed
    If Not IsEqualFileContent(newAddress, CStr(ws.Range("B5").Value2), message) Then GoTo Unconfirmed
    PutValue ws.Range("B9"), "移動・内容確認済み " & Format$(Now, "yyyy/mm/dd hh:nn:ss")
    ws.Range("B1").ClearContents
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then
        PutValue ws.Range("B1"), "移動結果確認待ち"
        GoTo Unconfirmed
    End If
    message = "管理簿の移動と内容の一致を確認しました。最新を再読込してください。"
    VerifyMoveJournal = True
    Exit Function
Unconfirmed:
    message = "移動結果を確定できません。移動元・移動先・バックアップを確認してください。再送しません。"
End Function

End Sub

Public Sub RestoreBackup()
    Dim address As Variant
    Dim target As Variant
    Dim state As Object
    Dim message As String
    If Not IsOwnerAuthorized() Or ControllerFolderRole(ThisWorkbook.FullName) <> "Staff" Then
        Notify "オーナー認証と常勤フォルダからの起動が必要です。"
        Exit Sub
    End If
    address = Application.InputBox("退避した管理簿の直接URLを入力してください。", "バックアップの復旧確認コピー", Type:=2)
    If VarType(address) = vbBoolean Then Exit Sub
    target = Application.InputBox("同じ退避フォルダ内の未使用の直接URLを入力してください。名前は復旧確認_から始めます。", "復旧先", Type:=2)
    If VarType(target) = vbBoolean Then Exit Sub
    If Not IsSameWorkbookAddress(AddressFolder(CStr(address)), Setting("BackupFolder")) Or _
        Not IsSameWorkbookAddress(AddressFolder(CStr(target)), Setting("BackupFolder")) Or _
        Left$(AddressFileName(CStr(target)), 5) <> "復旧確認_" Or _
        Not IsDirectWorkbookAddress(CStr(target)) Then
        Notify "元・先の退避フォルダと新しい復旧確認ファイル名を確認してください。"
        Exit Sub
    End If
    Set state = FileState(CStr(target), message)
    If state Is Nothing Then
        Notify message
        Exit Sub
    End If
    If state("Exists") Then
        Notify "復旧先に同名のファイルがあります。上書きしません。"
        Exit Sub
    End If
    If MsgBox("退避元から復旧確認ファイルを作成しますか？", vbYesNo Or vbQuestion Or vbDefaultButton2) <> vbYes Then Exit Sub
    If ServerCopyOrMove(CStr(address), CStr(target), False, message) Then
        If IsEqualFileContent(CStr(address), CStr(target), message) Then
            message = "復旧確認ファイルを作成し、元のバックアップとの一致を確認しました。"
        Else
            message = "復旧確認ファイルの内容を確定できません。再送せず確認してください。"
        End If
    End If
    Notify message
End Sub

Public Function IsImportRule(ByVal rule As Variant) As Boolean
    If Len(Trim$(rule(0))) = 0 Or Len(Trim$(rule(1))) = 0 Then Exit Function
    Select Case rule(3)
        Case "Right"
            Select Case rule(4)
                Case "Auto", "Tab", "Colon":
                Case Else: Exit Function
            End Select
        Case "Next"
            If rule(4) <> "Line" Then Exit Function
        Case Else: Exit Function
    End Select
    Select Case rule(2)
        Case "CommonNumber": IsImportRule = (rule(5) = "Digits18")
        Case "PostalCode": IsImportRule = (rule(5) = "Postal")
        Case "PhoneNumber": IsImportRule = (rule(5) = "Phone")
        Case "PersonName", "Furigana", "Address": IsImportRule = (rule(5) = "Trim")
    End Select
End Function

Public Function ParseWorkflowImport(ByVal inputText As String, ByVal ruleData As Variant, ByVal profile As String, ByRef issues As String) As Object
    Dim result As Object
    Dim seen As Object
    Dim rule As Variant
    Dim lines As Variant
    Dim lineText As String
    Dim name As String
    Dim value As String
    Dim splitAt As Long
    Dim row As Long
    Dim i As Long
    Dim count As Long

    Set result = NewDictionary()
    Set seen = NewDictionary()
    issues = ""
    lines = Split(Replace(Replace(inputText, vbCrLf, vbLf), vbCr, vbLf), vbLf)
    For row = LBound(ruleData, 1) To UBound(ruleData, 1)
        If CStr(ruleData(row, 1)) = profile And Len(profile) > 0 Then
            rule = Array(CStr(ruleData(row, 1)), CStr(ruleData(row, 2)), CStr(ruleData(row, 3)), _
                CStr(ruleData(row, 4)), CStr(ruleData(row, 5)), CStr(ruleData(row, 6)))
            If Not IsImportRule(rule) Then
                issues = issues & "抽出ルールが無効です。行 " & row & vbCrLf
            Else
                count = 0
                value = ""
                For i = LBound(lines) To UBound(lines)
                    lineText = CleanImportText(CStr(lines(i)))
                    If rule(3) = "Next" Then
                        If lineText = rule(1) Then
                            count = count + 1
                            If i < UBound(lines) Then value = CleanImportText(CStr(lines(i + 1)))
                        End If
                    Else
                        splitAt = ImportSeparator(lineText, CStr(rule(4)))
                        If splitAt > 0 Then
                            name = CleanImportText(Left$(lineText, splitAt - 1))
                            If name = rule(1) Then
                                count = count + 1
                                value = CleanImportText(Mid$(lineText, splitAt + 1))
                            End If
                        End If
                    End If
                Next
                If count > 0 Then
                    If seen.Exists(rule(2)) Or count > 1 Then
                        If result.Exists(rule(2)) Then result.Remove rule(2)
                        issues = issues & rule(1) & "：候補が複数あります。" & vbCrLf
                    ElseIf ImportValueValid(CStr(rule(5)), value) Then
                        result.Add rule(2), value
                    Else
                        issues = issues & rule(1) & "：空欄または形式不正です。" & vbCrLf
                    End If
                    seen(rule(2)) = True
                End If
            End If
        End If
    Next
    Set ParseWorkflowImport = result
End Function

Private Function CleanImportText(ByVal text As String) As String
    CleanImportText = Trim$(Replace(Replace(Replace(text, ChrW(&HFEFF), ""), ChrW(&H200B), ""), ChrW(&H3000), " "))
End Function

Private Function ImportSeparator(ByVal lineText As String, ByVal mode As String) As Long
    If mode <> "Colon" Then ImportSeparator = InStr(lineText, vbTab)
    If ImportSeparator = 0 And mode <> "Tab" Then ImportSeparator = InStr(Replace(lineText, "：", ":"), ":")
End Function

Private Function ImportValueValid(ByVal mode As String, ByVal value As String) As Boolean
    If Len(value) = 0 Then Exit Function
    Select Case mode
        Case "Digits18": ImportValueValid = IsDigits(value, 18)
        Case "Postal"
            ImportValueValid = IsDigits(Replace(value, "-", ""), 7)
        Case "Phone"
            value = Replace(value, "-", "")
            If Len(value) >= 9 And Len(value) <= 11 Then ImportValueValid = IsDigits(value, Len(value))
        Case "Trim": ImportValueValid = True
    End Select
End Function

Public Function WorkflowImportPreview(ByVal inputText As String, ByVal rules As Variant, ByVal profile As String) As String
    Dim values As Object

End Function

Private Function IsMatchingReviewCase(ByVal left As CaseRecord, ByVal right As CaseRecord, ByVal rules As Object, ByVal isSingle As Boolean) As Boolean
    Dim field As Variant
    Dim fieldValue As Variant
    Dim used As Boolean
    For Each field In rules.Keys
        Select Case field
            Case "CommonNumber":
            Case "SourceYear"
                If rules(field)(1) And Len(ReviewText(right.values(field))) > 0 Then
                    If left.sourceYear <> ReviewText(right.values(field)) Then Exit Function
                    used = True
                End If
            Case Else
                If rules(field)(1) Or field = "TaxYear" Or field = "DocumentName" Or field = "SubmissionFormat" Then
                    fieldValue = right.values(field)
                    If Len(ReviewText(fieldValue)) > 0 Then
                        If Not left.sourceColumns.Exists(field) Then Exit Function
                        If StrComp(ReviewText(left.values(field)), ReviewText(fieldValue), vbTextCompare) <> 0 Then Exit Function
                        used = True
                    End If
                End If
        End Select
    Next
    IsMatchingReviewCase = used Or isSingle
End Function

Private Function ReviewText(ByVal value As Variant) As String
    If Not IsError(value) And Not IsNull(value) Then ReviewText = Trim$(CStr(value))
End Function

Public Sub BuildReviewReport(ByVal entries As Collection, ByVal rules As Object)
    Dim people As Object
    Dim internalCase As CaseRecord
    Dim outside As CaseRecord
    Dim target As CaseRecord
    Dim group As Collection
    Dim matches As Collection
    Dim tentative As Object
    Dim counts As Object
    Dim commonChecks As Object
    Dim issues As Collection
    Dim record As Variant
    Dim field As Variant
    Dim id As Variant
    Dim number As String
    Dim editableKey As String
    Dim message As String
    Dim priority As Long

    Set people = NewDictionary()
    Set tentative = NewDictionary()
    Set counts = NewDictionary()
    Set commonChecks = NewDictionary()
    Set issues = New Collection
    For Each id In Cases.Keys
        Set internalCase = Cases(id)
        If Len(internalCase.PersonKey) > 0 Then AddReviewMember people, internalCase.PersonKey, internalCase
    Next
    For Each outside In entries
        number = outside.PersonKey
        Set matches = New Collection
        If people.Exists(number) Then
            Set group = people(number)
            For Each internalCase In group
                If IsMatchingReviewCase(internalCase, outside, rules, group.Count = 1) Then matches.Add internalCase
            Next
            If matches.Count = 1 Then
                Set target = matches(1)
                tentative.Add CStr(outside.sourceRow), target
                If Not counts.Exists(target.Key) Then counts.Add target.Key, 0
                counts(target.Key) = CLng(counts(target.Key)) + 1
            ElseIf matches.Count > 1 Then
                issues.Add Array(number, outside.sourceRow, "重複候補", "", "", "", "", ReviewCategoryPriority(1))
            Else
                issues.Add Array(number, outside.sourceRow, "事案対応不明", "", "", "", "", ReviewCategoryPriority(2))
            End If
        ElseIf Len(number) = 0 Then
            issues.Add Array("", outside.sourceRow, "18桁の共通番号を確認", "", "", "", "", ReviewCategoryPriority(2))
        Else
            issues.Add Array(number, outside.sourceRow, "登録漏れ候補", "", "", "", "", ReviewCategoryPriority(1))
        End If
    Next
    For Each outside In entries
        If tentative.Exists(CStr(outside.sourceRow)) Then
            Set target = tentative(CStr(outside.sourceRow))
            If counts(target.Key) > 1 Or Len(target.invalidReason) > 0 Then
                issues.Add Array(outside.PersonKey, outside.sourceRow, "重複・要確認", "", "", "", "", ReviewCategoryPriority(1))
            Else
                For Each field In rules.Keys
                    If field <> "SourceYear" And field <> "CommonNumber" Then
                        If target.sourceColumns.Exists(field) Then
                            If Not SameValue(target.values(field), outside.values(field)) Then
                                message = "不一致"
                                priority = 5
                                If Len(ReviewText(target.values(field))) = 0 Then
                                    message = "記入漏れ・補完候補"
                                    priority = IIf(fields(field)(4) = "Yes", 3, 4)
                                ElseIf Len(ReviewText(outside.values(field))) = 0 Then
                                    message = "整理簿の記入漏れ"
                                End If
                                If rules(field)(1) Or field = "TaxYear" Or field = "DocumentName" Or field = "SubmissionFormat" Then priority = 2
                                priority = ReviewPriority(CStr(field), priority)
                                id = outside.PersonKey & "|" & field
                                Select Case field
                                    Case "PersonName", "Furigana", "Address", "PostalCode", "PhoneNumber"
                                        If Not commonChecks.Exists(id) Then
                                            commonChecks.Add id, True
                                            Set group = people(outside.PersonKey)
                                            editableKey = ""
                                            If group.Count = 1 Then editableKey = target.Key
                                            issues.Add Array(outside.PersonKey, outside.sourceRow, message, field, _
                                                ReviewText(target.values(field)), ReviewText(outside.values(field)), editableKey, priority)
                                        End If
                                    Case Else
                                        issues.Add Array(outside.PersonKey, outside.sourceRow, message, field, _
                                            ReviewText(target.values(field)), ReviewText(outside.values(field)), target.Key, priority)
                                End Select
                            End If
                        End If
                    End If
                Next
            End If
        End If
    Next
    DisplayReviewReport issues
End Sub

Public Function SaveReviewCorrections() As String
    Dim plans As Object
    Dim rowsByCase As Object
    Dim plan As Object
    Dim chosen As Collection
    Dim completed As Collection
    Dim entry As Variant
    Dim key As Variant
    Dim row As Variant
    Dim value As Variant
    Dim record As CaseRecord
    Dim message As String
    Dim good As Long
    Dim bad As Long
    Dim yearsToRefresh As Object
    Dim attempted As Long

    If reviewPage Is Nothing Or reviewRows Is Nothing Then Exit Function
    If Not BeginAction() Then Exit Function
    On Error GoTo Failed
    Set plans = NewDictionary()
    Set rowsByCase = NewDictionary()
    Set yearsToRefresh = NewDictionary()
    Set completed = New Collection
    reviewPage.Unprotect
    For Each row In reviewRows.Keys
        entry = reviewRows(row)
        value = reviewPage.Cells(CLng(row), 7).Value2
        If reviewPage.Cells(CLng(row), 7).HasFormula Or IsError(value) Then
            reviewPage.Cells(CLng(row), 8).Value2 = "数式・エラーは入力できません"
            bad = bad + 1
        ElseIf Len(CStr(value)) > 0 Then
            key = entry(0)
            If Not plans.Exists(key) Then
                Set plan = NewDictionary()
                plans.Add key, plan
                Set chosen = New Collection
                rowsByCase.Add key, chosen
            End If
            Set plan = plans(key)
            Set chosen = rowsByCase(key)
            chosen.Add CLng(row)
            If plan.Exists(entry(1)) Then plan("__Conflict") = True
            plan(entry(1)) = value
        End If
    Next
    For Each key In plans.Keys
        attempted = attempted + 1
        Set plan = plans(key)
        Set chosen = rowsByCase(key)
        message = ""
        If Not Cases.Exists(key) Or plan.Exists("__Conflict") Then
            message = "対象不明または同じ項目に複数の訂正があります。"
        Else
            Set record = reviewSnapshots(key)
            For Each entry In record.values.Keys
                If Not plan.Exists(entry) Then plan.Add entry, record.values(entry)
            Next
            If ValidateDetailChoices(plan, record, message) Then
                If SaveCase(record, plan, False, message) Then
                    For Each row In chosen
                        completed.Add row
                    Next
                    good = good + chosen.Count
                    yearsToRefresh(record.sourceYear) = True
                    message = ""
                ElseIf Len(message) = 0 Then
                    message = "保存できませんでした。"
                End If
            End If
        End If
        If Len(message) > 0 Then
            For Each row In chosen
                PutValue reviewPage.Cells(CLng(row), 8), message
            Next
            bad = bad + chosen.Count
        End If
        If PendingSave() Then Exit For
    Next
    For Each key In yearsToRefresh.Keys
        message = ""
        RefreshOneYear CStr(key), message
    Next
    For Each row In completed
        reviewRows.Remove CStr(row)
        With reviewPage.Range("A" & row & ":L" & row)
            .ClearContents
            .Locked = True
            .EntireRow.Hidden = True
        End With
    Next
    SaveReviewCorrections = "更新済み " & good & "項目 / 失敗 " & bad & "項目"
    If attempted < plans.Count Then SaveReviewCorrections = SaveReviewCorrections & " / 未処理 " & (plans.Count - attempted) & "事案"
    If PendingSave() Then SaveReviewCorrections = SaveReviewCorrections & vbCrLf & "未確定の保存があります。オーナーに連絡してください。"
Finished:
    On Error Resume Next
    reviewPage.Protect
    isBusy = False
    Exit Function
Failed:
    SaveReviewCorrections = "更新確認済み " & good & "項目。処理が中断されました。保存を再送せずオーナーに連絡してください。"
    Resume Finished
End Function

Public Sub OpenPendingReport()
    Dim basis As Variant
    Dim filter As Variant
    Dim limit As Variant
    Dim year As Variant
    Dim office As Variant
    Dim status As Variant
    Dim order As Variant
    Dim fieldId As String
    Dim criterion As String
    Dim description As String
    Dim threshold As Date

    If Cases Is Nothing Then
        Notify "先に管理簿を読み込んでください。"
        Exit Sub
    End If
    basis = Application.InputBox("1 申請日 / 2 設定済みの日付項目", "抽出する日付項目", 1, Type:=1)
    If VarType(basis) = vbBoolean Then Exit Sub
    Select Case basis
        Case 1: fieldId = "ApplicationDate"
        Case 2: fieldId = CStr(Sheet("Settings").Range("S3").Value2)
        Case Else: Exit Sub
    End Select
    If Not fields.Exists(fieldId) Then
        Notify "日付項目が登録されていません。設定で日付型の項目IDを指定してください。"
        Exit Sub
    End If

    End If
    filter = Application.InputBox("1 2か月超 / 2 3か月超 / 3 任意月数超 / 4 指定日以前 / 5 日付未入力", "抽出条件", 1, Type:=1)
    If VarType(filter) = vbBoolean Then Exit Sub
    Select Case filter
        Case 1, 2
            threshold = DateAdd("m", -(CLng(filter) + 1), Date)
            criterion = "Before"
            description = (CLng(filter) + 1) & "か月超"
        Case 3
            limit = Application.InputBox("1～1200の整数を入力してください。", "経過月数", Type:=1)
            If VarType(limit) = vbBoolean Then Exit Sub
            If limit < 1 Or limit > 1200 Or limit <> Fix(limit) Then Exit Sub
            threshold = DateAdd("m", -CLng(limit), Date)
            criterion = "Before"
            description = CLng(limit) & "か月超"
        Case 4
            limit = Application.InputBox("例 2026/11/10", "指定日以前", Type:=2)
            If VarType(limit) = vbBoolean Then Exit Sub
            If Not IsDate(limit) Then
                Notify "日付を入力してください。"
                Exit Sub
            End If
            threshold = DateValue(CStr(limit))
            criterion = "Through"
            description = Format$(threshold, "yyyy/mm/dd") & "以前"
        Case 5
            criterion = "Blank"
            description = "日付未入力"
        Case Else
            Exit Sub
    End Select
    year = Application.InputBox("空欄は全年度", "年度", Type:=2)
    If VarType(year) = vbBoolean Then Exit Sub
    office = Application.InputBox("空欄は全署", "署", Type:=2)
    If VarType(office) = vbBoolean Then Exit Sub
    status = Application.InputBox("空欄は全状態", "処理状態", Type:=2)
    If VarType(status) = vbBoolean Then Exit Sub
    order = Application.InputBox("1 古い日付順 / 2 新しい日付順 / 3 経過期間順 / 4 署別 / 5 年度別", "並べ替え", 1, Type:=1)
    If VarType(order) = vbBoolean Then Exit Sub
    If order < 1 Or order > 5 Or order <> Fix(order) Then Exit Sub
    BuildPendingReport Array(fieldId, criterion, threshold, Trim$(CStr(year)), Trim$(CStr(office)), Trim$(CStr(status)), CLng(order), description)
End Sub

Public Function PendingDateMatches(ByVal value As Variant, ByVal criterion As String, ByVal threshold As Date) As Boolean
    If IsError(value) Or IsNull(value) Then Exit Function
    If criterion = "Blank" Then
        PendingDateMatches = Len(CStr(value)) = 0
    ElseIf IsNumeric(value) And VarType(value) <> vbString And Len(CStr(value)) > 0 Then
        Select Case criterion
            Case "Before": PendingDateMatches = (CDbl(value) < CDbl(threshold))
            Case "Through": PendingDateMatches = (CDbl(value) < CDbl(threshold) + 1)
        End Select
    End If
End Function

Public Sub BuildPendingReport(ByVal criteria As Variant)
    Dim book As Workbook
    Dim ws As Worksheet
    Dim entry As CaseRecord
    Dim key As Variant
    Dim closed As Object
    Dim output As Variant
    Dim count As Long
    Dim row As Long
    Dim sortColumn As Long
    Dim sortOrder As XlSortOrder
    Dim fieldId As String
    Dim day As Variant
    Dim label As String

    On Error GoTo Failed
    If Cases Is Nothing Then Exit Sub
    If Cases.Count = 0 Then
        Notify "対象の事案がありません。"
        Exit Sub
    End If
    fieldId = CStr(criteria(0))
    Set closed = ConfiguredList(15)
    ReDim output(1 To Cases.Count, 1 To 8)
    For Each key In Cases.Keys
        Set entry = Cases(key)
        If Not IsCaseClosed(entry.values, closed) And Len(entry.invalidReason) = 0 Then
            If (Len(criteria(3)) = 0 Or entry.sourceYear = criteria(3)) And _
                (Len(criteria(4)) = 0 Or entry.values("Office") = criteria(4)) And _
                (Len(criteria(5)) = 0 Or entry.values("Status") = criteria(5)) Then
                If entry.sourceColumns.Exists(fieldId) Then
                    day = entry.values(fieldId)
                    If PendingDateMatches(day, CStr(criteria(1)), CDate(criteria(2))) Then
                        count = count + 1
                        output(count, 1) = entry.sourceYear
                        output(count, 2) = entry.values("Office")
                        output(count, 3) = entry.PersonKey
                        output(count, 4) = entry.values("InquiryNumber")
                        output(count, 5) = entry.values("PersonName")
                        output(count, 6) = entry.values("DocumentName")
                        output(count, 7) = entry.values("Status")
                        If Len(CStr(day)) > 0 Then output(count, 8) = day
                    End If
                End If
            End If
        End If
    Next
    Set book = Workbooks.Add(xlWBATWorksheet)
    Set ws = book.Worksheets(1)
    Set pendingPage = ws
    ws.Name = "未処理リスト"
    label = IIf(fieldId = "ApplicationDate", "申請日", fieldId)
    ws.Range("A1").Value2 = "未処理リスト"
    ws.Range("A2").Value2 = label & " / " & criteria(7) & " / 年度 " & ReportFilter(criteria(3)) & _
        " / 署 " & ReportFilter(criteria(4)) & " / 状態 " & ReportFilter(criteria(5))
    ws.Range("A3").Value2 = Format$(Date, "yyyy/mm/dd") & " 出力  " & count & "件"
    ws.Range("A6:H6").Value2 = Array("年度", "署", "共通番号", "お問い合わせ番号", "氏名", "文書名", "処理状態", label)
    FormatWorkflowReport ws, "H"
    ws.Columns("C").ColumnWidth = 16
    ws.Columns("F").ColumnWidth = 30
    If count > 0 Then
        ws.Range("A7:G" & (count + 6)).NumberFormat = "@"
        ws.Range("H7:H" & (count + 6)).NumberFormat = "yyyy/mm/dd"
        ws.Range("A7").Resize(count, 8).Value2 = output
        sortColumn = 8
        sortOrder = xlAscending
        Select Case criteria(6)
            Case 2: sortOrder = xlDescending
            Case 4: sortColumn = 2
            Case 5: sortColumn = 1
        End Select
        ws.Range("A6:H" & (count + 6)).Sort Key1:=ws.Cells(7, sortColumn), Order1:=sortOrder, Header:=xlYes
    End If
    AddWorkflowButton ws, "PendingPreview", "印刷プレビュー", "PreviewPendingReport", ws.Range("K2:L3")
    AddWorkflowButton ws, "PendingPrint", "印刷", "PrintPendingReport", ws.Range("K5:L6")
    ApplyPendingPrint ws, count + 6
    Exit Sub
Failed:
    Notify SafeFailure("未処理リスト", Err.Number)
End Sub

    If Len(value) = 0 Then ReportFilter = "すべて"

Public Sub CreateLedgerYear()
    Dim entry As Variant
    Dim name As Variant
    Dim address As String
    Dim message As String
    Dim book As Workbook
    Dim page As Worksheet
    Dim ws As Worksheet
    Dim item As Variant
    Dim col As Long
    On Error GoTo Failed
    If Not CanMoveLedger(message) Then GoTo Finished
    If Not IsOwnerAuthorized() Then
        message = "年度シートの作成にはオーナー認証が必要です。"
        GoTo Finished
    End If
    If PendingSave() Or Len(CStr(Sheet("MoveRecovery").Range("B1").Value2)) > 0 Then
        message = "未確定の保存または移動があります。先に照合してください。"
        GoTo Finished
    End If
    entry = Application.InputBox("作成する年度を西暦4桁で入力してください。", "年度追加", Type:=2)
    If VarType(entry) = vbBoolean Then Exit Sub
    If Not IsDigits(entry, 4) Then Exit Sub
    address = LocateFixedLedger(message, True)
    If Len(address) = 0 Then GoTo Finished
    Set book = OpenLedger(address, False, message)
    If book Is Nothing Then GoTo Finished
    For Each ws In book.Worksheets
        If StrComp(ws.Name, CStr(entry), vbTextCompare) = 0 Then
            message = "同名の年度シートがあります。"
            GoTo Finished
        End If
    Next
    Set page = book.Worksheets.Add(After:=book.Worksheets(book.Worksheets.Count))
    page.Name = CStr(entry)
    For Each item In LedgerColumnDefinitions()
        col = col + 1
        page.Cells(1, col).Value2 = item(1)
        page.Columns(col).NumberFormat = IIf(item(2) = "Date", "yyyy/mm/dd", "@")
    Next
    book.Save
    If Not book.Saved Then GoTo Failed
    book.Close False
    Set book = OpenLedger(address, True, message)
    If book Is Nothing Then GoTo Failed
    Set page = book.Worksheets(CStr(entry))
    If CStr(page.Range("A1").Value2) <> "共通番号" Then GoTo Failed
    message = "年度シートを作成しました。設定ツールで年度・シートを登録してください。"
Finished:
    If Not book Is Nothing Then book.Close False
    If Len(message) > 0 Then Notify message
    Exit Sub
Failed:
    message = "年度作成の結果を確認できません。管理簿を開いて年度シートを確認してから操作してください。"
    Resume Finished
End Sub

Public Sub ArchiveLedgerYear()
    Dim year As Variant
    Dim filename As Variant
    Dim address As String
    Dim message As String
    Dim source As Workbook
    Dim archive As Workbook
    On Error GoTo Failed
    If Not IsOwnerAuthorized() Or Not CanMoveLedger(message) Then
        message = "年度保管にはオーナー認証と常勤フォルダからの起動が必要です。"
        GoTo Finished
    End If

    If Not years.Exists(CStr(year)) Then
        message = "対象年度がありません。"
        GoTo Finished
    End If
    filename = Application.InputBox("保管先の未使用の直接URLを入力してください。ファイル名は「保管_」で始め、.xlsxにしてください。", "年度の保管", Type:=2)
    If VarType(filename) = vbBoolean Then Exit Sub
    If Not IsDirectWorkbookAddress(CStr(filename)) Or Not IsHttpsAddress(CStr(filename)) Then Exit Sub
    If Left$(AddressFileName(CStr(filename)), 3) <> "保管_" Or LCase$(Right$(CStr(filename), 5)) <> ".xlsx" Then Exit Sub
    Dim target As Object
    Set target = FileState(CStr(filename), message)
    If target Is Nothing Then GoTo Finished
    If target("Exists") Then
        message = "保管先がすでに存在します。上書きしません。"
        GoTo Finished
    End If
    address = ResolveYearAddress(CStr(year), message)

    source.Worksheets(years(CStr(year))(1)).Copy Before:=archive.Worksheets(1)
    archive.SaveAs CStr(filename), xlOpenXMLWorkbook
    message = "年度シートを保管しました。正式な更新先は変わりません。"
Finished:
    If Not archive Is Nothing Then archive.Close False
    If Not source Is Nothing Then source.Close False
    If Len(message) > 0 Then Notify message
    Exit Sub
Failed:
    message = "保管結果を確認できません。保存先を開いて確認してください。"
    Resume Finished
End Sub

Public Function IsHttpsAddress(ByVal address As String) As Boolean
    IsHttpsAddress = (StrComp(Left$(address, 8), "https://", vbTextCompare) = 0)
End Function

Public Function CanonicalAddress(ByVal address As String) As String
    Dim slash As Long
    Dim host As String
    Dim path As String
    Dim segment As Variant
    If Not IsHttpsAddress(address) Then
        If Mid$(address, 2, 2) = ":\" Or Left$(address, 2) = "\\" Then CanonicalAddress = Replace(address, "/", "\")
        Exit Function
    End If
    slash = InStr(9, address, "/")
    If slash = 0 Then
        host = Mid$(address, 9)
    Else
        host = Mid$(address, 9, slash - 9)
    End If
    If Len(host) = 0 Or InStr(host, ".") = 0 Or host Like "*[!A-Za-z0-9.-]*" Then Exit Function
    If InStr(address, "?") > 0 Or InStr(address, "\") > 0 Then Exit Function
    If slash = 0 Then
        CanonicalAddress = "https://" & LCase$(host)
        Exit Function
    End If
    path = DecodeAddressPath(Mid$(address, slash))
    If Len(path) = 0 Then Exit Function
    If InStr(path, "\") > 0 Or InStr(path, vbCr) > 0 Or InStr(path, vbLf) > 0 Or InStr(path, Chr$(0)) > 0 Then Exit Function
    If InStr(path, "/:") > 0 Or InStr(path, "//") > 0 Or InStr(1, path, "/_layouts/", vbTextCompare) > 0 Then Exit Function
    For Each segment In Split(path, "/")
        If segment = "." Or segment = ".." Then Exit Function
    Next
    If Right$(path, 1) = "/" Then path = Left$(path, Len(path) - 1)
    CanonicalAddress = "https://" & LCase$(host) & path
End Function

Private Function DecodeAddressPath(ByVal path As String) As String
    Dim i As Long
    Dim octet As Long
    Dim code As Long
    Dim pending As Long
    Dim minimum As Long
    Dim pair As String
    Dim result As String
    i = 1
    Do While i <= Len(path)
        pair = Mid$(path, i + 1, 2)
        If Mid$(path, i, 1) = "%" And Len(pair) = 2 And UCase$(pair) Like "[0-9A-F][0-9A-F]" Then
            octet = CLng("&H" & pair)
            If pending > 0 Then
                If octet < 128 Or octet > 191 Then Exit Function
                code = code * 64 + octet - 128
                pending = pending - 1
                If pending = 0 Then
                    If code < minimum Or code > &H10FFFF Or (code >= &HD800& And code <= &HDFFF&) Then Exit Function
                    If code > 65535 Then
                        code = code - 65536
                        result = result & ChrW(&HD800& + code \ 1024) & ChrW(&HDC00& + code Mod 1024)
                    Else
                        result = result & ChrW(code)
                    End If
                End If
            ElseIf octet < 128 Then
                '区切りをエンコードしたリンクは、別パスに解釈しない。
                If octet = 47 Or octet = 92 Then Exit Function
                result = result & ChrW(octet)
            ElseIf octet >= 194 And octet <= 223 Then
                code = octet - 192
                pending = 1
                minimum = 128
            ElseIf octet >= 224 And octet <= 239 Then
                code = octet - 224
                pending = 2
                minimum = 2048
            ElseIf octet >= 240 And octet <= 244 Then
                code = octet - 240
                pending = 3
                minimum = 65536
            Else
                Exit Function
            End If
            i = i + 3
        Else
            If pending > 0 Then Exit Function
            result = result & Mid$(path, i, 1)
            i = i + 1
        End If
    Loop
    If pending = 0 Then DecodeAddressPath = result
End Function

Public Function IsSameWorkbookAddress(ByVal first As String, ByVal second As String) As Boolean
    Dim leftAddress As String
    Dim rightAddress As String
    leftAddress = CanonicalAddress(first)
    rightAddress = CanonicalAddress(second)
    If Len(leftAddress) = 0 Or Len(rightAddress) = 0 Then Exit Function
    IsSameWorkbookAddress = (StrComp(leftAddress, rightAddress, vbTextCompare) = 0)
End Function

Public Function IsDirectWorkbookAddress(ByVal address As String) As Boolean
    Dim path As String
    Dim extension As String
    path = CanonicalAddress(address)
    If Len(path) = 0 Then Exit Function
    extension = LCase$(Mid$(path, InStrRev(path, ".") + 1))
    IsDirectWorkbookAddress = (extension = "xlsm" Or extension = "xlsx" Or extension = "xlsb" Or extension = "xls")
End Function

Public Function AddressFileName(ByVal address As String) As String
    Dim path As String
    path = Replace(CanonicalAddress(address), "\", "/")
    If Len(path) > 0 Then AddressFileName = Mid$(path, InStrRev(path, "/") + 1)
End Function

Public Function AddressFolder(ByVal address As String) As String
    Dim slash As Long
    slash = InStrRev(address, IIf(IsHttpsAddress(address), "/", "\"))
    If slash > 0 Then AddressFolder = Left$(address, slash - 1)
End Function

Public Function OpenAddressWorkbook(ByVal address As String, ByVal isReadOnly As Boolean, _
    ByVal allowTrustedMacros As Boolean, ByRef message As String) As Workbook
    Dim candidate As Workbook
    Dim book As Workbook
    Dim previousEvents As Boolean
    Dim previousSecurity As Long

    message = ""
    If Not IsDirectWorkbookAddress(address) Then
        message = "Excelファイルの直接URLまたは検証用の絶対パスを指定してください。"
        Exit Function
    End If
    For Each candidate In Application.Workbooks
        If IsSameWorkbookAddress(candidate.FullName, address) Or _
            StrComp(candidate.Name, AddressFileName(address), vbTextCompare) = 0 Then
            message = "対象または同名のブックを閉じてから操作してください。"
            Exit Function
        End If
    Next
    previousEvents = Application.EnableEvents
    previousSecurity = Application.AutomationSecurity
    On Error GoTo Failed
    Application.EnableEvents = False
    If Not allowTrustedMacros Then Application.AutomationSecurity = msoAutomationSecurityForceDisable
    Set book = Application.Workbooks.Open(Filename:=ExcelOpenAddress(address), UpdateLinks:=0, ReadOnly:=isReadOnly, _
        Notify:=False, AddToMru:=False, IgnoreReadOnlyRecommended:=True)
    If Not IsSameWorkbookAddress(book.FullName, address) Then
        message = "Excelが開いたブックのアドレスが設定と一致しません。直接URLを確認してください。"
    ElseIf Not isReadOnly And book.ReadOnly Then
        message = "読み取り専用です。編集権限と他の利用者の使用状況を確認してください。"
    ElseIf book.Date1904 Then
        message = "1904年日付システムのブックは対象外です。"
    Else
        If Not isReadOnly And IsHttpsAddress(address) Then
            'オートセーブが既に無効な環境では、書き込みを試みない。
            If book.AutoSaveOn Then book.AutoSaveOn = False
            If book.AutoSaveOn Then Err.Raise 5, , "自動保存を停止できません。"
        End If
        Set OpenAddressWorkbook = book
        Set book = Nothing
    End If
Finished:
    On Error Resume Next
    If Not book Is Nothing Then book.Close SaveChanges:=False
    Application.AutomationSecurity = previousSecurity
    Application.EnableEvents = previousEvents
    On Error GoTo 0
    Exit Function
Failed:
    message = "Excelで開けません。直接URL、Officeのサインイン、権限、使用状況を確認してください。エラー番号 " & Err.Number
    Resume Finished
End Function

Public Function ExcelOpenAddress(ByVal address As String) As String
    Dim i As Long
    Dim ch As String
    If Not IsHttpsAddress(address) Then
        ExcelOpenAddress = address
        Exit Function
    End If
    For i = 1 To Len(address)
        ch = Mid$(address, i, 1)
        If ch = "#" Then
            ch = "%23"
        ElseIf ch = "%" Then
            If Not UCase$(Mid$(address, i + 1, 2)) Like "[0-9A-F][0-9A-F]" Then ch = "%25"
        End If
        ExcelOpenAddress = ExcelOpenAddress & ch
    Next
End Function
