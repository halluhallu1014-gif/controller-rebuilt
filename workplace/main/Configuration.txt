Option Explicit
Option Private Module

Public Function Setting(ByVal settingName As String) As String
    Dim rowIndex As Long
    For rowIndex = 2 To LastRow(Sheet("Settings"))
        If CStr(Sheet("Settings").Cells(rowIndex, 1).Value2) = settingName Then
            Setting = CStr(Sheet("Settings").Cells(rowIndex, 2).Value2)
            Exit Function
        End If
    Next
End Function

Public Function LoadConfiguration(ByRef message As String) As Boolean
    Dim fieldDefinitions As Object
    Dim yearDefinitions As Object
    Dim ws As Worksheet
    Dim rowIndex As Long
    Dim fieldId As String
    Dim definition As Variant
    Dim target As Range
    Dim usedCells As Object
    Dim year As String
    Dim address As String
    Dim id As Variant
    Dim yearProblem As String
    Dim previous As Variant
    Dim hasYearSettings As Boolean
    On Error GoTo Failed
    If ThisWorkbook.Date1904 Then
        message = "CONTROLLERが1904年日付システムです。日付のずれを防ぐため処理を中止します。標準の1900年日付システムの別ブックで構築してください。"
        Exit Function
    End If
    Set fieldDefinitions = NewDictionary()
    Set yearDefinitions = NewDictionary()
    Set usedCells = NewDictionary()
    Set ws = Sheet("Fields")
    For rowIndex = 2 To LastRow(ws)
        fieldId = CStr(ws.Cells(rowIndex, 1).Value2)
        If Len(fieldId) > 0 Then
            message = "Fields 行" & rowIndex & "：項目ID重複、型、Header、Sheet、Cellを確認してください。"
            If fieldDefinitions.Exists(fieldId) Then Exit Function
            definition = Array(CStr(ws.Cells(rowIndex, 2).Value2), CStr(ws.Cells(rowIndex, 3).Value2), _
                CStr(ws.Cells(rowIndex, 4).Value2), CStr(ws.Cells(rowIndex, 5).Value2), CStr(ws.Cells(rowIndex, 6).Value2), CStr(ws.Cells(rowIndex, 11).Value2))
            If Len(definition(0)) = 0 Then Exit Function
            If definition(1) <> "Text" And definition(1) <> "Date" Then Exit Function
            If definition(4) <> "Yes" And definition(4) <> "No" Then Exit Function
            Set target = Sheet(definition(2)).Range(definition(3))
            If target.CountLarge <> 1 Then Exit Function
            If target.MergeCells Then
                If target.Address <> target.MergeArea.Cells(1, 1).Address Then Exit Function
            End If
            If target.HasFormula Then Exit Function
            If definition(2) = "Settings" Or definition(2) = "Fields" Or definition(2) = "Recovery" Or definition(2) = "MoveRecovery" Or definition(2) = "Calendar" Then Exit Function
            If IsReservedUiCell(target) Then
                message = "Fields 行" & rowIndex & "：" & fieldId & "の入力セルが検索・関連事案一覧・操作欄と重複しています。別のセルを設定してください。"
                Exit Function
            End If
            If usedCells.Exists(target.Parent.Name & "!" & target.Address) Then Exit Function
            usedCells.Add target.Parent.Name & "!" & target.Address, True
            fieldDefinitions.Add fieldId, definition
        End If
    Next
    For Each id In Array("CommonNumber", "PersonName")
        message = "Fields：必須定義がありません。" & CStr(id)
        If Not fieldDefinitions.Exists(id) Then Exit Function
    Next
    If fieldDefinitions("CommonNumber")(1) <> "Text" Then Exit Function
    Set ws = Sheet("Settings")
    For rowIndex = 2 To LastRow(ws)
        year = CStr(ws.Cells(rowIndex, 4).Value2)
        If Len(year) > 0 Then
            hasYearSettings = True
            If Not IsDigits(year, 4) Then Exit Function
            message = "Settings 年度表 行" & rowIndex & "：年度重複、Enabled、直接アドレス、シート名を確認してください。"
            If ws.Cells(rowIndex, 7).Value2 <> "No" Then
                yearProblem = ""
                If CStr(ws.Cells(rowIndex, 7).Value2) <> "Yes" Then yearProblem = "EnabledはYesまたはNoにしてください。"
                address = Setting("StaffLedger")
                If Len(CStr(ws.Cells(rowIndex, 6).Value2)) = 0 Then yearProblem = "年度シート名を設定してください。"
                If ws.Cells(rowIndex, 8).Value2 <> "Yes" And ws.Cells(rowIndex, 8).Value2 <> "No" Then yearProblem = "AllowWriteはYesまたはNoにしてください。"
                If yearDefinitions.Exists(year) Then
                    previous = yearDefinitions(year)
                    previous(5) = "年度が重複しています。Settingsの年度表を確認してください。"
                    yearDefinitions(year) = previous
                Else
                    yearDefinitions.Add year, Array(address, CStr(ws.Cells(rowIndex, 6).Value2), CStr(ws.Cells(rowIndex, 8).Value2), address, Setting("GeneralLedger"), yearProblem)
                End If
            End If
        End If
    Next

    Set fields = fieldDefinitions
    If Not hasYearSettings Then
        If Not DiscoverYears(yearDefinitions, message) Then Exit Function
    End If
    If yearDefinitions.Count = 0 Then
        message = "有効な管理簿の年度がありません。設定ツールで対象年度を確認してください。"
        Exit Function
    End If
    message = ""
    If Not ValidatePresentation(message) Then Exit Function
    Set fields = fieldDefinitions
    Set years = yearDefinitions
    LoadConfiguration = True
    Exit Function
Failed:
    message = message & " 定義したセルが存在し、単一セルまたは結合セルの左上で、数式なしであることを確認してください。"
End Function


Private Function DiscoverYears(ByVal definitions As Object, ByRef message As String) As Boolean
    Dim address As String
    Dim book As Workbook
    Dim ws As Worksheet
    Dim year As String
    Dim columns As Object
    Dim issue As String
    On Error GoTo Failed
    address = LocateFixedLedger(message)
    If Len(address) = 0 Then Exit Function
    Set book = OpenLedger(address, True, message)
    If book Is Nothing Then Exit Function
    For Each ws In book.Worksheets
        year = YearFromSheet(ws.Name)
        If Len(year) > 0 Then
            Set columns = ColumnMap(ws, year, issue)
            If Not columns Is Nothing Then
                If definitions.Exists(year) Then
                    message = year & "年度の管理簿シートが複数あります。設定ツールで使用するシートを指定してください。"
                    GoTo Finished
                End If
                definitions.Add year, Array(address, ws.Name, "Yes", address, "", "")
            ElseIf Len(message) = 0 Then
                message = ws.Name & "：" & issue
            End If
        End If
    Next
    DiscoverYears = definitions.Count > 0
    If DiscoverYears Then
        message = ""
    ElseIf Len(message) = 0 Then
        message = "管理簿に年度を特定できるシートがありません。設定ツールで使用するシートを指定してください。"
    End If
Finished:
    book.Close False
    Exit Function
Failed:
    message = "管理簿の年度と見出しを確認できませんでした。"
    If Not book Is Nothing Then book.Close False
End Function

Private Function YearFromSheet(ByVal name As String) As String
    Dim value As Long
    name = Trim$(name)
    If Right$(name, 2) = "年度" Then name = Left$(name, Len(name) - 2)
    If IsDigits(name, 4) Then
        value = CLng(name)
    ElseIf name Like "令和#" Or name Like "令和##" Then
        value = 2018 + CLng(Mid$(name, 3))
    ElseIf name = "令和元" Then
        value = 2019
    End If
    If value >= 1900 And value <= 2099 Then YearFromSheet = CStr(value)
End Function

Public Function FieldCell(ByVal fieldId As String) As Range
    Dim definition As Variant
    definition = fields(fieldId)
    Set FieldCell = Sheet(definition(2)).Range(definition(3))
End Function

Public Function HeaderFor(ByVal fieldId As String, ByVal year As String, ByRef message As String) As String
    Dim rowIndex As Long
    Dim count As Long
    Dim ws As Worksheet
    HeaderFor = fields(fieldId)(0)
    Set ws = Sheet("Fields")
    For rowIndex = 2 To LastRow(ws)
        If CStr(ws.Cells(rowIndex, 8).Value2) = year And CStr(ws.Cells(rowIndex, 9).Value2) = fieldId Then
            count = count + 1
            HeaderFor = CStr(ws.Cells(rowIndex, 10).Value2)
        End If
    Next
    If count > 1 Then message = year & "年度：Fieldsの年度別見出しが重複しています。" & fieldId
End Function

Public Function ColumnMap(ByVal ws As Worksheet, ByVal year As String, ByRef message As String) As Object
    Dim headers As Object
    Dim result As Object
    Dim used As Object
    Dim id As Variant
    Dim heading As String
    Dim value As Variant
    Dim position As Long
    Dim last As Long
    Set headers = NewDictionary()
    Set result = NewDictionary()
    Set used = NewDictionary()
    message = ""
    last = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For position = 1 To last
        value = ws.Cells(1, position).Value2
        If IsError(value) Then
            message = "管理簿の1行目にExcelエラーがあります。"
            Exit Function
        End If
        heading = Trim$(Replace(CStr(value), ChrW(&H3000), " "))
        If Len(heading) > 0 Then
            If headers.Exists(heading) Then
                headers(heading) = 0
            Else
                headers.Add heading, position
            End If
        End If
    Next
    For Each id In fields.Keys
        heading = Trim$(Replace(HeaderFor(CStr(id), year, message), ChrW(&H3000), " "))
        If Len(message) > 0 Then Exit Function
        If headers.Exists(heading) Then
            position = CLng(headers(heading))
            If position = 0 Or used.Exists(heading) Then
                message = "管理簿の見出し「" & heading & "」が重複しています。"
                Exit Function
            End If
            used.Add heading, True
            result.Add id, position
        ElseIf fields(id)(4) = "Yes" Then
            message = "管理簿の1行目に必須の見出し「" & heading & "」がありません。"
            Exit Function
        End If
    Next
    If headers.Exists("照合番号") Then
        If CLng(headers("照合番号")) > 0 Then result.Add "ComparisonNumber", CLng(headers("照合番号"))
    End If
    Set ColumnMap = result
End Function

Public Function ValidateValues(ByVal values As Object, ByRef message As String) As Boolean
    Dim id As Variant
    Dim value As Variant
    Dim number As String
    Dim normalized As Variant
    For Each id In fields.Keys
        message = "入力項目 " & CStr(id) & "："
        If Not values.Exists(id) Then
            message = message & "入力がありません。"
            Exit Function
        End If
        value = values(id)
        If IsError(value) Or IsNull(value) Then
            message = message & "Excelエラーは保存できません。"
            Exit Function
        End If
        If fields(id)(4) = "Yes" And Len(CStr(value)) = 0 Then
            message = message & "必須項目です。"
            Exit Function
        End If
        If Not NormalizeValue(CStr(id), value, normalized) Then
            message = message & "形式が不正です。日付はExcel日付、郵便番号は7桁、電話番号は数字で入力してください。"
            Exit Function
        End If
        values(id) = normalized
    Next
    message = "共通番号は18桁の文字列で入力してください。"
    If Not IsDigits(values("CommonNumber"), 18) Then Exit Function
    message = ""
    ValidateValues = True
End Function

Public Function IsClosedState(ByVal values As Object, ByVal closedStates As Object) As Boolean
    Dim status As String
    If Not values.Exists("Status") Then Exit Function
    If IsError(values("Status")) Or IsNull(values("Status")) Then Exit Function
    status = Trim$(CStr(values("Status")))
    IsClosedState = status = "終了【終】" Or IsTransferStatus(status) Or status = "終了" Or status = "他グループ・署へ移管" Or closedStates.Exists(status)
End Function

Public Function HasCompletionValue(ByVal values As Object, ByVal fieldId As String) As Boolean
    If Not values.Exists(fieldId) Then Exit Function
    If IsError(values(fieldId)) Or IsNull(values(fieldId)) Then Exit Function
    HasCompletionValue = Len(CStr(values(fieldId))) > 0
End Function

Public Function IsCaseClosed(ByVal values As Object, ByVal closedStates As Object) As Boolean
    IsCaseClosed = HasCompletionValue(values, "CompletedDate") Or IsClosedState(values, closedStates)
End Function

Public Function HasCompletionMismatch(ByVal values As Object, ByVal closedStates As Object) As Boolean
    If HasCompletionValue(values, "CompletedDate") Then
        HasCompletionMismatch = Not IsClosedState(values, closedStates)
    Else
        HasCompletionMismatch = HasCompletionValue(values, "IssuedDate") Or IsClosedState(values, closedStates)
    End If
End Function

Public Function IsTransferStatus(ByVal status As String) As Boolean
    IsTransferStatus = (status = "署へ移行【終】" Or status = "他グループへ移行【終】")
End Function

Public Sub ApplyWorkflowStatus(ByVal values As Object, ByVal previousStatus As String)
    Dim status As String
    status = CStr(values("Status"))
    If IsTransferStatus(status) Then Exit Sub
    If HasCompletionValue(values, "IssuedDate") Then
        values("Status") = "終了【終】"
        If Not HasCompletionValue(values, "CompletedDate") Then values("CompletedDate") = values("IssuedDate")
    ElseIf HasCompletionValue(values, "ApprovedDate") Then
        values("Status") = "決裁済"
    ElseIf HasCompletionValue(values, "DraftDate") Then
        values("Status") = "起案済"
    End If
End Sub

Public Function ValidateCompletion(ByVal values As Object, ByVal columns As Object, ByRef message As String) As Boolean
    If Not HasCompletionValue(values, "CompletedDate") Then
        If HasCompletionMismatch(values, ConfiguredList(15)) Then
            If columns.Exists("CompletedDate") Then
                message = "完了日が入力されていません。完了日を入力してから、再度保存してください。"
            Else
                message = "この年度の管理簿に完了日欄がありません。完了日列のある管理簿を設定してから保存してください。"
            End If
            Exit Function
        End If
    End If
    ValidateCompletion = True
End Function

Public Function ConfiguredList(ByVal column As Long) As Object
    Dim result As Object
    Dim ws As Worksheet
    Dim rowIndex As Long
    Dim value As String
    Set result = NewDictionary()
    result.CompareMode = vbTextCompare
    Set ws = Sheet("Settings")
    For rowIndex = 2 To ws.Cells(ws.Rows.Count, column).End(xlUp).Row
        If Not IsError(ws.Cells(rowIndex, column).Value2) Then
            value = Trim$(CStr(ws.Cells(rowIndex, column).Value2))
            If Len(value) > 0 Then result(value) = True
        End If
    Next
    Set ConfiguredList = result
End Function
