Option Explicit
Option Private Module

Public Function IsWebAddress(ByVal address As String) As Boolean
    IsWebAddress = IsHttpsAddress(address)
End Function

Public Function OpenLedger(ByVal address As String, ByVal readOnly As Boolean, ByRef message As String) As Workbook
    Set OpenLedger = OpenAddressWorkbook(address, readOnly, False, message)
End Function

Public Function ReadYear(ByVal year As String, ByRef message As String, Optional ByVal knownAddress As String = "") As Object
    Dim definition As Variant
    Dim book As Workbook
    Dim ws As Worksheet
    Dim columns As Object
    Dim result As Object
    Dim data As Variant
    Dim formulas As Variant
    Dim rowIndex As Long
    Dim record As CaseRecord
    Dim key As String
    On Error GoTo Failed
    definition = years(year)
    If Len(CStr(definition(5))) > 0 Then
        message = CStr(definition(5))
        Exit Function
    End If
    If Len(knownAddress) = 0 Then knownAddress = ResolveYearAddress(year, message)
    If Len(knownAddress) = 0 Then Exit Function
    definition(0) = knownAddress
    years(year) = definition
    Set book = OpenLedger(definition(0), True, message)
    If book Is Nothing Then Exit Function
    definition(0) = book.FullName
    years(year) = definition
    Set ws = LedgerSheet(book, CStr(definition(1)), message)
    If ws Is Nothing Then GoTo Rejected
    Set columns = ColumnMap(ws, year, message)
    If columns Is Nothing Then
        message = book.Name & " / " & ws.Name & "：" & message
        GoTo Rejected
    End If
    data = ws.Range(ws.Cells(1, 1), ws.Cells(LastRow(ws), LastColumn(ws))).Value2
    formulas = ws.Range(ws.Cells(1, 1), ws.Cells(LastRow(ws), LastColumn(ws))).Formula
    Set result = NewDictionary()
    For rowIndex = 2 To UBound(data, 1)
        If HasRecordData(data, rowIndex, columns) Then
            Set record = ReadRecord(data, formulas, rowIndex, columns, year, definition)
            key = record.Key
            If result.Exists(key) Then
                result(key).invalidReason = "同一年度の識別キー重複。更新禁止。"
                record.invalidReason = "同一年度の識別キー重複。更新禁止。"
                key = year & "|DuplicateRow|" & rowIndex
            End If
            result.Add key, record
        End If
    Next
    book.Close SaveChanges:=False
    If yearColumnMaps Is Nothing Then Set yearColumnMaps = NewDictionary()
    Set yearColumnMaps(year) = columns
    Set ReadYear = result
    Exit Function
Rejected:
    book.Close SaveChanges:=False
    Exit Function
Failed:
    message = year & "年度の読込を中止しました。シート名、見出し、セル型を確認してください。詳細：" & Err.Description
    On Error Resume Next
    If Not book Is Nothing Then book.Close SaveChanges:=False
    On Error GoTo 0
End Function

Private Function HasRecordData(ByRef data As Variant, ByVal rowIndex As Long, ByVal columns As Object) As Boolean
    Dim id As Variant
    For Each id In columns.Keys
        If IsError(data(rowIndex, columns(id))) Then
            HasRecordData = True
            Exit Function
        End If
        If Len(CStr(data(rowIndex, columns(id)))) > 0 Then
            HasRecordData = True
            Exit Function
        End If
    Next
End Function

Private Function ReadRecord(ByRef data As Variant, ByRef formulas As Variant, ByVal rowIndex As Long, _
    ByVal columns As Object, ByVal year As String, ByVal definition As Variant) As CaseRecord
    Dim record As New CaseRecord
    Dim id As Variant
    Dim colIndex As Long
    Dim original() As Variant
    Dim normalized As Variant
    record.sourceYear = year
    record.sourceAddress = definition(0)
    record.sourceSheet = definition(1)
    record.sourceRow = rowIndex
    Set record.sourceColumns = columns
    ReDim original(1 To 1, 1 To UBound(data, 2))
    For colIndex = 1 To UBound(data, 2)
        original(1, colIndex) = formulas(rowIndex, colIndex)
    Next
    record.originalRow = original
    If columns.Exists("ComparisonNumber") Then
        If Not IsError(data(rowIndex, columns("ComparisonNumber"))) Then record.comparisonNumber = CStr(data(rowIndex, columns("ComparisonNumber")))
    End If
    For Each id In fields.Keys
        record.values.Add id, ""
        If columns.Exists(id) Then
            If IsError(data(rowIndex, columns(id))) Then
                record.invalidReason = "行" & rowIndex & "：" & CStr(id) & "がExcelエラーです。更新禁止。"
            Else
                record.values(id) = data(rowIndex, columns(id))
                If Not NormalizeValue(CStr(id), record.values(id), normalized) Then
                    record.invalidReason = "行" & rowIndex & "：" & CStr(id) & "の保存型が不正です。更新禁止。"
                End If
            End If
        End If
    Next
    If Len(record.CaseIdentity) = 0 Then record.invalidReason = "行" & rowIndex & "：18桁の共通番号の形式が不正です。更新禁止。"
    Set ReadRecord = record
End Function

Public Function LedgerSheet(ByVal book As Workbook, ByVal sheetName As String, ByRef message As String) As Worksheet
    Dim candidate As Worksheet
    Dim available As String
    For Each candidate In book.Worksheets
        If StrComp(candidate.Name, sheetName, vbTextCompare) = 0 Then
            Set LedgerSheet = candidate
            Exit Function
        End If
        If Len(available) > 0 Then available = available & "、"
        available = available & candidate.Name
    Next
    message = book.Name & "：設定したシート「" & sheetName & "」がありません。利用可能なシート：" & available
End Function

Private Function LocateRow(ByVal ws As Worksheet, ByVal columns As Object, ByVal caseIdentity As String, _
    ByVal isNew As Boolean, ByRef message As String) As Long
    Dim row As Long
    Dim last As Long
    Dim value As Variant
    Dim col As Long
    col = columns("CommonNumber")
    last = ws.Cells(ws.Rows.Count, col).End(xlUp).Row
    For row = 2 To last
        value = ws.Cells(row, col).Value2
        If IsDigits(value, 18) Then
            If CStr(value) = caseIdentity Then
                If LocateRow <> 0 Then
                    message = "同一年度の共通番号が重複しています。書き込みを中止します。"
                    LocateRow = 0
                    Exit Function
                End If
                LocateRow = row
            End If
        End If
    Next
End Function

Public Function SaveCase(ByVal record As CaseRecord, ByVal values As Object, ByVal isNew As Boolean, ByRef message As String) As Boolean
    Dim book As Workbook
    Dim ws As Worksheet
    Dim columns As Object
    Dim definition As Variant
    Dim rowIndex As Long
    Dim id As Variant
    Dim changed As Object
    Dim number As String
    Dim personKey As String
    Dim writeStarted As Boolean
    Dim events As Boolean
    Dim normalizedOriginal As Variant
    Dim offices As Object
    events = Application.EnableEvents
    On Error GoTo Failed
    If PendingSave() Then
        message = "前回の保存結果が未確認です。再送せずオーナーに連絡してください。"
        Exit Function
    End If
    If Len(CStr(Sheet("MoveRecovery").Range("B1").Value2)) > 0 Then
        message = "管理簿移動の結果が未確認です。常勤の管理担当が移動結果を確認するまで保存を中止します。"
        Exit Function
    End If
    If Not LoadConfiguration(message) Then Exit Function
    If Not ValidateValues(values, message) Then Exit Function
    If isNew Then
        ApplyWorkflowStatus values, ""
    Else
        ApplyWorkflowStatus values, CStr(record.values("Status"))
    End If
    If Not years.Exists(record.sourceYear) Then
        message = "有効な保存先年度がありません。Settingsを確認してください。保存は実行していません。"
        Exit Function
    End If
    If yearErrors.Exists(record.sourceYear) Then
        message = "この年度は読込に失敗しています。設定を修正し、最新を再読込してください。保存は実行していません。"
        Exit Function
    End If
    definition = years(record.sourceYear)
    If definition(2) <> "Yes" Then
        message = "この年度のAllowWriteはYesではありません。保存は実行していません。"
        Exit Function
    End If
    If ControllerFolderRole(ThisWorkbook.FullName) = "" And Not IsLocalDevelopment() And Not IsLocalSingleLedger() Then
        message = "CONTROLLERの起動フォルダを特定できません。"
        Exit Function
    End If
    definition(0) = ResolveYearAddress(record.sourceYear, message, True)
    If Len(message) > 0 Then Exit Function
    years(record.sourceYear) = definition
    If Not isNew Then
        If Len(record.invalidReason) > 0 Then
            message = record.invalidReason
            Exit Function
        End If
        If Not IsSameWorkbookAddress(record.sourceAddress, CStr(definition(0))) Or record.sourceSheet <> definition(1) Then
            message = "保存先設定が読込時から変わっています。最新を再読込してください。保存は実行していません。"
            Exit Function
        End If
        If Not SameValue(record.values("CommonNumber"), values("CommonNumber")) Then
            message = "既存事案の共通番号は通常更新では変更できません。保存は実行していません。"
            Exit Function
        End If
    End If
    personKey = CStr(values("CommonNumber"))
    Set book = OpenLedger(definition(0), False, message)
    If book Is Nothing Then Exit Function
    definition(0) = book.FullName
    years(record.sourceYear) = definition
    Set ws = LedgerSheet(book, CStr(definition(1)), message)
    If ws Is Nothing Then GoTo Rejected
    Set columns = ColumnMap(ws, record.sourceYear, message)
    If columns Is Nothing Then GoTo Rejected
    If Not ValidateCompletion(values, columns, message) Then GoTo Rejected
    rowIndex = LocateRow(ws, columns, personKey, isNew, message)
    If Len(message) > 0 Then GoTo Rejected
    If isNew Then
        If rowIndex <> 0 Then
            message = "同一年度に同じ識別キーが存在します。新規登録を中止しました。"
            GoTo Rejected
        End If
        rowIndex = LastRow(ws) + 1
    Else
        If rowIndex = 0 Then
            message = "対象事案が管理簿にありません。最新を再読込してください。"
            GoTo Rejected
        End If
    End If
    Set changed = NewDictionary()
    For Each id In fields.Keys
        If Not columns.Exists(id) Then
            If Len(CStr(values(id))) > 0 Then
                message = "この年度に存在しない項目へ入力されています。" & CStr(id) & "。保存は実行していません。"
                GoTo Rejected
            End If
        ElseIf isNew Then
            changed.Add id, values(id)
        Else
            If NormalizeValue(CStr(id), record.values(id), normalizedOriginal) Then
                If SameValue(values(id), normalizedOriginal) Then values(id) = record.values(id)
            End If
            If Not SameValue(values(id), record.values(id)) Then changed.Add id, values(id)
        End If
    Next
    If changed.Exists("Office") And Len(CStr(values("Office"))) > 0 And Not IsLocalSingleLedger() And Not IsLocalDevelopment() Then
        Set offices = ConfiguredList(14)
        If offices.Count > 0 And Not offices.Exists(CStr(values("Office"))) Then
            message = "署はSettingsの署一覧から選んでください。"
            GoTo Rejected
        End If
    End If
    If changed.Count = 0 Then
        message = "変更はありません。書込みは実行していません。"
        GoTo Rejected
    End If
    For Each id In changed.Keys
        If ws.Cells(rowIndex, columns(id)).HasFormula Then
            message = "変更対象の" & CStr(id) & "に数式があります。数式を保護して保存を中止しました。"
            GoTo Rejected
        End If
    Next
    If isNew Then
        If LocateRow(ws, columns, personKey, True, message) <> 0 Or Len(message) > 0 Or rowIndex <> LastRow(ws) + 1 Then
            message = "登録位置が変わりました。最新を再読込してください。"
            GoTo Rejected
        End If
    Else
        If LocateRow(ws, columns, personKey, False, message) <> rowIndex Or Len(message) > 0 Then
            message = "対象行が移動しました。最新を再読込してください。"
            GoTo Rejected
        End If
        For Each id In changed.Keys
            If Not record.sourceColumns.Exists(id) Then
                message = "見出し対応が変わりました。最新を再読込してください。"
                GoTo Rejected
            End If
            If Not SameValue(ws.Cells(rowIndex, columns(id)).Formula, record.originalRow(1, record.sourceColumns(id))) Then
                message = "「" & fields(id)(0) & "」が管理簿で変更されています。入力は保持しています。最新の値を確認してください。"
                GoTo Rejected
            End If
        Next
    End If
    If Not PrepareJournal(record, definition, values, changed, message) Then GoTo Rejected
    Application.EnableEvents = False
    writeStarted = True
    '非管理列・数式を保護するため、登録時も定義済み列だけに書く。
    For Each id In changed.Keys
        PutValue ws.Cells(rowIndex, columns(id)), changed(id), (fields(id)(1) = "Date")
    Next
    book.Save
    If Not book.Saved Then Err.Raise vbObjectError + 100, "Storage", "Save not confirmed"
    book.Close SaveChanges:=False
    Set book = Nothing
    Application.EnableEvents = events
    SaveCase = VerifyRecordedSave(message)
    Exit Function
Rejected:
    book.Close SaveChanges:=False
    Exit Function
Failed:
    If writeStarted Then
        message = "保存結果不明です。入力とRecoveryを保持しています。再保存せずオーナーに連絡してください。"
    Else
        message = SafeFailure(record.sourceYear & "年度の保存準備", Err.Number) & " 管理簿への書込みは開始していません。"
    End If
    On Error Resume Next
    If Not book Is Nothing Then book.Close SaveChanges:=False
    Application.EnableEvents = events
    On Error GoTo 0
End Function

Private Function PrepareJournal(ByVal record As CaseRecord, ByVal definition As Variant, ByVal values As Object, _
    ByVal changed As Object, ByRef message As String) As Boolean
    Dim ws As Worksheet
    Dim id As Variant
    Dim rowIndex As Long
    On Error GoTo Failed
    If ThisWorkbook.ReadOnly Or Len(ThisWorkbook.Path) = 0 Or ThisWorkbook.FileFormat <> 52 Then
        message = "CONTROLLERを保存可能なxlsmとして保存してください。管理簿への書込みは実行していません。"
        Exit Function
    End If
    Set ws = Sheet("Recovery")
    ws.Cells.ClearContents
    PutValue ws.Range("B1"), "結果確認待ち"
    PutValue ws.Range("B2"), record.sourceYear
    PutValue ws.Range("B3"), definition(0)
    PutValue ws.Range("B4"), definition(1)
    PutValue ws.Range("B5"), IIf(record.sourceRow = 0, "登録", "更新")
    PutValue ws.Range("D4"), CStr(values("CommonNumber"))
    PutValue ws.Range("D5"), NewTransactionId()
    rowIndex = 6
    For Each id In values.Keys
        PutValue ws.Cells(rowIndex, 1), CStr(id)
        PutValue ws.Cells(rowIndex, 2), values(id), (fields(id)(1) = "Date")
        If changed.Exists(id) Then
            PutValue ws.Cells(rowIndex, 3), record.values(id), (fields(id)(1) = "Date")
            PutValue ws.Cells(rowIndex, 4), "Yes"
        End If
        rowIndex = rowIndex + 1
    Next
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Failed
    PrepareJournal = True
    Exit Function
Failed:
    message = "保存予定内容をCONTROLLER内へ退避できません。管理簿への書込みは実行していません。Recoveryが残る場合は確認してください。"
End Function

Public Function ReconcileSave(ByRef message As String) As Boolean
    If Not IsOwnerAuthorized() Then
        message = "保存結果はオーナーに確認を依頼してください。"
        Exit Function
    End If
    HideOwnerWorkspace
    ReconcileSave = VerifyRecordedSave(message)
End Function

Private Function VerifyRecordedSave(ByRef message As String) As Boolean
    Dim recovery As Worksheet
    Dim book As Workbook
    Dim ws As Worksheet
    Dim columns As Object
    Dim expected As Object
    Dim rowIndex As Long
    Dim index As Long
    Dim number As String
    Dim year As String
    Dim id As Variant
    Dim definition As Variant
    On Error GoTo Failed
    If Not PendingSave() Then
        message = "照合待ちの保存はありません。"
        Exit Function
    End If
    Set recovery = Sheet("Recovery")
    year = CStr(recovery.Range("B2").Value2)
    If Not years.Exists(year) Then GoTo Unconfirmed
    definition = years(year)
    definition(0) = ResolveYearAddress(year, message)
    If Len(message) > 0 Then GoTo Unconfirmed
    years(year) = definition
    If Not IsSameWorkbookAddress(CStr(definition(0)), CStr(recovery.Range("B3").Value2)) Or definition(1) <> CStr(recovery.Range("B4").Value2) Then GoTo Unconfirmed
    Set expected = NewDictionary()
    For index = 6 To LastRow(recovery)
        id = CStr(recovery.Cells(index, 1).Value2)
        If id = "CommonNumber" Or recovery.Cells(index, 4).Value2 = "Yes" Then expected.Add id, recovery.Cells(index, 2).Value2
    Next
    If Not IsDigits(expected("CommonNumber"), 18) Then GoTo Unconfirmed
    Set book = OpenLedger(definition(0), True, message)
    If book Is Nothing Then GoTo Unconfirmed
    Set ws = LedgerSheet(book, CStr(definition(1)), message)
    If ws Is Nothing Then GoTo Unconfirmed
    Set columns = ColumnMap(ws, year, message)
    If columns Is Nothing Then GoTo Unconfirmed
    rowIndex = LocateRow(ws, columns, CStr(expected("CommonNumber")), False, message)
    If rowIndex = 0 Or Len(message) > 0 Then GoTo Unconfirmed
    For Each id In expected.Keys
        If columns.Exists(id) Then
            If Not SameValue(ws.Cells(rowIndex, columns(id)).Value2, expected(id)) Then GoTo Unconfirmed
        ElseIf Len(CStr(expected(id))) > 0 Then
            GoTo Unconfirmed
        End If
    Next
    book.Close SaveChanges:=False
    Set book = Nothing
    WriteHistory recovery
    recovery.Range("B1").ClearContents
    ThisWorkbook.Save
    If Not ThisWorkbook.Saved Then GoTo Unconfirmed
    message = "保存確認済み：管理簿を開き直して保存対象の項目との一致を確認しました。"
    VerifyRecordedSave = True
    Exit Function
Unconfirmed:
    If Not book Is Nothing Then book.Close SaveChanges:=False
    PutValue recovery.Range("B1"), "結果確認待ち"
    message = "保存結果を確認できません。Recoveryと入力を保持し、再送を禁止しています。管理担当が管理簿と保存予定内容を照合してください。" & vbCrLf & message
    Exit Function
Failed:
    On Error Resume Next
    If Not book Is Nothing Then book.Close SaveChanges:=False
    PutValue Sheet("Recovery").Range("B1"), "結果確認待ち"
    On Error GoTo 0
    message = "保存結果照合に失敗しました。再保存せず管理担当へ確認してください。Recoveryと入力は保持しています。"
End Function
