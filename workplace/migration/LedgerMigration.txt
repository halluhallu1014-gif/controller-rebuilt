Option Explicit

Public Sub MigrateLedger()
    Dim source As Variant
    Dim target As Variant
    Dim tabName As Variant
    Dim year As Variant
    Dim message As String
    source = Application.InputBox("旧管理簿の直接URLまたは検証用の絶対パスを入力してください。", "旧管理簿の移行", Type:=2)
    If VarType(source) = vbBoolean Then Exit Sub
    tabName = Application.InputBox("旧管理簿のシート名を入力してください。", "旧管理簿の移行", Type:=2)
    If VarType(tabName) = vbBoolean Then Exit Sub
    target = Application.InputBox("あらかじめ保存した移行先の管理簿.xlsmの直接URLを入力してください。", "旧管理簿の移行", Type:=2)
    If VarType(target) = vbBoolean Then Exit Sub
    year = Application.InputBox("移行先の年度シート名を西暦4桁で入力してください。", "旧管理簿の移行", Type:=2)
    If VarType(year) = vbBoolean Then Exit Sub
    If MsgBox("管理簿.xlsmに年度シート " & year & " を追加しますか？", vbQuestion Or vbYesNo Or vbDefaultButton2) <> vbYes Then Exit Sub
    MigrateLedgerYear CStr(source), CStr(tabName), CStr(target), CStr(year), message
    MsgBox message, vbInformation, "旧管理簿の移行"
End Sub

Public Function MigrateLedgerYear(ByVal source As String, ByVal sourceSheet As String, ByVal target As String, _
    ByVal year As String, ByRef message As String) As Boolean
    Dim oldBook As Workbook
    Dim newBook As Workbook
    Dim page As Worksheet
    Dim existing As Worksheet
    Dim oldValues As Variant
    Dim output As Variant
    Dim headings As Variant
    Dim headers As Object
    Dim used As Object
    Dim seen As Object
    Dim mapping(0 To 23) As Long
    Dim header As Variant
    Dim masked As Long
    Dim branch As Long
    Dim number As String
    Dim label As String
    Dim missing As String
    Dim lastCol As Long
    Dim lastRow As Long
    Dim last As Range
    Dim row As Long
    Dim col As Long
    Dim offset As Long
    Dim count As Long
    Dim expected As Variant
    Dim actual As Variant
    Dim saveStarted As Boolean
    Dim previousEvents As Boolean
    previousEvents = Application.EnableEvents
    On Error GoTo Failed
    message = "移行先の管理簿.xlsmと、西暦4桁の新しい年度を指定してください。"
    If AddressFileName(target) <> "管理簿.xlsm" Or Not IsDigits(year, 4) Or IsSameWorkbookAddress(source, target) Then Exit Function
    Set oldBook = OpenAddressWorkbook(source, True, False, message)
    If oldBook Is Nothing Then GoTo Finished
    Set page = oldBook.Worksheets(sourceSheet)
    headings = Array("共通番号", "署", "氏名", "申請書種別", "提出形式", "年分", "状態", "申請日", _
        "起案日", "決裁日", "施行日", "完了日", "メモ1", "メモ2", "メモ3", "発送文書1", "発送文書2", _
        "発送文書3", "決裁番号", "お問い合わせ番号", "フリガナ", "郵便番号", "住所", "電話番号")
    Set headers = CreateObject("Scripting.Dictionary")
    Set used = CreateObject("Scripting.Dictionary")
    Set seen = CreateObject("Scripting.Dictionary")
    lastCol = page.Cells(1, page.Columns.Count).End(xlToLeft).Column
    Set last = page.Cells.Find(What:="*", LookIn:=xlFormulas, LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious, MatchCase:=False, SearchFormat:=False)
    If last Is Nothing Or lastCol < 2 Then Err.Raise 5, , "旧管理簿の見出しを確認してください。"
    lastRow = last.Row
    oldValues = page.Range(page.Cells(1, 1), page.Cells(lastRow, lastCol)).Value2
    For col = 1 To lastCol
        If IsError(oldValues(1, col)) Then Err.Raise 5, , "見出しがExcelエラーです。"
        label = Trim$(Replace(CStr(oldValues(1, col)), "　", " "))
        Select Case label
            Case "MaskedNumber", "№", "Number（№）", "13桁番号": label = "旧13桁番号"
            Case "氏名(フリガナ)", "氏名（フリガナ）", "ふりがな": label = "フリガナ"
            Case "問い合わせ番号": label = "お問い合わせ番号"
        End Select
        If Len(label) > 0 Then
            If headers.Exists(label) Then Err.Raise 5, , "見出しが重複しています。" & label
            headers.Add label, col
        End If
    Next
    For offset = 0 To 23
        If headers.Exists(headings(offset)) Then
            mapping(offset) = headers(headings(offset))
            used.Add headings(offset), True
        End If
    Next
    If mapping(0) = 0 Then
        If Not headers.Exists("旧13桁番号") Or Not headers.Exists("枝番") Then Err.Raise 5, , "旧13桁番号と枝番が見つかりません。"
        masked = headers("旧13桁番号")
        branch = headers("枝番")
        used.Add "旧13桁番号", True
        used.Add "枝番", True
    End If
    If mapping(2) = 0 Then Err.Raise 5, , "氏名の見出しが見つかりません。"
    For Each header In headers.Keys
        If Not used.Exists(header) Then
            If Len(missing) > 0 Then missing = missing & "、"
            missing = missing & header
        End If
    Next
    ReDim output(1 To lastRow, 1 To 24)
    For offset = 0 To 23
        output(1, offset + 1) = headings(offset)
    Next
    count = 1
    For row = 2 To lastRow
        If Application.CountA(page.Range(page.Cells(row, 1), page.Cells(row, lastCol))) > 0 Then
            If mapping(0) > 0 Then
                If Not IsDigits(oldValues(row, mapping(0)), 18) Then Err.Raise 5, , row & "行の共通番号が18桁文字列ではありません。"
                number = CStr(oldValues(row, mapping(0)))
            Else
                If Not RestoreNumber(oldValues(row, masked), number) Or Not IsDigits(oldValues(row, branch), 5) Then _
                    Err.Raise 5, , row & "行の旧番号・枝番を確認してください。"
                number = number & CStr(oldValues(row, branch))
            End If
            If seen.Exists(number) Then Err.Raise 5, , row & "行の共通番号が同一年度内で重複します。"
            seen.Add number, True
            count = count + 1
            output(count, 1) = number
            For offset = 1 To 23
                If mapping(offset) > 0 Then
                    If IsError(oldValues(row, mapping(offset))) Then Err.Raise 5, , row & "行にExcelエラーがあります。"
                    output(count, offset + 1) = oldValues(row, mapping(offset))
                End If
            Next
        End If
    Next
    oldBook.Close False
    Set oldBook = Nothing
    Set newBook = OpenAddressWorkbook(target, False, False, message)
    If newBook Is Nothing Then GoTo Finished
    If newBook.FileFormat <> xlOpenXMLWorkbookMacroEnabled Then Err.Raise 5, , "移行先はマクロ有効形式で保存してください。"
    For Each existing In newBook.Worksheets
        If StrComp(existing.Name, year, vbTextCompare) = 0 Then Err.Raise 5, , "移行先年度が既にあります。"
    Next
    Application.EnableEvents = False
    Set page = newBook.Worksheets.Add(After:=newBook.Worksheets(newBook.Worksheets.Count))
    page.Name = year
    page.Range("A:X").NumberFormat = "@"
    page.Range("H:L").NumberFormat = "yyyy/mm/dd"
    page.Range("A1").Resize(count, 24).Value2 = output
    saveStarted = True
    newBook.Save
    If Not newBook.Saved Then Err.Raise 5, , "保存未確認"
    newBook.Close False
    Set newBook = Nothing
    Set newBook = OpenAddressWorkbook(target, True, False, message)
    If newBook Is Nothing Then Err.Raise 5, , message
    Set page = newBook.Worksheets(year)
    actual = page.Range("A1").Resize(count, 24).Value2
    For row = 1 To count
        For col = 1 To 24
            If IsError(actual(row, col)) Then Err.Raise 5, , "再読込にExcelエラーがあります。"
            If CStr(actual(row, col)) <> CStr(output(row, col)) Then Err.Raise 5, , "再読込した値が一致しません。"
        Next
    Next
    For row = 2 To count
        If Not IsDigits(actual(row, 1), 18) Then Err.Raise 5, , "共通番号が文字列として保存されませんでした。"
    Next
    message = (count - 1) & "件を移行し、再読込を確認しました。"
    If Len(missing) > 0 Then message = message & vbCrLf & "旧管理簿で保管する列：" & missing
    MigrateLedgerYear = True
Finished:
    On Error Resume Next
    If Not oldBook Is Nothing Then oldBook.Close False
    If Not newBook Is Nothing Then newBook.Close False
    Application.EnableEvents = previousEvents
    Exit Function
Failed:
    message = Err.Description
    If saveStarted Then message = "移行結果未確定。再送せず移行先の年度シートを確認してください。" & vbCrLf & message
    Resume Finished
End Function

Public Function IsDigits(ByVal value As Variant, ByVal size As Long) As Boolean
    Dim index As Long
    Dim text As String
    If VarType(value) <> vbString Then Exit Function
    text = CStr(value)
    If Len(text) <> size Then Exit Function
    For index = 1 To size
        If Mid$(text, index, 1) < "0" Or Mid$(text, index, 1) > "9" Then Exit Function
    Next
    IsDigits = True
End Function

Public Function RestoreNumber(ByVal masked As Variant, ByRef number As String) As Boolean
    Dim text As String
    Dim body As String
    Dim index As Long
    number = ""
    If VarType(masked) <> vbString Then Exit Function
    text = CStr(masked)
    If IsDigits(text, 13) Then
        number = text
        RestoreNumber = True
        Exit Function
    End If
    If Len(text) <> 20 Then Exit Function
    If Left$(text, 4) <> "MN1:" Or Mid$(text, 18, 1) <> ":" Then Exit Function
    body = Mid$(text, 5, 13)
    If Not IsDigits(body, 13) Then Exit Function
    For index = 13 To 1 Step -1
        number = number & CStr((CLng(Mid$(body, index, 1)) + 3) Mod 10)
    Next
    If Right$(text, 2) <> NumberChecksum(number) Then
        number = ""
        Exit Function
    End If
    RestoreNumber = True
End Function

Private Function NumberChecksum(ByVal number As String) As String
    Dim index As Long
    Dim total As Long
    For index = 1 To 13
        total = total + index * CLng(Mid$(number, index, 1))
    Next
    NumberChecksum = Format$(total Mod 97, "00")
End Function


Public Function IsHttpsAddress(ByVal address As String) As Boolean
    IsHttpsAddress = (StrComp(Left$(address, 8), "https://", vbTextCompare) = 0)
End Function

Public Function CanonicalAddress(ByVal address As String) As String
    Dim slash As Long
    Dim host As String
    Dim path As String
    Dim segment As Variant
    If Not IsHttpsAddress(address) Then
        If Mid$(address, 2, 2) = ":\" Or Left$(address, 2) = "\\" Then CanonicalAddress = Replace(address, "/", "\")
        Exit Function
    End If
    slash = InStr(9, address, "/")
    If slash = 0 Then
        host = Mid$(address, 9)
    Else
        host = Mid$(address, 9, slash - 9)
    End If
    If Len(host) = 0 Or InStr(host, ".") = 0 Or host Like "*[!A-Za-z0-9.-]*" Then Exit Function
    If InStr(address, "?") > 0 Or InStr(address, "\") > 0 Then Exit Function
    If slash = 0 Then
        CanonicalAddress = "https://" & LCase$(host)
        Exit Function
    End If
    path = DecodeAddressPath(Mid$(address, slash))
    If Len(path) = 0 Then Exit Function
    If InStr(path, "\") > 0 Or InStr(path, vbCr) > 0 Or InStr(path, vbLf) > 0 Or InStr(path, Chr$(0)) > 0 Then Exit Function
    If InStr(path, "/:") > 0 Or InStr(path, "//") > 0 Or InStr(1, path, "/_layouts/", vbTextCompare) > 0 Then Exit Function
    For Each segment In Split(path, "/")
        If segment = "." Or segment = ".." Then Exit Function
    Next
    If Right$(path, 1) = "/" Then path = Left$(path, Len(path) - 1)
    CanonicalAddress = "https://" & LCase$(host) & path
End Function

Private Function DecodeAddressPath(ByVal path As String) As String
    Dim i As Long
    Dim octet As Long
    Dim code As Long
    Dim pending As Long
    Dim minimum As Long
    Dim pair As String
    Dim result As String
    i = 1
    Do While i <= Len(path)
        pair = Mid$(path, i + 1, 2)
        If Mid$(path, i, 1) = "%" And Len(pair) = 2 And UCase$(pair) Like "[0-9A-F][0-9A-F]" Then
            octet = CLng("&H" & pair)
            If pending > 0 Then
                If octet < 128 Or octet > 191 Then Exit Function
                code = code * 64 + octet - 128
                pending = pending - 1
                If pending = 0 Then
                    If code < minimum Or code > &H10FFFF Or (code >= &HD800& And code <= &HDFFF&) Then Exit Function
                    If code > 65535 Then
                        code = code - 65536
                        result = result & ChrW(&HD800& + code \ 1024) & ChrW(&HDC00& + code Mod 1024)
                    Else
                        result = result & ChrW(code)
                    End If
                End If
            ElseIf octet < 128 Then
                '区切りをエンコードしたリンクは、別パスに解釈しない。
                If octet = 47 Or octet = 92 Then Exit Function
                result = result & ChrW(octet)
            ElseIf octet >= 194 And octet <= 223 Then
                code = octet - 192
                pending = 1
                minimum = 128
            ElseIf octet >= 224 And octet <= 239 Then
                code = octet - 224
                pending = 2
                minimum = 2048
            ElseIf octet >= 240 And octet <= 244 Then
                code = octet - 240
                pending = 3
                minimum = 65536
            Else
                Exit Function
            End If
            i = i + 3
        Else
            If pending > 0 Then Exit Function
            result = result & Mid$(path, i, 1)
            i = i + 1
        End If
    Loop
    If pending = 0 Then DecodeAddressPath = result
End Function

Public Function IsSameWorkbookAddress(ByVal first As String, ByVal second As String) As Boolean
    Dim leftAddress As String
    Dim rightAddress As String
    leftAddress = CanonicalAddress(first)
    rightAddress = CanonicalAddress(second)
    If Len(leftAddress) = 0 Or Len(rightAddress) = 0 Then Exit Function
    IsSameWorkbookAddress = (StrComp(leftAddress, rightAddress, vbTextCompare) = 0)
End Function

Public Function IsDirectWorkbookAddress(ByVal address As String) As Boolean
    Dim path As String
    Dim extension As String
    path = CanonicalAddress(address)
    If Len(path) = 0 Then Exit Function
    extension = LCase$(Mid$(path, InStrRev(path, ".") + 1))
    IsDirectWorkbookAddress = (extension = "xlsm" Or extension = "xlsx" Or extension = "xlsb" Or extension = "xls")
End Function

Public Function AddressFileName(ByVal address As String) As String
    Dim path As String
    path = Replace(CanonicalAddress(address), "\", "/")
    If Len(path) > 0 Then AddressFileName = Mid$(path, InStrRev(path, "/") + 1)
End Function

Public Function AddressFolder(ByVal address As String) As String
    Dim slash As Long
    slash = InStrRev(address, IIf(IsHttpsAddress(address), "/", "\"))
    If slash > 0 Then AddressFolder = Left$(address, slash - 1)
End Function

Public Function OpenAddressWorkbook(ByVal address As String, ByVal isReadOnly As Boolean, _
    ByVal allowTrustedMacros As Boolean, ByRef message As String) As Workbook
    Dim candidate As Workbook
    Dim book As Workbook
    Dim previousEvents As Boolean
    Dim previousSecurity As Long

    message = ""
    If Not IsDirectWorkbookAddress(address) Then
        message = "Excelファイルの直接URLまたは検証用の絶対パスを指定してください。"
        Exit Function
    End If
    For Each candidate In Application.Workbooks
        If IsSameWorkbookAddress(candidate.FullName, address) Or _
            StrComp(candidate.Name, AddressFileName(address), vbTextCompare) = 0 Then
            message = "対象または同名のブックを閉じてから操作してください。"
            Exit Function
        End If
    Next
    previousEvents = Application.EnableEvents
    previousSecurity = Application.AutomationSecurity
    On Error GoTo Failed
    Application.EnableEvents = False
    If Not allowTrustedMacros Then Application.AutomationSecurity = msoAutomationSecurityForceDisable
    Set book = Application.Workbooks.Open(Filename:=ExcelOpenAddress(address), UpdateLinks:=0, ReadOnly:=isReadOnly, _
        Notify:=False, AddToMru:=False, IgnoreReadOnlyRecommended:=True)
    If Not IsSameWorkbookAddress(book.FullName, address) Then
        message = "Excelが開いたブックのアドレスが設定と一致しません。直接URLを確認してください。"
    ElseIf Not isReadOnly And book.ReadOnly Then
        message = "読み取り専用です。編集権限と他の利用者の使用状況を確認してください。"
    ElseIf book.Date1904 Then
        message = "1904年日付システムのブックは対象外です。"
    Else
        If Not isReadOnly And IsHttpsAddress(address) Then
            'オートセーブが既に無効な環境では、書き込みを試みない。
            If book.AutoSaveOn Then book.AutoSaveOn = False
            If book.AutoSaveOn Then Err.Raise 5, , "自動保存を停止できません。"
        End If
        Set OpenAddressWorkbook = book
        Set book = Nothing
    End If
Finished:
    On Error Resume Next
    If Not book Is Nothing Then book.Close SaveChanges:=False
    Application.AutomationSecurity = previousSecurity
    Application.EnableEvents = previousEvents
    On Error GoTo 0
    Exit Function
Failed:
    message = "Excelで開けません。直接URL、Officeのサインイン、権限、使用状況を確認してください。エラー番号 " & Err.Number
    Resume Finished
End Function

Public Function ExcelOpenAddress(ByVal address As String) As String
    Dim i As Long
    Dim ch As String
    If Not IsHttpsAddress(address) Then
        ExcelOpenAddress = address
        Exit Function
    End If
    For i = 1 To Len(address)
        ch = Mid$(address, i, 1)
        If ch = "#" Then
            ch = "%23"
        ElseIf ch = "%" Then
            If Not UCase$(Mid$(address, i + 1, 2)) Like "[0-9A-F][0-9A-F]" Then ch = "%25"
        End If
        ExcelOpenAddress = ExcelOpenAddress & ch
    Next
End Function
